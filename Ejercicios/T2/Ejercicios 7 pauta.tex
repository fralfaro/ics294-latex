\section{Ejercicios:}


\begin{enumerate}
    \item En una regresión lineal simple, el coeficiente asociado a la variable
        \(X_1\) fue estimado en:
        \[
          \hat\beta_1 = 2.7,
          \qquad s.e.(\hat\beta_1) = 0.9,
          \qquad n = 250.
        \]
        Considere un nivel de significancia del 5\%.

        \begin{enumerate}[label=\alph*)]
          \item Plantee las hipótesis nula y alternativa para evaluar si
                \( \beta_1 \) es significativamente distinto de cero.
          \item Calcule el estadístico \( t \) de prueba, compare su valor
                con el valor crítico correspondiente y determine si se
                rechaza la hipótesis nula.
          \item Construya un intervalo de confianza al 95\% para
                \( \beta_1 \) e interprete su significado.
        \end{enumerate}

        \textbf{Solución.}
        \begin{enumerate}[label=\alph*)]
          \item Hipótesis a testear:
                \[
                  H_0: \beta_1 = 0
                  \qquad\text{versus}\qquad
                  H_1: \beta_1 \neq 0.
                \]
          \item El estadístico \( t \) es:
                \[
                  t
                  = \frac{\hat\beta_1 - 0}{s.e.(\hat\beta_1)}
                  = \frac{2.7}{0.9}
                  = 3.0.
                \]
                Con 248 grados de libertad (suponiendo pocos regresores)
                y \(n = 250\), podemos aproximar el valor crítico bilateral
                usando la normal estándar:
                \[
                  t_{2.5\%} \approx 1.96.
                \]
                Dado que \(|3.0| > 1.96\), \textbf{rechazamos} \( H_0 \).
                Existe evidencia suficiente para afirmar que \( \beta_1 \) es
                significativamente distinto de cero al 5\% de significancia.

          \item El intervalo de confianza al 95\% es:
                \[
                  \left[
                    \hat\beta_1 - t_{2.5\%} \times s.e.(\hat\beta_1),
                    \hat\beta_1 + t_{2.5\%} \times s.e.(\hat\beta_1)
                  \right]
                  =
                  \left[
                    2.7 - 1.96 \times 0.9,
                    2.7 + 1.96 \times 0.9
                  \right]
                  =
                  \left[
                    0.934, 4.466
                  \right].
                \]
                \textbf{Interpretación:} Con 95\% de confianza, se espera que
                el verdadero efecto de \( X_1 \) sobre la variable dependiente
                esté entre 0.934 y 4.466 unidades. Es decir, un aumento en
                \( X_1 \) está asociado, en promedio, a un aumento positivo y
                significativo en la variable dependiente.
        \end{enumerate}

\item Se tiene un modelo de regresión lineal de la forma:
$$y=\beta_0 + \beta_1 x_1+\beta_2x_2+\beta_3 x3 + u$$
los resultados de la regresión con 250 datos son los siguientes:
        \[
          \hat{\beta} = \begin{bmatrix} 3 \\ 9 \\ 5 \\2 \end{bmatrix},
        \]
y 
        \[
          (X'X)^{-1} =
          \begin{bmatrix}
1 & 0  & 0  &  0\\
0 & 1  & 0  & -1\\
0 & 0  & 2  & -1\\
0 & -1 & -1 &  1
\end{bmatrix},
          \quad \hat{\sigma}^2 = 2
        \]
\begin{enumerate}
    \item Plantee la hipótesis que testea que $\beta_1=4\beta_3$.

    \item ¿Se puede rechazar la hipótesis nula con la información que disponemos con una significancia del 5\%?

\end{enumerate}

\textbf{solución}
\begin{enumerate}
    \item $$H_0 : \beta_1-4\beta_3 = 0$$
$$H_1 : \beta_1-4\beta_3 \neq 0$$
\item el estadistico t esta dado por:
$$t = \frac{\beta_1-4\beta_3}{s.e.(\beta_1-4\beta_3)}$$
y $s.e.(\beta_1-4\beta_3)$ esta dado por:
$$s.e.(\beta_1-4\beta_3)=\sqrt{Var(\beta_1-4\beta_3)}=\sqrt{Var(\beta_1)+16Var(\beta_3)-8Cov(\beta_1,\beta_3)}$$
$$\sqrt{Var(\beta_1)+16Var(\beta_3)-8Cov(\beta_1,\beta_3)}=\sqrt{1+16\cdot 1-8 \cdot (-1)}=\sqrt{25}=5$$
por tanto 
$$t = \frac{9-4\cdot2}{5}=0.2$$
y con el valor $t_{1-\alpha/2,246}=1.96$. No se rechaza $H_0$, y $\beta_1=4\beta_3$. 
\end{enumerate}

\item Considere la siguiente regresión lineal múltiple estimada con \(n = 75\) observaciones:

\[
y \;=\; \beta_{0} + \beta_{1}x_{1} + \beta_{2}x_{2} + \beta_{3}x_{3} + \beta_{4}x_{4} + u.
\]

El modelo arroja un coeficiente de determinación
\[
R^2 = 0.62.
\]

El investigador sospecha que las variables \(x_{3}\) y \(x_{4}\) no aportan capacidad explicativa.  
Para contrastar su hipótesis estima el siguiente modelo:

\[
y \;=\; \gamma_{0} + \gamma_{1}x_{1} + \gamma_{2}x_{2} + e,
\qquad\text{obteniendo}\quad
R^2 = 0.55.
\]
Formule y realice un test de hipótesis que permita decidir, con un 5 \% de significancia, si las variables \(x_{3}\) y \(x_{4}\) son relevantes para el modelo. ($F_{0.95,(2,70)}=3.13$)

\textbf{Solución}

\\
{Relevancia conjunta de $x_3$ y $x_4$}

\textit{Hipótesis:}\quad
$H_0:\ \beta_3=\beta_4=0$ \quad vs. \quad $H_1:$ al menos uno $\neq 0$.

\textit{Estadístico $F$:}
\[
F \;=\; \frac{(R_c^2-R_r^2)/q}{(1-R_c^2)/(n-k_c-1)}
\;=\;
\frac{(0.62-0.55)/2}{(1-0.62)/(75-4-1)}
\;=\;
\frac{0.07/2}{0.38/70}
\;=\; 6.447.
\]

\textit{Regla de decisión (5\%):}\quad
$F_{0.95,(2,70)}=3.13$.

\textit{Conclusión:}\quad
Como $6.447>3.13$, se \textbf{rechaza} $H_0$. Las variables $x_3$ y $x_4$ son \textbf{relevantes}

\item Una empresa de ventas de neumaticos esta realizando un modelo de regresión lineal multiple para predecir las ventas del proximo año, para esto han utilizado el siguiente modelo:
$$Ventas = \beta_0 + \beta_1ADV + \beta_2PROD+\beta_3CLI + u$$
Donde $ADV$ son el gasto en propagandas, $PROD$ la producción del año anterior y $CLI$ la cantidad de clientes. Se sospecha que estas ventas son diferentes por Zonas geograficas, por lo que se estima un modelo para cada Zona: NORTE, CENTRO y SUR, obteniendo: 
$$Ventas_{g} = \beta_{0g} + \beta_{1g}ADV + \beta_{2g}PROD+\beta_{3g}CLI + u$$
para $g \in {(NORTE,CENTRO,SUR)}$. Usando como datos las 50 tiendas que tienen a nivel nacional se tienen los siguientes resultados:
\begin{center}
    \begin{tabular}{|c|c|}
    \hline
        Modelo & SSR \\
        \hline
        Base & 250 \\
        \hline
        Norte & 80 \\
        \hline
        Centro & 70 \\
        \hline
        Sur & 50 \\
        \hline
    \end{tabular}
\end{center}
Testee, con una significancia del 5\%, si las ventas por Zonas son diferentes. ($F_{0.95,(8,38)}=2.19$)

\textbf{Solución}

Modelo base: $SSR_r=250$; modelos por zona: $SSR_N=80$, $SSR_C=70$, $SSR_S=50$.
\[
\sum_g SSR_g = 80+70+50 = 200,\qquad
q = (k_g G - k_r) = (4\cdot 3 - 4) = 8,\qquad
\text{gl}_{\text{den}} = N - k_g G = 50 - 12 = 38.
\]

\textit{Estadístico $F$:}
\[
F \;=\; \frac{(SSR_r-\sum_g SSR_g)/q}{\left(\sum_g SSR_g\right)/\text{gl}_{\text{den}}}
\;=\;
\frac{(250-200)/8}{200/38}
\;=\;
\frac{50/8}{5.263}
\;=\; 1.188.
\]

\textit{Regla de decisión (5\%):}\quad
$F_{0.95,(8,38)}=2.19$.

\textit{Conclusión:}\quad
Como $1.188<2.19$, \textbf{no se rechaza} $H_0$. No hay evidencia de diferencias significativas en las ventas por zona al 5\%.

\end{enumerate}
