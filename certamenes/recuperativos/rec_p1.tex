\section{Formulario}
\begin{itemize}

\item El intervalo de confianza de $100(1-\alpha)\%$ para la media poblacional $\mu$ cuando $\sigma$ es \emph{\textbf{conocido}}:
$\color{red}{\bar{x} \pm z_{\frac{\alpha}{2}}\frac{\sigma}{\sqrt{n}}}$

\item El intervalo de confianza de $100(1-\alpha)\%$ para la diferencia de medias
$\mu_x-\mu_y$ es:
${\color{red}(\bar{X}-\bar{Y}) \pm z_{\frac{\alpha}{2}}\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}}$

\item  Estimadores MCO.
{\color{red}
$$\hat{\beta}_0 = \bar{y} - \hat{\beta}_1 \bar{x} \quad \quad   \hat{\beta}_1 = \frac{\sum_{i=1}^{n}(x_i - \bar{x})(y_i-\bar{y})}{\sum_{i=1}^{n}(x_i - \bar{x})^2}$$}
    
\item 
	{\color{red}	\[ SST = \sum_{i=1}^{n} (y_i - \bar{y})^2 \quad \quad  SSE = \sum_{i=1}^{n} (\hat{y}_i - \bar{y})^2 \quad \quad SSR = \sum_{i=1}^{n} \hat{u}_i^2  \]}
		
		
    \item Vector de estimadores:    
{\color{red}    $\hat{\beta}=(X^TX)^{-1}X^Ty$}

\item {\color{red}\[
A = \begin{bmatrix}
a & b \\
c & d
\end{bmatrix}
\qquad
A^{-1} = \frac{1}{det(A)} \begin{bmatrix}
d & -b \\
-c & a
\end{bmatrix}
\]}
\end{itemize}
