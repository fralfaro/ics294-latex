\section*{Formulario}
\begin{itemize}

\item Estadistico t para $\beta_j = a_j$

$$\color{red}{t_{\beta_j}=\frac{\hat{\beta_j}-a_j}{s.e.(\hat{\beta_j})}}$$

\item El intervalo de confianza de $100(1-\alpha)\%$ para el estimador $\beta_j$:

$${\color{red}\hat{\beta_j} \pm t_{\frac{\alpha}{2},n-k-1}s.e.(\hat{\beta_j})}$$


\item para el estadistico F entre dos modelos:

$$\color{red}{F=\frac{SSR_r-SSR_{ur}}{SSR_{ur}}\frac{n-k-1}{j}}$$
$$\color{red}{F=\frac{R^2_{ur}-R^2_{r}}{1-R^2_{ur}}\frac{n-k-1}{j}}$$

\item Test de Chow para Grupos.

    \[\color{red}
    F = \frac{(SSR_r - SSR_{ur}) / ((G - 1)(k + 1))}{SSR_{ur} / (n - G(k + 1))} \sim F((G - 1)(k + 1), n - G(k + 1))
    \]

Con

$$\color{red}SSR_{ur} = \sum_{g=1}^G SSR_g$$

\item Estadistico t para igualdad entre parametros:
 \[\color{red}t_{\text{estadístico}} = \frac{\hat{\beta}_1 - \hat{\beta}_2 - 0}{\sqrt{\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2)}}\]


\item  Varianza de una suma.
{\color{red}
$$\text{Var}(kA + cB) = k^2\text{Var}(A) + c^2\text{Var}(B) + 2kc\text{COV}(A, B)$$}

\item $t_{0.95,n} = 1.64$ cuando $n >50$.
\item $t_{0.975,n} = 1.96$ cuando $n >50$.
\item $t_{0.99,n}=2.33$ cuando $n>50$
