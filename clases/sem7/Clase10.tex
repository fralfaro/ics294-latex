\begin{frame}
\frametitle{Teorema de Gauss-Markov}
\begin{itemize}
    \item \textbf{Teorema:} Bajo los supuestos 1 a 5, el estimador de Mínimos Cuadrados Ordinarios (MCO) de $\beta$ es de mínima varianza entre los estimadores lineales insesgados de $\beta$.
    \item El estimador MCO es el estimador lineal insesgado de menor varianza (MELI / BLUE).
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Supuesto 6: Normalidad de $u$}
\begin{itemize}
    \item Las perturbaciones $u$ se distribuyen como $u \sim N(0, \sigma^2)$.
    \item Este supuesto es importante porque para realizar inferencia acerca de los $\beta$ y la construcción de intervalos de confianza se necesita la distribución de sus estimadores $\hat{\beta}$ (no basta con conocer su esperanza y varianza).
\end{itemize}
\end{frame}

\begin{frame}{Normalidad}
\begin{itemize}
    \item  Como u es la suma de muchos factores distintos no observados que afectan a y, se puede apelar
al teorema del límite central (TLC)  para concluir que u tiene una distribución
aproximadamente normal.

\item Se asume que todos los factores
no observados afectan a y por separado en forma aditiva.  Si u
es una  función no aditiva de los factores no observados, entonces el argumento del TLC no
puede emplearse.
\end{itemize}
\end{frame}
\begin{frame}{Supuestos del modelo lineal clásico (MLC)}

\begin{itemize}
\item Consisten en todos los supuestos de Gauss-Markov más el supuesto  del error
distribuido normalmente.
    \item Si se satisfacen los supuestos del MCL los estimadores de MCO son insesgados de varianza mínima, no es ya necesario restringir la
comparación a estimadores lineales en y.
\end{itemize}

{\color{blue} $$ y|X\sim N(\beta_0+\beta_1x_1+...+\beta_kx_k, \sigma^2)$$}
    
\end{frame}



\begin{frame}[shrink=10]
\frametitle{Distribución del estimador de MCO}
\begin{itemize}
    \item Teorema: bajo los supuestos 1 a 6 se cumple $\hat{\beta}_j|X \sim N(\beta_j, \text{Var}(\hat{\beta}_j|X))$.
    \item Esto implica que, condicional en las variables regresoras $X$,
    $$\frac{\hat{\beta}_j - \beta_j}{\sqrt{\text{Var}(\hat{\beta}_j|X)}} \sim N(0, 1).$$
    \item Pero, recuerden que no conocemos $\text{Var}(\hat{\beta}_j)$ porque no conocemos $\sigma^2$.
    \item En su lugar, vamos a usar la estimación: $\hat{\text{Var}}(\hat{\beta}_j)$.
    \item Nos queda que:
    $$\color{red}t_{\hat{\beta_j}}=\frac{\hat{\beta}_j - \beta_j}{\text{s.e.}(\hat{\beta}_j)} \sim t_{n-k-1}$$ 
    donde $\text{s.e.}(\hat{\beta}_j) = \sqrt{\hat{\text{Var}}(\hat{\beta}_j|X)}$.
    \item \textcolor{red}{El término en rojo va a ser clave para hacer inferencia y construir intervalos de confianza.}
\end{itemize}
\end{frame}

\begin{frame}[shrink=10]
\frametitle{Test de hipótesis sobre un parámetro individual}
\begin{itemize}
    \item Tenemos el modelo poblacional
    $$Y_i = \beta_0 + \beta_1X_{i1} + \ldots + \beta_kX_{ik} + u_i$$
    \item Hasta aquí, aprendimos a encontrar el estimador $\hat{\beta}$ de los $\beta$.
    \item Además, aprendimos a estimar su varianza y su función de distribución.
    \item Ahora queremos usar todo esto para testear hipótesis acerca del valor de $\beta$.
    
\end{itemize}
\end{frame}

\begin{frame}[shrink=10]
\frametitle{Test de hipótesis sobre un parámetro individual}
\begin{itemize}
    \item Suponga que quiere testear si el coeficiente $\beta_j$ que acompaña a $x_j$ es significativamente distinto al valor $a_j$. Donde $a_j$ es una constante (un número). Es decir:
    \begin{align*}
        H_0 &: \beta_j = a_j \\
        H_1 &: \beta_j \neq a_j
    \end{align*}
    \item {\color{red} Para el caso particular de $a_j = 0$, el test busca determinar si la variable $x_j$ es significativa o relevante para el modelo.

     Si se rechaza $H_0$, entonces $x_j$ es estadísticamente significativa.}
        
    \item Para el test se usa el siguiente estadístico $t$ construido bajo la hipótesis nula:
    $$t_{\hat{\beta}_j} = \frac{\hat{\beta}_j - a_j}{s.e.(\hat{\beta}_j)} \sim t_{n-k-1}$$
    \item $t_{\hat{\beta}_j}$ mide en cuántas desviaciones estándar estimadas de $\hat{\beta}_j$, se aleja $\hat{\beta}_j$ de $a_j$. Valores de $t_{\hat{\beta}_j}$ alejados de $a_j$ son evidencia en contra de la hipótesis nula.
\end{itemize}
\end{frame}

\begin{itemize}
    \item {\color{blue}Supongamos que se fija la probabilidad de cometer error tipo I en $\alpha$. Nivel de significancia
o probabilidad de rechazar $H_0$ cuando en realidad es verdadera.}
    \item {\color{red}Sea $t_{\alpha/2}$ el percentil $(1 - \frac{\alpha}{2})100$ de la distribución t-Student de $n - k - 1$ grados de libertad.}
    \item Si $\left|\frac{\hat{\beta}_j - a_j}{s.e.(\hat{\beta}_j)}\right| \geq t_{\alpha/2}$, se rechaza $H_0$.
\end{itemize}

\begin{frame}{Intervalos de Confianza para un Parámetro Individual}
    \begin{itemize}
        \item El intervalo de confianza al $(1-\alpha)100\%$ para $\beta_j$ es:
        \[
        \hat{\beta}_j - t_{\alpha/2} \text{s.e.}(\hat{\beta}_j) \leq \beta_j \leq \hat{\beta}_j + t_{\alpha/2} \text{s.e.}(\hat{\beta}_j)
        \]
        \item Esto nos indica que con un $(1 - \alpha)100\%$ de confianza, $\beta_j$ estará dentro de ese intervalo.
        \item Si $\beta_j$ no está en el intervalo, entonces se rechaza la hipótesis nula al $(\alpha 100\%)$ de significancia.
    \end{itemize}
\end{frame}

\begin{frame}
\frametitle{Ejemplo de Test de Hipótesis}


\begin{enumerate}
    \item \textbf{Paso 1:} Planteamos las hipótesis:
    \[
    H_0 : \beta_1 = 0 \quad \text{vs} \quad H_1 : \beta_1 \neq 0
    \]
    
    \item \textbf{Paso 2:} Calculamos el estadístico t:
    \[
    t_{\hat{\beta}_1} = \frac{42.0576 - 0}{6.5498} = 6.4212
    \]
    
    \item \textbf{Paso 3:} Regla $\Rightarrow$ rechazamos $H_0$ si $|6.42| > t_{2.5}$, con gl=932.
    
    \item \textbf{Paso 4:} Dado que $|6.42| \geq 1.96$, rechazamos $H_0$. Con un nivel de significatividad del $5\%$, concluimos que $\beta_1$ es significativamente diferente de cero.
\end{enumerate}
\end{frame}
\begin{frame}[shrink=10]{Ejemplo}

   
    \begin{itemize}
        \item Tomemos $\alpha 100 = 5\% \rightarrow t_{\alpha/2} =  1.96$
        \item Recordando la fórmula:

        \item Reemplazando:

        \]
        \item Entonces:
        \[

        \]
        \item Esto significa que con 95\% de confianza, $\beta_1$ está entre esos valores.
    \end{itemize}

\begin{frame}{Valor-p (p-value)}


    ¿Cuál es el menor nivel de
significancia al que se habría rechazado la hipótesis nula? 

{\color{blue}Este nivel se conoce como el valor-p
de la prueba.}

El valor-p es la probabilidad de observar un
estadístico t tan extremo como el que se encontró si la hipótesis nula es verdadera.

Esto significa
que valores-p pequeños son evidencia contra la hipótesis nula.



\end{frame}

\begin{frame}
\frametitle{Ejemplo: ecuación de salarios}
Se busca testear si hay retornos positivos a la experiencia.

\begin{itemize}
    \item Paso 1: $H_0: \beta_2 =0$, $H_1: \beta_2 > 0$
    \item Paso 2: El estadístico $t$ es $t_{\hat{\beta}_2} = \frac{0.0041}{0.0017} \approx 2.41$
    \item Paso 3: Comparar $t_{\hat{\beta}_2}$ contra $t_{0.05}$. En este caso hay $n - k - 1 = 522$ gl, por lo que podemos usar valores de  la normal. El valor crítico a un nivel de significancia del 5\% es 1.645.
    \item Paso 4: Por lo tanto, $2.41 > 1.645$ y se concluye que $\beta_2$ es estadísticamente mayor que 0 con un nivel de significancia del 5\%.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Ejercicio de cierre}

En el modelo de salarios se obtiene:
\[
\hat{\beta}_{educ}=0.075, \qquad se(\hat{\beta}_{educ})=0.025.
\]
Queremos testear \(H_0:\beta_{educ}=0\) contra \(H_1:\beta_{educ}\neq 0\), con \(\alpha=5\%\).

\begin{enumerate}
    \item Calcule el estadístico \(t\).
    \item Use el valor crítico aproximado \(1.96\). ¿Rechaza \(H_0\)?
    \item Interprete el resultado.
\end{enumerate}

\vspace{0.2cm}
\textbf{Respuesta:}
\[
t=\frac{0.075-0}{0.025}=3.
\]
Como \(|3|>1.96\), rechazamos \(H_0\). La educación tiene un efecto estadísticamente distinto de cero al 5\%.

\end{frame}

\begin{frame}
\frametitle{Conclusiones}
\begin{itemize}
    \item Hemos visto cómo se puede hacer tests de hipótesis simples sobre estimadores del MCO.
    \item El test más importante es determinar si el parámetro es significativamente diferente de cero.
    \item El estadístico \(t\) compara la distancia entre el estimador y el valor nulo en unidades de error estándar.
    \item Los intervalos de confianza y los valores-p entregan formas complementarias de reportar evidencia estadística.
   
\end{itemize}
\end{frame}
