\item Beta mediante MC2E:
\[
\hat\beta_1^{MC2E} \;=\; \frac{\operatorname{Cov}(\hat{X},Y)}{\operatorname{Var}(\hat{X})} \;=\; \frac{\sum_i (\hat{X}_i - \bar {\hat{X}})(Y_i - \bar Y)}{\sum_i (\hat{X}_i - \bar {\hat{X}})(\hat{X}_i - \bar {\hat{X}})}.
\]

\item Beta mediante IV:
$$\beta^{IV}=\frac{Cov(y,z)}{Cov(z,x)}=\frac{\sum (y_i - \bar{y})(z_i - \bar{z})}{\sum (x_i - \bar{x})(z_i - \bar{z})}$$

\item intercepto:
\[
\hat\beta_0 \;=\; \bar y \;-\; \hat\beta_1\,\bar{\hat{x}}
