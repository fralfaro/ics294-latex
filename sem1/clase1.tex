\section{Introducción}
\begin{frame}{¿Qué es econometría?}
 \begin{itemize}
     \item  Es una disciplina que emplea la teor\'ia econ\'omica y m\'etodos estad\'isticos para
contrastar distintas teor\'ias, pronosticar variables y evaluar pol\'iticas p\'ublicas
utilizando bases de datos.
\item  Es una aplicaci\'on del metodo cient\'ifico en econom\'ia. Nos permite refutar
hip\'otesis econ\'omicas.
\item  Nos permite contestar preguntas como por ejemplo:
\begin{enumerate}
    \item Cu\'al es la brecha salarial por g\'enero?
    \item Cu\'al es el efecto de haber completado la universidad en el salario?
\end{enumerate}

 \end{itemize}
 
\end{frame}


\begin{frame}{Econometría moderna}


  La econometr\'ia moderna se enfoca en contestar 3 tipos de preguntas:
  \begin{itemize}
      \item Estimar como la esperanza condicional de una variable depende de otra.
      \item Estimar el impacto causal de una variable sobre la otra.
      \item Predecir el comportamento de una variable hacia el futuro.
  \end{itemize}

\end{frame}

\begin{frame}{M\'etodo econom\'etrico}
    
\begin{itemize}
    \item Definir objetivo: qu\'e quiero estudiar?
    \item  Estructura:
    {\small
\begin{enumerate}
    \item {\color{blue} Modelo Econ\'omico:} modelos que explican el comportamiento de una variable económica en
funci\'on de otras.\\

 $Y = f (x_1; x_2; ...; x_k)$\\
Ejemplo 1: Funci\'on de producci\'on $Y = AK^{\alpha}L^{\beta}$\\
Ejemplo 2: Retornos educaci\'on\\
salario = f (educ; experiencia; ingfamiliar ; habilidad)
\item {\color{blue}Modelo econom\'etrico:} permite cuantificar, contrastar y evaluar el modelo
econ\'omico.
\begin{enumerate}
    \item Especificar la funcion f (:).
    \item Las variables econ\'omicas son aleatorias por lo que lo incorporamos con un termino de
error:\\ 
$y = f (x_1; x_2;...; x_k) + u$
\item Los datos. 
\end{enumerate} 

\end{enumerate}}
\end{itemize}

\end{frame}

\begin{frame}{Causalidad y correlación}

\begin{itemize}
    \item  Muchas veces nos interesa conocer el efecto causal de una variable sobre otra.
    \item  El concepto {\color{blue} ceteris paribus } es clave para la causalidad: efecto de X sobre Y
{\it manteniendo todos los demas factores relevantes constantes}.
 Muy dif\'icil en la pr\'actica en ciencias sociales.
 \item  Vamos a ver en el curso los supuestos necesarios y t\'ecnicas para
hacerlo.
\end{itemize}

\pause

\begin{alertblock}{Idea clave: correlación $\neq$ causalidad}
Que dos variables $X$ e $Y$ se muevan juntas (\textbf{correlación}) \emph{no} implica
que $X$ \textbf{cause} a $Y$. La asociación puede deberse a:
\begin{itemize}
    \item un \textbf{tercer factor} $Z$ que afecta a ambas (confusor),
    \item \textbf{causalidad inversa} ($Y$ causa a $X$), o
    \item pura \textbf{coincidencia} (correlación espuria).
\end{itemize}
\end{alertblock}

\end{frame}

\begin{frame}{Definiciones clave: esperanza condicional y causalidad}

\begin{block}{Definición: Esperanza condicional}
La \textbf{esperanza condicional} de $Y$ dado $X=x$, denotada $\mathbb{E}[Y \mid X=x]$,
es el valor promedio de $Y$ entre todas las unidades de la población que comparten el
valor $X=x$. La econometría descriptiva busca estimar cómo $\mathbb{E}[Y \mid X]$ cambia con $X$.
\end{block}

\pause

\begin{block}{Definición: Efecto causal (\textit{ceteris paribus})}
El \textbf{efecto causal} de $X$ sobre $Y$ es el cambio en $Y$ atribuible a una
variación de $X$, \emph{manteniendo constantes} todos los demás factores relevantes.
Formalmente, si $Y_i(x)$ es el resultado potencial de la unidad $i$ cuando $X=x$, el
efecto causal individual es $Y_i(x') - Y_i(x)$.
\end{block}

\centering
{\small \textit{Estimar $\mathbb{E}[Y\mid X]$ es fácil con datos; estimar el efecto causal requiere supuestos adicionales.}}

\end{frame}

\begin{frame}{Ejemplo}
\begin{itemize}
\item  Informaci\'on: En una base de individuos de 35 años de Chile, encuentro que las
personas que terminaron la universidad ganan 30\% m\'as que las que no la terminaron.

 \item Pregunta: De lo anterior, puedo concluir que ir a la universidad causa un
aumento del ingreso de los individuos?   
\end{itemize}
\end{frame}


\begin{frame}{Ejemplo}

No puedo concluir eso.
\begin{itemize}
    \item Ir a la universidad est\'a asociado con un mont\'on de otras caracter\'isticas que
tambien afectan ingreso.
\item Por ejemplo, la habilidad, el ingreso de los padres, etc.
\end{itemize}

\pause

\begin{block}{Definición: Sesgo de variable omitida}
Ocurre cuando una variable $Z$ que (i) \textbf{afecta a $Y$} y (ii) está
\textbf{correlacionada con $X$} se deja fuera del análisis. La asociación observada
entre $X$ e $Y$ mezcla entonces el efecto de $X$ con el de $Z$:
\[
\underbrace{\mathbb{E}[Y\mid X=1]-\mathbb{E}[Y\mid X=0]}_{\text{diferencia observada}}
= \underbrace{\text{efecto causal}}_{\text{lo que buscamos}} + \underbrace{\text{sesgo de selección}}_{\text{contaminación por }Z}.
\]
En el ejemplo, $X=$ ir a la universidad, $Z=$ habilidad: las personas más hábiles
estudian más \emph{y} ganan más, inflando la brecha del 30\%.
\end{block}

\end{frame}

    \begin{frame}{Tipos de Datos }
      

\begin{itemize}
    \item {\color{blue}Experimentales}: el investigador asigna en forma aleatoria la exposici\'on.
    \item {\color{blue} Pseudo-experimental}: el investigador controla la exposici\'on pero no
asigna en forma aleatoria.
\item {\color{blue}Observacional (o no-experimental)}: el investigador no controla la exposici\'on, depende de variables que el no controla.
\end{itemize}

{\color{blue} Rara vez usa datos experimentales
extremadamente caros, problemas éticos.}
    \end{frame}

    \begin{frame}{Tipos de datos}

   {\color{blue}Corte transversal: }Muestra de individuos, hogares, empresas, regiones, países, etc., recolectados en un tiempo dado.
A veces el estudio no se realiza exactamente en el mismo periodo de tiempo para todas las unidades evaluadas, pero diferencias temporales menores se ignoran.
Se asume que los datos fueron obtenidos por muestreo aleatorio (random sampling) desde la población subyacente.

        
    \end{frame}


\begin{frame}{Tipos de datos}

{\color{blue}Series de tiempo: }Observaciones de una o más variables en el tiempo.
Ej. Precios de acciones, dólar, IPC, ventas de automóviles, etc.
Eventos pasados influencian eventos futuros.
Rezagos en el comportamiento son prevalentes en ciencias sociales.
Tiempo es una dimensión importante en datos de series de tiempo.
Frecuencia $\Delta$ tiempo=1, día, año, semestre, trimestre, etc. 
A diferencia de datos de corte transversal, datos de series de tiempo no pueden asumirse independientes.

\end{frame}

\begin{frame}{Tipos de datos}

{\color{blue}Datos longitudinales o de panel: }Consiste en una serie de tiempo para cada miembro del set de datos.
Ejemplo: 10 años de datos de salario, educación y empleo sobre un set de individuos.
Diferencias con los anteriores es que los mismos individuos, firmas o países, son seguidos por varios años.

Varias ventajas:
Control de ciertas características no observadas de individuos, firmas, etc.
Facilita estudios de causa-efecto (inferencia causal).
Permite estudiar los efectos del rezago en el comportamiento o resultado.

\end{frame}

\section{Conceptos } 

\begin{frame}{Muestreo }

Los datos experimentales son obtenidos sobre conjuntos de individuos u objetos, sobre los cuales se quiere conocer algunas características. 
     
     Llamaremos {\bf unidad de observaci\'on} a estos individuos
y la totalidad de estas unidades de observación se denomina {\bf población.}

    El Muestreo es la disciplina que trata con el conjunto de m\'etodos, t\'ecnicas y
procedimientos para tomar u obtener una muestra particular  de la población a efecto de
realizar inferencias a partir de la misma.
%Existen dos grandes categor\'ias de muestreo: el Muestreo Probabil\'istico y el Muestreo No Probabil\'istico.

 
\end{frame}




 \begin{frame}{Muestreo}
     Se dice que una muestra es representativa de una poblaci\'on si toda unidad de observaci\'on podr\'a aparecer en la muestra, y esto con una probabilidad conocida.


  \begin{figure}
    \centering
    \includegraphics[width=0.5\linewidth]{sem1/image.png}
\end{figure}   
 \end{frame}


\begin{frame}{¿Por qué muestrear la población?}

    \begin{itemize}
        \item Imposibilidad física de chequear todos los ítems en la población.
        \item El costo de estudiar todos los elementos en la población.
        \item Los resultados del muestreo son normalmente adecuados.
    \end{itemize}
\end{frame}

\begin{frame}{Objetivos del muestreo}

{\color{blue} {\bf Estimar los parámetros de la población.}}

\begin{itemize}
\item Variable: lo que se desea medir en la muestra. Ej. Nivel de educación, nivel de ingresos, etc.
\item Parámetros:  valor numérico que describe una característica de una población completa. Promedio, desviación estándar, betas, etc.
\item Estadístico: calculado sobre la muestra.
\item Los parámetros normalmente son desconocidos, se estiman a través del estadístico.
\end{itemize}
\end{frame}

\begin{frame}{Parámetro vs.\ Estadístico (formalización)}

\begin{block}{Definición: Población y muestra}
La \textbf{población} es el conjunto completo de todas las unidades de observación de
interés (tamaño $N$, posiblemente infinito). Una \textbf{muestra} es un subconjunto de
$n$ unidades extraído de la población mediante algún procedimiento de muestreo.
\end{block}

\pause

\begin{block}{Definición: Parámetro vs.\ Estadístico}
Un \textbf{parámetro} es un valor \emph{fijo y generalmente desconocido} que describe a
la población (p.ej. la media poblacional $\mu$, la varianza $\sigma^2$, un coeficiente
$\beta_j$).\\[0.4em]
Un \textbf{estadístico} es una función de los datos de la \emph{muestra}, por lo tanto
es una \emph{variable aleatoria} (p.ej. $\bar{X}$, $S^2$, $\hat{\beta}_j$). Se usa para
\textbf{estimar} el parámetro:
\[
\mu \;\;\text{(parámetro)} \quad\longleftrightarrow\quad \bar{X}=\frac{1}{n}\sum_{i=1}^{n}X_i \;\;\text{(estadístico)}.
\]
\end{block}

\end{frame}

\begin{frame}{Muestreo aleatorio} 

\begin{itemize}
    \item {\bf  Muestreo aleatorio simple:} muestra seleccionada de manera que cada elemento o individuo de la población tenga las mismas posibilidades de que se le incluya.
    \item {\bf Muestreo aleatorio sistemático:} Los elementos o individuos de la población se ordenan de alguna manera. Se selecciona un punto aleatorio de inicio y posteriormente se elige cada k-esimo miembro de la población. 
A nivel técnico, el muestreo sistemático no crea una muestra verdaderamente aleatoria. Sólo la selección del primer elemento de muestreo sistemático es una selección de probabilidad. Una vez que el primer elemento es seleccionado, algunos de los elementos tendrán una probabilidad cero de selección.
\end{itemize}
 
\end{frame}
 

\begin{frame}{Muestreo aleatorio}
\begin{itemize}
    \item {\bf Muestreo aleatorio estratificado:} La población se divide en subgrupos denominados estratos y se selecciona al azar una muestra de cada estrato.
    \item {\bf  Muestreo por conglomerados (cluster):} La población se divide en conglomerados a partir de límites geográficos o de otra clase. Luego se seleccionan los conglomerados al azar y se toma una muestra aleatoria con elementos de cada grupo. 
\end{itemize}
 \end{frame}

\section{Ejemplo aplicado}

\begin{frame}{Ejemplo aplicado: la encuesta CASEN (Chile)}

\begin{block}{Contexto chileno}
La \textbf{CASEN} (Caracterización Socioeconómica Nacional, Ministerio de Desarrollo
Social) es una encuesta de \textbf{corte transversal} con diseño de muestreo
\textbf{probabilístico estratificado por conglomerados}: se estratifica por región y
zona (urbano/rural) y se seleccionan manzanas (conglomerados) y luego hogares.
\end{block}

\begin{itemize}
    \item \textbf{Población:} todos los hogares de Chile.
    \item \textbf{Muestra:} $\sim$200.000 personas encuestadas.
    \item \textbf{Parámetro:} ingreso autónomo promedio del país, $\mu$.
    \item \textbf{Estadístico:} promedio muestral $\bar{X}$ (con \emph{factores de expansión}).
\end{itemize}

\pause
\centering
{\small Otras fuentes oficiales: \textbf{INE} (Encuesta de Empleo, IPC) y
\textbf{Banco Central} (Cuentas Nacionales, PIB) producen \emph{series de tiempo}.}

\end{frame}


\begin{frame}[fragile]{Ejemplo aplicado: descripción de datos en R}

\begin{exampleblock}{Cargar CASEN y describir el ingreso (R)}
\begin{verbatim}
# Cargar datos de la encuesta CASEN
datos <- read.csv("casen.csv")

# Estadistica DESCRIPTIVA: resumen del ingreso
summary(datos$ingreso)          # min, media, mediana, max
sd(datos$ingreso)               # desviacion estandar

# Comparar ingreso promedio por nivel educativo
aggregate(ingreso ~ educacion, data = datos, FUN = mean)
\end{verbatim}
\end{exampleblock}

{\small La \textbf{media muestral} (estadístico) estima el ingreso promedio del país
(parámetro). La diferencia de medias por educación es \emph{correlación}, no
necesariamente el efecto causal de educar.}

\end{frame}


\begin{frame}[fragile]{Ejemplo aplicado: lo mismo en Python}

\begin{exampleblock}{Cargar CASEN y describir el ingreso (Python)}
\begin{verbatim}
import pandas as pd

# Cargar datos de la encuesta CASEN
datos = pd.read_csv("casen.csv")

# Estadistica DESCRIPTIVA
print(datos["ingreso"].describe())   # media, std, cuartiles

# Ingreso promedio por nivel educativo
print(datos.groupby("educacion")["ingreso"].mean())
\end{verbatim}
\end{exampleblock}

\end{frame}


\section{Ejercicios}

\begin{frame}{Ejercicio Resuelto 1}

\begin{block}{Ejercicio 1 — Clasificación de datos y muestreo}
Para cada estudio, indique (i) el \textbf{tipo de datos} (corte transversal, serie de
tiempo o panel) y (ii) el \textbf{diseño} (experimental u observacional):
\begin{enumerate}
    \item El PIB trimestral de Chile reportado por el Banco Central, 2010--2024.
    \item La CASEN 2022: ingreso de hogares chilenos en un mismo año.
    \item Un seguimiento a los mismos 500 egresados de la USM durante 10 años.
\end{enumerate}
\end{block}

\pause

\begin{exampleblock}{Solución}
\begin{enumerate}
    \item \textbf{Serie de tiempo}, observacional (una variable observada en el tiempo; el investigador no la controla).
    \item \textbf{Corte transversal}, observacional (muchas unidades, un mismo periodo).
    \item \textbf{Panel} (longitudinal), observacional (mismas unidades seguidas en el tiempo).
\end{enumerate}
\end{exampleblock}

\end{frame}


\begin{frame}{Ejercicio Resuelto 2}

\begin{block}{Ejercicio 2 — Media muestral (muestreo)}
Una población de 1.000 individuos tiene ingreso mensual con media $\mu=1.500$ y
desviación estándar $\sigma=300$. Se toma una \textbf{muestra aleatoria simple} de
$n=100$. Encuentre la media esperada y la desviación estándar de la media muestral
$\bar{X}$.
\end{block}

\pause

\begin{exampleblock}{Solución}
$\bar{X}$ es un \textbf{estadístico}; $\mu$ es el \textbf{parámetro}. Por propiedades del muestreo aleatorio:
\[
\mathbb{E}[\bar{X}] = \mu = 1.500, \qquad
\sigma_{\bar{X}} = \frac{\sigma}{\sqrt{n}} = \frac{300}{\sqrt{100}} = 30.
\]
La media muestral es \textbf{insesgada} (centrada en $\mu$) y su dispersión cae con $n$:
más datos $\Rightarrow$ estimación más precisa.
\end{exampleblock}

\end{frame}


\begin{frame}{Ejercicio Resuelto 3}

\begin{block}{Ejercicio 3 — Correlación vs.\ causalidad (sesgo de selección)}
Una empresa ofrece un curso de software. 4 trabajadores lo tomaron ($D=1$) y 4 no
($D=0$); $R=1$ marca a quienes fueron \emph{asignados al azar}. Salarios $Y$:
\begin{center}\small
\begin{tabular}{c|cccc|cccc}
id & 1 & 2 & 3 & 4 & 5 & 6 & 7 & 8\\ \hline
$Y$ & 950 & 900 & 880 & 820 & 780 & 740 & 700 & 650\\
$D$ & 1 & 1 & 1 & 1 & 0 & 0 & 0 & 0\\
$R$ & 1 & 1 & 1 & 0 & 1 & 1 & 0 & 0\\
\end{tabular}
\end{center}
Calcule la diferencia \emph{ingenua} y la diferencia usando solo los aleatorios ($R=1$). ¿Cuál es el sesgo?
\end{block}

\pause

\begin{exampleblock}{Solución}
Ingenua: $\bar{Y}_{D=1}-\bar{Y}_{D=0} = 887{,}5 - 717{,}5 = 170$.\\
Solo aleatorios: $\bar{Y}_{D=1,R=1}-\bar{Y}_{D=0,R=1} = 910 - 760 = 150 \approx$ efecto causal.\\
\textbf{Sesgo} $= 170 - 150 = 20$: la comparación ingenua \emph{sobreestima} porque
quienes se inscriben voluntariamente difieren (motivación, habilidad).
\end{exampleblock}

\end{frame}

\section{Resumen}

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 1}
\begin{enumerate}
    \item La \textbf{econometría} combina teoría económica y métodos estadísticos para
    estimar relaciones, evaluar causalidad y predecir.
    \item \textbf{Correlación $\neq$ causalidad}: el supuesto \textit{ceteris paribus}
    y el cuidado del \textbf{sesgo de variable omitida} son centrales.
    \item Los \textbf{tipos de datos} son corte transversal, series de tiempo y panel;
    el \textbf{diseño} puede ser experimental u observacional.
    \item El \textbf{muestreo} probabilístico (simple, sistemático, estratificado,
    conglomerados) permite hacer inferencia a la población.
    \item \textbf{Parámetro} (poblacional, fijo) vs.\ \textbf{estadístico} (muestral,
    aleatorio); la estadística \textbf{inferencial} generaliza a la población.
\end{enumerate}
\end{block}
