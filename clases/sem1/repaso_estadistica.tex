\begin{frame}
\frametitle{Estadística  Inferencial}

\begin{itemize}
\item  Utiliza los datos muestrales para \textit{\textbf{inferir}} sobre la población.
\item Requiere hacer supuesto sobre la población de interés. Uso de
las teoría de \textit{\textbf{probabilidades}}.
\end{itemize}
\end{frame}

\begin{frame}

\frametitle{INTERVALOS DE CONFIANZA (IC)}
\begin{itemize}
\item En lugar de ofrecer un valor único, se ofrece un rango de valores donde es más probable que el parámetro se encuentre.
\item Este rango de valores lo llamaremos Intervalo de confianza (IC).
\item Ventaja: da una idea de la precisión que se tiene al estimar.
\item Para construir un IC hay que decidir previamente el nivel de confianza que se quiere tener.
\end{itemize}
\end{frame}



\begin{frame}{Teorema central del l\'imite }
    \begin{itemize}
% 
    \item La distribución muestral del promedio es la distribución de probabilidad de todos los posibles promedios de muestras de un determinado tamaño ($n$), seleccionadas desde una población.
        \item Para una población con media $\mu$ y varianza $\sigma^2$ la distribución muestral de los promedios de todas las posibles muestras de tamaño n generadas desde la población serán, aproximadamente, normalmente distribuidas.
        
\item La media de la distribución muestral será igual a $\mu$ y la varianza  igual a $\sigma^2/n$.
    \end{itemize}
\end{frame}

\begin{frame}{Supuestos TCL}
    

\begin{enumerate}
    \item \textbf{Muestreo aleatorio:} Las observaciones deben ser i.i.d. $(X_1, X_2, \ldots, X_n)$.
    
    \item \textbf{Tamaño muestral adecuado (regla de dedo):} 
    $$n \text{ debe ser } \begin{cases}
        \geq 30 & \text{para distribuciones simétricas (normal)} \\
        \geq 50 & \text{para distribuciones  asimétricas} \\
        \geq 100 & \text{para distribuciones muy asimétricas} 
    \end{cases}$$
    
    \item \textbf{Varianza finita:} $\sigma^2 < \infty$
    
    \item \textbf{Media finita:} $\mu < \infty$
\end{enumerate}
\end{frame}

\item $\sigma^2$ es la varianza poblacional (supongamos que es conocida por el momento)

\end{itemize}

\begin{frame}
\frametitle{OTROS NIVELES DE CONFIANZA}
Si $\alpha$ es un número entre 0 y 1. El intervalo de confianza de $100(1-\alpha)\%$ para la media poblacional $\mu$ cuando $\sigma$ es conocido se obtiene de la siguiente forma:

\begin{itemize}
\item Se identifica: $z_{\frac{\alpha}{2}}$
% \item Luego: $$P(z_{\frac{\alpha}{2}}\leq \frac{\bar{X}-\mu}{\frac{\sigma}{\sqrt{n}}}\leq -z_{\frac{\alpha}{2}})=1-\alpha$$
\item Finalmente el IC de $100(1-\alpha)\%$ es: $$\color{red}{\bar{x} \pm z_{\frac{\alpha}{2}}\frac{\sigma}{\sqrt{n}}}$$
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA: NIVEL DE CONFIANZA Y PRECISIÓN}

Si tomamos como estimador puntual $\bar{X}$

\begin{itemize}
\item Con un 95\% de confianza el verdadero valor del parámetro $\mu$ no estará mas lejos de:

$$E_{95}=(1.96)\frac{\sigma }{\sqrt{n}}$$

\item Con un 99\% de confianza el verdadero valor del parámetro $\mu$ no estará mas lejos de:

$$E_{99}=(2.58)\frac{\sigma }{\sqrt{n}}$$
\item Llamaremos a $E$ \emph{\textbf{error absoluto}} de la estimación
%\item Llamaremos $ER=E/\bar{X}$ \emph{\textbf{error relativo}} de la estimación

\end{itemize}


\end{frame}

%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

\begin{frame}
\frametitle{ NIVEL DE CONFIANZA Y PRECISIÓN.}


\begin{itemize}
\item Si para una misma muestra aumentamos el nivel de confianza
(por ejemplo de 95\% a 99\%) lograremos una mejora en la
\textbf{confiabilidad}... pero se pierde \textbf{precisión} (mayor error absoluto)
\begin{itemize}
\item el error absoluto ($E_{95}$) del IC de {\color{blue}{95\%}}:
$${\color{blue}{E_{95}}}=(1.96)\frac{\sigma }{\sqrt{n}} $$
\item el error absoluto ($E_{99}$) del IC de {\color{blue}{99\%}}:
$${\color{blue}{E_{99}}}=(2.58)\frac{\sigma }{\sqrt{n}} $$
\item confiabilidad $(1-\alpha)$ $\uparrow \Rightarrow$ precisión  $\downarrow$
\end{itemize}
\end{itemize}
\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA: DETERMINACIÓN DEL TAMAÑO
MUESTRAL}

\begin{itemize}
\item Antes de realizar un muestreo se puede definir el nivel de
confianza y precisión (error absoluto) deseados y luego
determinar el tamaño muestral ($n$)
\item suponiendo $\sigma$ conocido y eligiendo $E$ (el error absoluto) y $\alpha$:
$${\color{blue}{E}}=-z_{\frac{{\color{blue}{\alpha}}}{2}}\sigma/\sqrt{{\color{red}{n}}}$$
\item luego, despejando $n$:
$$n=(z_{\frac{\alpha}{2}}\frac{\sigma}{E})^2$$
\item si el valor resultante no es entero se redondea hacia arriba.
\end{itemize}
\end{frame}

\section{Proporciones} 

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA PARA LAS PROPORCIONES}

\begin{itemize}

\item Sea $\bf{p}$ la proporción de \emph{éxitos} en una población.
\item \emph{\textbf{éxito:}} identifica a un individuo de una población con una
determinada característica.
\item ejemplos:
\begin{itemize}
\item proporción de fallas en una población de un
determinado artefacto.
\item proporción de enfermos en una población.
\item proporción de aprobados en un curso
\end{itemize}
\end{itemize}

\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA PARA LAS PROPORCIONES }
\begin{itemize}
\item Recordemos que la v.a. :
$$X\equiv \text{número de éxitos en una muestra de tamaño } n $$
Distribuye $Bin(n,p)$
\item Además si $n\geq 30$
$$X\stackrel{aprox.}\sim N(np,np(1-p))$$
\item Se pedirá adicionalmente que $np>5$ y $n(1-p)>5$
\end{itemize}
\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA PARA LAS PROPORCIONES }

\begin{itemize}
\item Un buen estimador puntual para $p$ es $\hat{p}=\frac{X}{n}$ la
fracción de éxitos en la muestra. 
\item Ahora: $$\hat{p}\stackrel{aprox.}\sim N(p,\frac{p(1-p)}{n})$$
y
$$\frac{\hat{p}-p}{\sqrt{\frac{p(1-p)}{n}}}\stackrel{aprox.}\sim N(0,1)$$
\end{itemize}
\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA PARA LAS PROPORCIONES }

Entonces se puede emplear la distribución normal para armar un IC
para las proporciones. En este caso:

$$P(z_{\frac{\alpha}{2}}\leq \frac{\hat{p}-p}{\sqrt{\frac{p(1-p)}{n}}}\leq -z_{\frac{\alpha}{2}} )\simeq 1-\alpha$$

%%Si $n$ es suficientemente grande, se puede estimar la varianza
%mediante:
%$$\frac{\hat{p}(1-\hat{p})}{n}$$
\end{frame}

%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA PARA LAS PROPORCIONES }

Es decir:
$$P(z_{\frac{\alpha}{2}}\leq \frac{\hat{p}-p}{\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}}\leq -z_{\frac{\alpha}{2}} )\simeq 1-\alpha$$

$$P(\hat{p}+z_{\frac{\alpha}{2}}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}\leq p \leq \hat{p}-z_{\frac{\alpha}{2}}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}} )\simeq 1-\alpha$$

El IC se construye como:
$$\color{red}{IC_{(1-\alpha)100\%}=\hat{p}\pm z_{\frac{\alpha}{2}}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}}$$
\end{frame}

%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{IC PARA LAS PROPORCIONES: EJEMPLO}
\textbf{
Un fabricante asegura, a una compañía que le compra un producto en
forma regular, que el porcentaje de productos defectuosos es 5\%. La compañía decide comprobar la afirmación del
fabricante seleccionando de su inventario 200 unidades del producto
y probándolas. En este procedimiento se detectan 19 unidades
defectuosas en la muestra. ¿Es coherente este resultado con la
afirmación del proveedor? Utilice una confiabilidad de 95\%.
}\end{frame}

%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{IC PARA LAS PROPORCIONES. RESPUESTA DE EJEMPLO}

$$\hat{p}=\frac{19}{200}=0.095$$
$$IC_{95\%}=\hat{p}\pm z_{0.025}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}$$
$$IC_{95\%}=0.095 \pm 1.96 \sqrt{\frac{0.095(1-0.095)}{200}}$$
$$IC_{95\%}=(0.05433;0.1356)$$
El proveedor afirma que $p$ es $0.05$. Este
valor no cae dentro del IC.

\begin{frame}
\frametitle{IC PARA LAS PROPORCIONES: DETERMINACIÓN DEL TAMAÑO
MUESTRAL}

\begin{itemize}
\item Aquí también se puede determinar $n$ fijando el nivel de
confianza y el largo del intervalo.
\item se requiere una estimación inicial de $p$, $p^*$.
\item Para obtener $p^*$ se puede extraer una muestra preliminar (más
pequeña), o usar $p^*=0.5$ (peor escenario).
\item En este caso el error absoluto es:
$$E=-z_{\frac{\alpha}{2}}\sqrt{\frac{p^*(1-p^*)}{n}}$$
\item luego, despejando $n$:
$$n=\frac{z_{\frac{\alpha}{2}}^2p^*(1-p^*)}{E^2}$$
\end{itemize}
\end{frame}

\section{Diferencia de media}

\begin{frame}
\frametitle{INTERVALOS DE CONFIANZA\\DIFERENCIA DE MEDIA DE DOS POBLACIONES}

Se cree que el salario promedio de recién egresados de la carrera de
ingeniería industrial es mayor que el salario promedio de recién
egresados de arquitectura. Se obtuvieron muestras aleatorias de cada
grupo de egresados:
\begin{itemize}
\item $X_1,....,X_{n_x}$: salarios de egresados de ing. industrial
\item $Y_1,....,Y_{n_y}$: salarios de egresados de arquitectura
\item Se supone que las muestras son \textbf{independientes} entre si
\end{itemize}

\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{DIFERENCIA DE MEDIA DE DOS POBLACIONES}


Llamaremos:
\begin{itemize}
\item $\mu_x$: media poblacional de salarios de egresados de ing. industrial
\item $\mu_y$: media poblacional de salarios de egresados de arquitectura
\end{itemize}

\end{frame}



%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{DIFERENCIA DE MEDIA DE DOS POBLACIONES}
La idea es armar un IC para: $$\mu_x-\mu_y$$
\begin{itemize}
\item Si el cero cae dentro del intervalo, los datos no dan evidencia
de diferencia entre las medias
\item Si el cero cae fuera del intervalo, los datos  dan evidencia
de diferencia entre las medias

\end{itemize}

\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------

%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{CASO $\sigma_x^2$ y $\sigma_y^2$ CONOCIDOS, CON $n_x$ y $n_y$ GRANDES}

\begin{itemize}
\item $\bar{X} \stackrel{aprox.}\sim N(\mu_x,\frac{\sigma_x^2}{n_x})$
\item $\bar{Y} \stackrel{aprox.}\sim N(\mu_y,\frac{\sigma_y^2}{n_y})$
\item $\bar{X}-\bar{Y}$ también es aproximadamente normal
\item $E(\bar{X}-\bar{Y})=E(\bar{X})-E(\bar{Y})=\mu_x-\mu_y$
\item
$V(\bar{X}-\bar{Y})=V(\bar{X})+V(\bar{Y})=\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}$
(porque las muestras son independientes)
\end{itemize}

\end{frame}

%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{}

Por ello,
$$Z=\frac{(\bar{X}-\bar{Y})-(\mu_x-\mu_y)}{\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}} \stackrel{aprox.}\sim N(0,1)$$
\end{frame}



%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{}

De aquí surge que:

$$P(z_{\frac{\alpha}{2}}\leq Z\leq -z_{\frac{\alpha}{2}}) \simeq 1-\alpha$$

$$P(z_{\frac{\alpha}{2}}\leq \frac{(\bar{X}-\bar{Y})-(\mu_x-\mu_y)}{\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}}\leq -z_{\frac{\alpha}{2}}) \simeq 1-\alpha$$

$P((\bar{X}-\bar{Y}) + z_{\frac{\alpha}{2}}\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}\leq \mu_x-\mu_y\leq (\bar{X}-\bar{Y})-z_{\frac{\alpha}{2}}\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}})$ $ \simeq 1-\alpha$

\end{frame}



%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{}

El IC de $(1-\alpha)100\%$ para la diferencia de medias
$\mu_x-\mu_y$ es:

$$\color{red}{IC_{(1-\alpha)100}=(\bar{X}-\bar{Y}) \pm z_{\frac{\alpha}{2}}\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}}$$

\end{frame}


%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{EJEMPLO}

En el ejemplo de salarios. Supongamos que:
\begin{itemize}
\item $\bar{X}=\$800000$
\item $\bar{Y}=\$680000$
\item $n_x=144$
\item $n_y=121$
\item $\sigma_x=60000$
\item $\sigma_y=50000$
\end{itemize}
Obtener un IC de 95\% para la diferencia de medias.
\end{frame}

%------------------------------------------------------------------------------------------------------------------------------------------------
%------------------------------------------------------------------------------------------------------------------------------------------------
\begin{frame}
\frametitle{}

El IC de 95\% es:
$$IC_{95}=(\bar{X}-\bar{Y}) \pm z_{0.025}\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}$$
