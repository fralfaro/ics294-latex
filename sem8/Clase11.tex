\begin{frame}{Objetivos de la Clase}

Al finalizar esta clase, el estudiante será capaz de:

\begin{itemize}
    \item[\checkmark] Formular una hipótesis sobre \textbf{una restricción lineal} de los parámetros del modelo (igualdad $\beta_1=\beta_2$ y combinaciones como $\beta_1+\beta_2=1$).
    \item[\checkmark] Derivar y calcular la \textbf{varianza de una combinación lineal} de estimadores MCO, $\text{Var}(c'\hat{\beta})$, usando la matriz de varianzas--covarianzas.
    \item[\checkmark] Construir el \textbf{estadístico $t$} para una restricción lineal y aplicar la regla de rechazo con $t_{n-k-1}$.
    \item[\checkmark] Aplicar el \textbf{método de reparametrización} ($\theta=\beta_1-\beta_2$) para reescribir el modelo y testear con un $t$ usual.
    \item[\checkmark] Construir un \textbf{intervalo de confianza} para una combinación lineal de parámetros.
\end{itemize}

\end{frame}

\begin{frame}{Introducción}
    \begin{itemize}
        \item La clase pasada, vimos cómo hacer un test de hipótesis sobre un estimador específico del MCO ($H_0:\beta_j=a_j$).
        \item Pero a veces, no es eso lo que buscamos sino respuestas a hipótesis más complejas.
            \begin{itemize}
                \item ¿Es el coeficiente sobre la variable A igual al coeficiente para la variable B? ($\beta_1=\beta_2$)
                \item ¿La elasticidad del gasto propio compensa exactamente la del rival? ($\beta_1+\beta_2=0$)
                \item ¿Tienen rendimientos constantes a escala? ($\beta_1+\beta_2=1$)
            \end{itemize}
        \item \textbf{Esta clase:} probar \textit{una} restricción lineal sobre los parámetros.
        \item \textbf{Próxima clase:} probar \textit{varias} restricciones simultáneas $\Rightarrow$ prueba $F$.
    \end{itemize}
\end{frame}

\begin{frame}{El problema general: una restricción lineal}
\begin{block}{Definición: restricción lineal de parámetros}
Una \textbf{restricción lineal} sobre $\beta=(\beta_0,\beta_1,\dots,\beta_k)'$ tiene la forma
\[
H_0:\ c'\beta = q, \qquad\text{es decir}\qquad c_0\beta_0 + c_1\beta_1 + \cdots + c_k\beta_k = q,
\]
donde $c=(c_0,\dots,c_k)'$ es un vector de constantes conocidas y $q$ es un escalar conocido.
\end{block}
\pause
\begin{itemize}
    \item Casos particulares de esta clase:
    \begin{itemize}
        \item Igualdad: $\beta_1=\beta_2 \;\Leftrightarrow\; \beta_1-\beta_2=0$ \quad ($c=(0,1,-1,0,\dots)'$, $q=0$).
        \item Suma: $\beta_1+\beta_2=1$ \quad ($c=(0,1,1,0,\dots)'$, $q=1$).
        \item Proporción: $\beta_1=2\beta_2 \;\Leftrightarrow\; \beta_1-2\beta_2=0$.
    \end{itemize}
    \item Como es \textbf{una sola} restricción, podemos resolverla con un estadístico $t$ (no necesitamos $F$ todavía).
\end{itemize}
\end{frame}

\section{Igualdad de parámetros}

\begin{frame}{Hipótesis de igualdad de parámetros}
    \begin{itemize}
        \item Imaginemos que queremos testear: $H_0: \beta_1 = \beta_2$.
        \item Planteamos el test como una resta:

        $H_0: \beta_1 - \beta_2 = 0$.

        $H_1: \beta_1 - \beta_2 \neq 0$
        \item Dado que los $\hat{\beta}$ son distribuidos como normales (bajo los supuestos del MLN), la diferencia también es una normal.
        \item Se puede, entonces, construir un test $t$ con el $t$-estadístico:
        \[t_{\text{estadístico}} = \frac{\hat{\beta}_1 - \hat{\beta}_2 - 0}{\sqrt{\hat{\text{Var}}(\hat{\beta}_1 - \hat{\beta}_2)}}\]
        \item El único ingrediente \textbf{nuevo} es: $\hat{\text{Var}}(\hat{\beta}_1 - \hat{\beta}_2)$.
    \end{itemize}
\end{frame}

\begin{frame}{Recordatorio: propiedades de la varianza}
\begin{block}{Propiedades (sean $k,c$ constantes y $A,B$ variables aleatorias)}
\begin{enumerate}
    \item $\text{Var}(kA) = k^2\,\text{Var}(A)$
    \item $\text{Var}(kA + cB) = k^2\text{Var}(A) + c^2\text{Var}(B) + 2kc\,\text{Cov}(A, B)$
\end{enumerate}
\end{block}
\pause
\begin{itemize}
    \item La propiedad clave es la (2): la varianza de una combinación lineal \textbf{no} es la simple suma de varianzas; aparece el término de \textbf{covarianza}.
    \item Los estimadores $\hat{\beta}_1$ y $\hat{\beta}_2$ \textbf{están correlacionados} (provienen de la misma muestra y regresores correlacionados), por lo que $\text{Cov}(\hat{\beta}_1,\hat{\beta}_2)\neq 0$ en general.
    \item Ignorar la covarianza es el error más frecuente en este tipo de test.
\end{itemize}
\end{frame}

\section{Varianza de una combinación lineal}

\begin{frame}[allowframebreaks]{Demostración: varianza de una combinación lineal $\text{Var}(c'\hat{\beta})$}
\begin{alertblock}{Proposición}
Sea $\hat{\beta}$ el estimador MCO con matriz de varianzas--covarianzas $\text{Var}(\hat{\beta})=\sigma^2(X'X)^{-1}$, y sea $c$ un vector de constantes. Entonces
\[
\text{Var}(c'\hat{\beta}) = c'\,\text{Var}(\hat{\beta})\,c = \sigma^2\, c'(X'X)^{-1}c.
\]
En el caso $c'\hat{\beta}=\hat{\beta}_1-\hat{\beta}_2$:
\[
\text{Var}(\hat{\beta}_1-\hat{\beta}_2) = \text{Var}(\hat{\beta}_1) + \text{Var}(\hat{\beta}_2) - 2\,\text{Cov}(\hat{\beta}_1,\hat{\beta}_2).
\]
\end{alertblock}

\begin{exampleblock}{Demostración}
\textbf{Paso 1.} Por definición de varianza de un escalar aleatorio,
\[
\text{Var}(c'\hat{\beta}) = \mathbb{E}\!\left[\big(c'\hat{\beta} - c'\mathbb{E}[\hat{\beta}]\big)^2\right]
= \mathbb{E}\!\left[\big(c'(\hat{\beta}-\mathbb{E}[\hat{\beta}])\big)^2\right].
\]

\textbf{Paso 2.} Como $c'(\hat{\beta}-\mathbb{E}[\hat{\beta}])$ es un escalar, es igual a su transpuesta, y su cuadrado es
\[
\big(c'(\hat{\beta}-\mathbb{E}[\hat{\beta}])\big)^2 = c'(\hat{\beta}-\mathbb{E}[\hat{\beta}])(\hat{\beta}-\mathbb{E}[\hat{\beta}])'c.
\]

\textbf{Paso 3.} Tomando esperanza y sacando las constantes $c$ fuera del operador,
\[
\text{Var}(c'\hat{\beta}) = c'\,\underbrace{\mathbb{E}\!\left[(\hat{\beta}-\mathbb{E}[\hat{\beta}])(\hat{\beta}-\mathbb{E}[\hat{\beta}])'\right]}_{=\ \text{Var}(\hat{\beta})}\,c = c'\,\text{Var}(\hat{\beta})\,c.
\]

\framebreak

\textbf{Paso 4.} Sustituyendo $\text{Var}(\hat{\beta})=\sigma^2(X'X)^{-1}$:
\[
\text{Var}(c'\hat{\beta}) = \sigma^2\, c'(X'X)^{-1}c.
\]

\textbf{Paso 5 (caso $\hat{\beta}_1-\hat{\beta}_2$).} Aquí $c=(0,1,-1,0,\dots,0)'$. Llamando $V=\text{Var}(\hat{\beta})$, el producto $c'Vc$ selecciona el bloque $2\times 2$ correspondiente a $(\beta_1,\beta_2)$:
\[
\begin{pmatrix}1 & -1\end{pmatrix}
\begin{pmatrix} V_{11} & V_{12}\\ V_{12} & V_{22}\end{pmatrix}
\begin{pmatrix}1\\ -1\end{pmatrix}
= V_{11} - 2V_{12} + V_{22},
\]
es decir, usando la propiedad de la varianza con $k=1,\ c=-1$:
\[
\text{Var}(\hat{\beta}_1-\hat{\beta}_2) = \text{Var}(\hat{\beta}_1) + \text{Var}(\hat{\beta}_2) - 2\,\text{Cov}(\hat{\beta}_1,\hat{\beta}_2).
\]
\hfill $\square$
\end{exampleblock}
\end{frame}

\begin{frame}[shrink=10]{De dónde salen las varianzas y covarianzas}
    \begin{itemize}
        \item Entonces, en este caso, usando propiedades de la varianza ($k = 1$, $c = -1$):
        \[\hat{\text{Var}}(\hat{\beta}_1 - \hat{\beta}_2) = \textcolor{red}{\hat{\text{Var}}(\hat{\beta}_1) + \hat{\text{Var}}(\hat{\beta}_2)} - \textcolor{blue}{2\hat{\text{Cov}}(\hat{\beta}_1, \hat{\beta}_2)}\]
        \item Recuerden la matriz: $\hat{\text{Var}}(\hat{\beta})= (X^{\top}X)^{-1}\hat{\sigma}^2$.
        \begin{enumerate}
            \item \textcolor{red}{$\hat{\text{Var}}(\hat{\beta}_j) = \hat{\sigma}^2_{j}$ es el elemento $(j + 1), (j + 1)$ de esa matriz (diagonal principal).}
            \item \textcolor{blue}{$\hat{\text{Cov}}(\hat{\beta}_j, \hat{\beta}_l) = \hat{\sigma}_{jl}$ es el elemento $(j + 1), (l + 1)$ de esa matriz.}
        \end{enumerate}
        \item \textbf{Regla de rechazo:} una vez construido el $t$-estadístico, se rechaza la hipótesis nula si $|t_{\text{estadístico}}| > t_{\alpha/2,\,n-k-1}$, donde $\alpha$ es el nivel de significancia requerido.
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Demostración: distribución del estadístico $t$}
\begin{alertblock}{Teorema}
Bajo los supuestos del modelo lineal normal (MLN), para la restricción $H_0: c'\beta=q$,
\[
t = \frac{c'\hat{\beta} - q}{\widehat{\text{ee}}(c'\hat{\beta})} \;\sim\; t_{\,n-k-1}\qquad\text{bajo }H_0,
\]
donde $\widehat{\text{ee}}(c'\hat{\beta})=\sqrt{\hat{\sigma}^2\,c'(X'X)^{-1}c}$. En particular, para $\hat{\beta}_1-\hat{\beta}_2$:
\[
t=\frac{\hat{\beta}_1-\hat{\beta}_2}{\sqrt{\hat{\text{Var}}(\hat{\beta}_1-\hat{\beta}_2)}}\sim t_{\,n-k-1}.
\]
\end{alertblock}

\begin{exampleblock}{Demostración (sketch)}
\textbf{Paso 1 (numerador, normal estándar).} Bajo $H_0$, $c'\hat{\beta}$ es una combinación lineal de normales, luego es normal con media $c'\beta=q$ y varianza $\sigma^2 c'(X'X)^{-1}c$. Estandarizando:
\[
Z=\frac{c'\hat{\beta}-q}{\sqrt{\sigma^2\,c'(X'X)^{-1}c}}\sim N(0,1).
\]
\framebreak

\textbf{Paso 2 (chi-cuadrado independiente).} Sabemos que
\[
\frac{(n-k-1)\,\hat{\sigma}^2}{\sigma^2}\sim \chi^2_{\,n-k-1},
\]
y que $\hat{\sigma}^2$ es independiente de $\hat{\beta}$ (los residuos son independientes de los valores ajustados bajo normalidad).

\textbf{Paso 3 (cociente $t$).} Por definición, $t=\dfrac{Z}{\sqrt{\chi^2_{\,n-k-1}/(n-k-1)}}\sim t_{\,n-k-1}$. Al formar este cociente, $\sigma^2$ se cancela y queda:
\[
t=\frac{c'\hat{\beta}-q}{\sqrt{\hat{\sigma}^2\,c'(X'X)^{-1}c}}
=\frac{c'\hat{\beta}-q}{\widehat{\text{ee}}(c'\hat{\beta})}\sim t_{\,n-k-1}.
\]
\hfill $\square$
\end{exampleblock}
La intuición: reemplazar $\sigma^2$ (desconocida) por $\hat{\sigma}^2$ introduce variabilidad extra, lo que ``engorda las colas'' y nos lleva de la normal a la $t$ de Student.
\end{frame}

$\mathbf{(X^{\top}X)^{-1}} =

            -0.00035 & -0.00002 & 0.00001 \\

        $\hat{\beta} =
        \begin{bmatrix}
            140 \\

        \end{bmatrix}$; $n=140$

Entonces: $\mathbf{(X^{\top}X)^{-1}\hat{\sigma}^2} =
        \begin{bmatrix}
            8497.04975 & -251.20725 & -49.67375 \\
            -251.20725 & \textcolor{red}{42.5775} & \textcolor{blue}{-2.8385} \\
            -49.67375 & \textcolor{blue}{-2.8385} & \textcolor{red}{1.41925} \\

        $\hat{\text{Var}}(\hat{\beta}_1)=\textcolor{red}{42.5775} \quad \hat{\text{Var}}(\hat{\beta}_2)=\textcolor{red}{1.41925} $
\end{frame}

\begin{frame}{Ejemplo: Test $\beta_1 = \beta_2$ con $\alpha = 0{,}05$ (procedimiento)}
Procedimiento:
        \begin{itemize}
            \item Paso 1: Hipótesis: $H_0: \beta_1 - \beta_2 = 0$\quad vs\quad
            $H_1: \beta_1 - \beta_2 \neq 0$.
            \item Paso 2: Construir el error estándar de la diferencia:
            \begin{itemize}
                \item $\sqrt{\hat{\text{Var}}(\hat{\beta}_1 - \hat{\beta}_2)} = \sqrt{42{,}58 + 1{.}42 - 2(-2{.}84)} = 7{,}05$
            \end{itemize}
            \item Paso 3: Estadístico: $t_e = \dfrac{\hat{\beta}_1-\hat{\beta}_2}{\widehat{\text{ee}}} = \dfrac{42 - 5}{7{,}05} = 5{.}3$
            \item Paso 4: Rechazo $H_0$ porque $5{.}3 > 1{.}98$, valor crítico de la $t$ con $137$ grados de libertad al $5\%$ de significancia.
            \item \textbf{Conclusión económica:} los efectos de las dos variables \textbf{no} son iguales.
        \end{itemize}
\end{frame}

\section{Reparametrización}

\begin{frame}{Método de reparametrización: la idea}
\begin{block}{Idea}
En vez de calcular ``a mano'' la $\hat{\text{Var}}(\hat{\beta}_1-\hat{\beta}_2)$, \textbf{reescribimos el modelo} introduciendo un parámetro nuevo $\theta=\beta_1-\beta_2$. El software entrega entonces $\hat{\theta}$ y $\widehat{\text{ee}}(\hat{\theta})$ \textbf{directamente}, y testeamos $\theta=0$ con un $t$ usual.
\end{block}
\pause
\begin{itemize}
    \item Es \textbf{exactamente equivalente} al test directo: produce el mismo $t$ y la misma conclusión.
    \item Ventaja práctica: no hay que extraer covarianzas de la matriz $(X'X)^{-1}$; basta leer la salida estándar de la regresión reparametrizada.
    \item Es la forma habitual de testear $\beta_1=\beta_2$ en R o Python.
\end{itemize}
\end{frame}

\begin{frame}{Reparametrización: justificación algebraica}
\begin{exampleblock}{De $\beta_1=\beta_2$ a $\theta=0$}
Partimos del modelo
\[
y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + u.
\]
\textbf{Paso 1.} Definimos $\theta=\beta_1-\beta_2 \Rightarrow \beta_1=\theta+\beta_2$.

\textbf{Paso 2.} Sustituimos en el modelo:
\[
y = \beta_0 + (\theta+\beta_2)x_1 + \beta_2 x_2 + u = \beta_0 + \theta x_1 + \beta_2\,(x_1+x_2) + u.
\]
\textbf{Paso 3.} Definimos la variable nueva $w = x_1+x_2$. El modelo a estimar es
\[
\boxed{\,y = \beta_0 + \theta\,x_1 + \beta_2\,w + u\,}.
\]
\end{exampleblock}
\pause
\begin{itemize}
    \item Ahora $\theta$ es \textbf{un coeficiente más} de la regresión: su $\hat{\theta}$ y $\widehat{\text{ee}}(\hat{\theta})$ salen directo.
    \item $H_0:\beta_1=\beta_2 \;\Leftrightarrow\; H_0:\theta=0$. Test: $t=\hat{\theta}/\widehat{\text{ee}}(\hat{\theta})$.
\end{itemize}
\end{frame}

\begin{frame}{Reparametrización para combinaciones generales}
\begin{itemize}
    \item El truco se generaliza a cualquier restricción $c'\beta=q$.
    \item \textbf{Ejemplo: rendimientos constantes a escala} $\beta_1+\beta_2=1$. Defina $\theta=\beta_1+\beta_2-1$:
    \[
    \beta_1 = \theta - \beta_2 + 1.
    \]
    Sustituyendo en $y=\beta_0+\beta_1 x_1+\beta_2 x_2+u$:
    \[
    y - x_1 = \beta_0 + \theta\, x_1 + \beta_2\,(x_2 - x_1) + u.
    \]
    Se regresa $(y-x_1)$ sobre $x_1$ y $(x_2-x_1)$, y se testea $\theta=0$.
    \pause
    \item \textbf{Ejemplo: $\beta_1=2\beta_2$.} Defina $\theta=\beta_1-2\beta_2$, entonces $\beta_1=\theta+2\beta_2$ y
    \[
    y = \beta_0 + \theta\,x_1 + \beta_2\,(2x_1+x_2) + u.
    \]
    \item En todos los casos: \textbf{una resta o suma de variables} y un $t$ sobre el coeficiente nuevo.
\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Intervalo de confianza}

\begin{frame}{Intervalo de confianza para una combinación lineal}
\begin{block}{IC al $(1-\alpha)\times 100\%$ para $c'\beta$}
\[
c'\hat{\beta} \;\pm\; t_{\alpha/2,\,n-k-1}\cdot \widehat{\text{ee}}(c'\hat{\beta}),
\qquad \widehat{\text{ee}}(c'\hat{\beta})=\sqrt{\hat{\text{Var}}(c'\hat{\beta})}.
\]
Para la diferencia:
\[
(\hat{\beta}_1-\hat{\beta}_2)\;\pm\; t_{\alpha/2,\,n-k-1}\cdot\sqrt{\hat{\text{Var}}(\hat{\beta}_1-\hat{\beta}_2)}.
\]
\end{block}
\pause
\begin{itemize}
    \item \textbf{Conexión con el test:} si el valor de $q$ bajo $H_0$ \textbf{cae fuera} del IC, se rechaza $H_0$ al nivel $\alpha$; si cae \textbf{dentro}, no se rechaza.
    \item En particular, para $H_0:\beta_1=\beta_2$ ($q=0$): se rechaza si el IC para $\beta_1-\beta_2$ \textbf{no contiene al cero}.
\end{itemize}
\end{frame}

\section{Ejercicios resueltos}

\begin{frame}{Ejercicio Resuelto 1: Test $\beta_1 = 2\beta_2$}
\begin{block}{Ejercicio 1}
Testee la $H_0$ de que el efecto de $x_1$ sobre $y$ es el doble que el efecto de $x_2$: $\beta_1=2\beta_2$, frente a la alternativa de que es diferente, con $\alpha=5\%$. Use los datos del ejemplo anterior: $\hat{\beta}_1=42$, $\hat{\beta}_2=5$, $\hat{\text{Var}}(\hat{\beta}_1)=42{.}5$, $\hat{\text{Var}}(\hat{\beta}_2)=1{.}42$, $\hat{\text{Cov}}(\hat{\beta}_1,\hat{\beta}_2)=-2{.}84$, $n=140$.
    \begin{itemize}
          \item $H_0: \beta_1 - 2\beta_2 = 0$
          \item $H_1: \beta_1 - 2\beta_2 \neq 0$.
  \end{itemize}
\end{block}
\end{frame}

\begin{frame}{Ejercicio Resuelto 1: Solución}
\begin{exampleblock}{Solución}
   \begin{itemize}
      \item \textbf{Paso 1.} Estadístico objetivo: $t_e = \dfrac{\hat{\beta}_1 - 2\hat{\beta}_2}{\sqrt{\hat{\text{Var}}(\hat{\beta}_1 - 2\hat{\beta}_2)}}$.
      \item \textbf{Paso 2.} Varianza de la combinación ($k=1$, $c=-2$):
            \begin{align*}
               \hat{\text{Var}}(\hat{\beta}_1 - 2\hat{\beta}_2) &= \hat{\text{Var}}(\hat{\beta}_1) + 4\,\hat{\text{Var}}(\hat{\beta}_2) - 4\,\hat{\text{Cov}}(\hat{\beta}_1, \hat{\beta}_2)\\
               &= 42{.}5 + 4(1{.}42) - 4(-2{.}84) = 59{.}5\\
                \Rightarrow\ \widehat{\text{ee}}(\hat{\beta}_1 - 2\hat{\beta}_2) &= \sqrt{59{.}5} = 7{.}68
            \end{align*}
        \item \textbf{Paso 3.} $t_e = \dfrac{42 - 2(5)}{7{.}68} = \dfrac{32}{7{.}68} = 4{.}16$.
        \item \textbf{Paso 4.} Como $|4{.}16| > 1{.}98$ ($t$ crítica con $137$ g.l. al $5\%$), \textbf{se rechaza} $H_0$: el efecto de $x_1$ no es exactamente el doble del de $x_2$.
   \end{itemize}
\end{exampleblock}
\end{frame}

\begin{frame}{Ejercicio Resuelto 2: $3\beta_2=2\beta_3$ (test + IC)}
\begin{block}{Ejercicio 2 (adaptado de Certamen 2 PC, T2)}
Modelo $y=\beta_0+\beta_1 x_1+\beta_2 x_2+\beta_3 x_3+u$, estimado con $n=150$. Se obtuvo $\hat{\beta}=(5{.}5,\,10{.}1,\,1{.}2,\,2{.}1)'$, $\hat{\sigma}^2=1{.}5$ y, de $(X'X)^{-1}$, los elementos relevantes del bloque $(\beta_2,\beta_3)$:
\[
[(X'X)^{-1}]_{22}=0{.}27,\quad [(X'X)^{-1}]_{33}=1{.}3,\quad [(X'X)^{-1}]_{23}=-0{.}16.
\]
\textbf{(a)} Plantee $H_0$ para $3\beta_2=2\beta_3$. \textbf{(b)} Testee al $5\%$. \textbf{(c)} Construya el IC al $95\%$ para $3\beta_2-2\beta_3$.
\end{block}
\end{frame}

\begin{frame}{Ejercicio Resuelto 2: Solución}
\begin{exampleblock}{Solución}
\textbf{(a)} $H_0: 3\beta_2 - 2\beta_3 = 0$\quad vs\quad $H_1: 3\beta_2 - 2\beta_3 \neq 0$.

\textbf{(b)} Primero las varianzas/covarianza ($\hat{\text{Var}}=\hat{\sigma}^2\cdot[(X'X)^{-1}]$):
\begin{align*}
\hat{\text{Var}}(\hat{\beta}_2)&=1{.}5(0{.}27)=0{.}405,\quad \hat{\text{Var}}(\hat{\beta}_3)=1{.}5(1{.}3)=1{.}95,\\
\hat{\text{Cov}}(\hat{\beta}_2,\hat{\beta}_3)&=1{.}5(-0{.}16)=-0{.}24.
\end{align*}
Con $c=(3,-2)$ sobre el bloque $(\beta_2,\beta_3)$:
\begin{align*}
\hat{\text{Var}}(3\hat{\beta}_2-2\hat{\beta}_3)&=9\,\hat{\text{Var}}(\hat{\beta}_2)+4\,\hat{\text{Var}}(\hat{\beta}_3)-12\,\hat{\text{Cov}}(\hat{\beta}_2,\hat{\beta}_3)\\
&=9(0{.}405)+4(1{.}95)-12(-0{.}24)=14{.}325.
\end{align*}
$\widehat{\text{ee}}=\sqrt{14{.}325}=3{.}785$. Estimación puntual: $3(1{.}2)-2(2{.}1)=-0{.}6$.
\[
t_e=\frac{-0{.}6}{3{.}785}=-0{.}16,\qquad |{-0{.}16}|<1{.}96 \Rightarrow \textbf{no se rechaza } H_0.
\]
\end{exampleblock}
\end{frame}

\begin{frame}{Ejercicio Resuelto 2: Solución (c)}
\begin{exampleblock}{Solución — Intervalo de confianza}
\textbf{(c)} IC al $95\%$ para la combinación $3\beta_2-2\beta_3$ (con $t_{0.975,\,n-k-1}\approx 1{.}96$):
\[
(3\hat{\beta}_2-2\hat{\beta}_3)\pm 1{.}96\cdot\widehat{\text{ee}} = -0{.}6 \pm 1{.}96(3{.}785) = -0{.}6 \pm 7{.}42.
\]
\[
\boxed{\ \text{IC}_{95\%} = [-8{.}02,\ 6{.}82]\ }
\]
\begin{itemize}
    \item El intervalo \textbf{contiene al cero}, coherente con no rechazar $H_0:3\beta_2-2\beta_3=0$.
    \item Conclusión: los datos son compatibles con la restricción $3\beta_2=2\beta_3$.
\end{itemize}
\end{exampleblock}
\end{frame}

\section{Ejemplo aplicado}

\begin{frame}{Ejemplo aplicado: gasto de campaña y votos}
\begin{block}{Contexto (datos de campaña, $n=173$)}
Modelo log--log (coeficientes son elasticidades):
\[
\text{voteA}=\beta_0+\beta_1\,\text{lexpendA}+\beta_2\,\text{lexpendB}+\beta_3\,\text{prtystrA}+u.
\]
Estimación: $\hat{\beta}_1=6{.}083$ (gasto propio), $\hat{\beta}_2=-6{.}615$ (gasto rival). De la matriz de var--cov: $\hat{\text{Var}}(\hat{\beta}_1)=0{.}146$, $\hat{\text{Var}}(\hat{\beta}_2)=0{.}144$, $\hat{\text{Cov}}(\hat{\beta}_1,\hat{\beta}_2)=-0{.}003$.

\textbf{Pregunta:} ¿un alza de $1\%$ en el gasto propio se compensa exactamente con un alza de $1\%$ del rival? Es decir, $H_0:\beta_1+\beta_2=0$.
\end{block}
\pause
\begin{itemize}
    \item Es una restricción de \textbf{suma}: $c=(0,1,1,0)'$, $q=0$.
\end{itemize}
\end{frame}

\begin{frame}{Ejemplo aplicado: solución}
\begin{exampleblock}{Test de $\beta_1+\beta_2=0$}
Varianza de la suma ($k=c=1$):
\[
\hat{\text{Var}}(\hat{\beta}_1+\hat{\beta}_2)=0{.}146+0{.}144+2(-0{.}003)=0{.}284,\quad \widehat{\text{ee}}=\sqrt{0{.}284}=0{.}533.
\]
\[
t_e=\frac{\hat{\beta}_1+\hat{\beta}_2}{\widehat{\text{ee}}}=\frac{6{.}083+(-6{.}615)}{0{.}533}=\frac{-0{.}532}{0{.}533}\approx -1{.}00.
\]
Con $169$ g.l., $|{-1{.}00}|<1{.}96$ $\Rightarrow$ \textbf{no se rechaza} $H_0$ al $5\%$. La hipótesis de compensación exacta es compatible con los datos.
\end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Ejemplo aplicado: implementación en R}
\begin{exampleblock}{Reparametrización en R (test de $\beta_1+\beta_2=0$)}
\begin{verbatim}
# Modelo original
m <- lm(voteA ~ lexpendA + lexpendB + prtystrA,
        data = datos)

# Opcion 1: test directo de la combinacion lineal
library(car)
linearHypothesis(m, "lexpendA + lexpendB = 0")

# Opcion 2: reparametrizacion theta = b1 + b2
# y = b0 + theta*lexpendA + b2*(lexpendB - lexpendA) + ...
datos$w <- datos$lexpendB - datos$lexpendA
m2 <- lm(voteA ~ lexpendA + w + prtystrA, data = datos)
summary(m2)   # el coef. de lexpendA es theta = b1 + b2
\end{verbatim}
\end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Ejemplo aplicado: implementación en Python}
\begin{exampleblock}{statsmodels (test de $\beta_1+\beta_2=0$)}
\begin{verbatim}
import statsmodels.formula.api as smf

m = smf.ols("voteA ~ lexpendA + lexpendB + prtystrA",
            data=datos).fit()

# Test directo de la restriccion lineal (Wald/t)
print(m.t_test("lexpendA + lexpendB = 0"))

# Reparametrizacion: theta = b1 + b2
datos["w"] = datos["lexpendB"] - datos["lexpendA"]
m2 = smf.ols("voteA ~ lexpendA + w + prtystrA",
             data=datos).fit()
print(m2.summary())  # coef de lexpendA es theta
\end{verbatim}
\end{exampleblock}
\end{frame}

\section{Cierre}

\begin{frame}{Puente a la próxima clase: ¿y si hay varias restricciones?}
\begin{itemize}
    \item Hoy probamos \textbf{una} restricción lineal con un estadístico $t$.
    \item ¿Qué pasa si queremos probar \textbf{varias} restricciones a la vez? Por ejemplo:
    \[
    H_0:\ \beta_1=0\ \text{y}\ \beta_2=0\ \text{y}\ \beta_3=0.
    \]
    \item Hacer varios test $t$ por separado \textbf{no} es correcto (problema de comparaciones múltiples).
    \item La herramienta adecuada es la \textbf{prueba $F$}, que compara el modelo restringido con el no restringido vía sumas de cuadrados. \textbf{Eso veremos la próxima clase.}
\end{itemize}
\end{frame}

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 11}
\begin{enumerate}
    \item Una restricción lineal $c'\beta=q$ (igualdad o combinación) se testea con un \textbf{estadístico $t$} con $n-k-1$ g.l.
    \item La varianza de una combinación lineal incluye el término de \textbf{covarianza}: $\text{Var}(\hat{\beta}_1-\hat{\beta}_2)=\text{Var}(\hat{\beta}_1)+\text{Var}(\hat{\beta}_2)-2\text{Cov}(\hat{\beta}_1,\hat{\beta}_2)$.
    \item En matriz: $\hat{\text{Var}}(c'\hat{\beta})=\hat{\sigma}^2\,c'(X'X)^{-1}c$.
    \item El \textbf{método de reparametrización} ($\theta=\beta_1-\beta_2$) entrega $\hat{\theta}$ y su ee directos: mismo resultado, sin manipular covarianzas.
    \item El \textbf{IC} para $c'\beta$ y el test son equivalentes: rechazo $\Leftrightarrow$ $q$ fuera del intervalo.
\end{enumerate}
\end{block}

\end{alertblock}

\end{frame}
