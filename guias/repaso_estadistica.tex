\section{Distribución normal}

La variable aleatoria X, que toma todos los valores reales, $-\infty<x<+\infty$, tiene una distribución normal o gaussiana si su función de densidad es de la forma:

$$f(x)={1\over \sqrt{2\pi \sigma}}\exp\left ({-{1\over 2} \left ( {x-\mu\over \sigma}\right )^2 }\right)$$

    Donde los parámetros $\mu$ y $\sigma$ representan la media y la desviación estándar de X y  satisfacen $-\infty<\mu<+\infty$, $0< \sigma$.

    Usamos la notación: $X\sim N(\mu, \sigma^2)$

\section{Distribución t de Student}



Se define como:
\[
T = \frac{Z}{\sqrt{U / \nu}}, \quad Z \sim \mathcal{N}(0,1), \quad U \sim \chi^2_\nu \
\]
donde $\nu$ son los grados de libertad.

 \[
\text{g.l.} = \text{número de observaciones} - \text{número de parámetros estimados}
\]



\begin{itemize}
    \item Simétrica respecto a cero, similar a la normal, pero con colas más pesadas.
    \item A mayor $\nu$, más cercana a la normal.
    \item Para $\nu \to \infty$, $\text{t}_\nu \to \mathcal{N}(0,1)$.
\end{itemize}

\section{Teorema Central  del Límite}

Sea $X_1, X_2, \dots, X_n$ una muestra aleatoria simple de variables independientes e idénticamente distribuidas (i.i.d.) con media $\mu$ y varianza $\sigma^2 < \infty$. La media muestral es:
\[
\bar{X} = \frac{1}{n} \sum_{i=1}^n X_i
\]
El Teorema Central del Límite establece que, cuando $n \to \infty$:
\[
Z_n = \frac{\bar{X} - \mu}{\sigma / \sqrt{n}} \ \xrightarrow{d} \ \mathcal{N}(0,1)
\]
donde $\xrightarrow{d}$ denota convergencia en distribución.


\begin{itemize}
    \item La distribución de $\bar{X}$ se aproxima a una normal estándar al aumentar $n$, aun si la distribución de cada $X_i$ no es normal.
   
    \item Para $n \geq 30$ suele aceptarse la aproximación normal en aplicaciones prácticas.
\end{itemize}

\section{Intervalos de confianza para media: $\sigma$ conocida vs. desconocida}

\subsection{$\sigma$ conocida}
Cuando la desviación estándar poblacional $\sigma$ es conocida:
\[
Z = \frac{\bar{X} - \mu_0}{\sigma / \sqrt{n}} \sim \mathcal{N}(0,1)
\]
Se utiliza la distribución normal estándar.

\subsection{$\sigma$ desconocida}
En la práctica, $\sigma$ se estima por la desviación estándar muestral:
\[
s = \sqrt{\frac{\sum_{i=1}^n (X_i - \bar{X})^2}{n-1}}
\]

La distribución es T-student:
\[
T = \frac{\bar{X} - \mu_0}{s / \sqrt{n}}
\]
El cual sigue una \textbf{distribución t de Student} con $n-1$ grados de libertad.

\vspace{4cm}

% Ecuación general con variable estandarizada
\[ P\left( a \ \le \ \frac{\bar{X} - \mu}{SE} \ \le \ b \right) = 1 - \alpha\]

% Definición de error estándar
\[
SE =
\begin{cases}
\dfrac{\sigma}{\sqrt{n}}, & \text{si } \sigma \text{ es conocida} \\
\dfrac{s}{\sqrt{n}}, & \text{si } \sigma \text{ es desconocida}
\end{cases}
\]

% Despeje para mu
\[
P\left( a \cdot SE \ \le \ \bar{X} - \mu \ \le \ b \cdot SE \right) = 1 - \alpha
\]
\[
P\left( \bar{X} - b \cdot SE \ \le \ \mu \ \le \ \bar{X} - a \cdot SE \right) = 1 - \alpha
\]

% Valores de a y b según el caso
\[
\text{Caso normal:} \quad a = z_{\alpha/2}, \quad b = -z_{\alpha/2}
\]
\[
\text{Caso t de Student:} \quad a = t_{\alpha/2,\,n-1}, \quad b =-t_{\alpha/2,\,n-1}
\]
