\begin{frame}{Objetivos de la Clase}

Al finalizar esta clase, el estudiante será capaz de:

\begin{itemize}
    \item[\checkmark] Reconocer problemas donde la variable \textbf{dependiente} es binaria.
    \item[\checkmark] Plantear y estimar el \textbf{Modelo de Probabilidad Lineal (MPL)} por MCO.
    \item[\checkmark] Interpretar los coeficientes del MPL como \textbf{cambios en la probabilidad} de éxito.
    \item[\checkmark] Demostrar que en el MPL $\text{Var}(u\mid x)=p(x)(1-p(x))$, es decir, \textbf{heterocedasticidad inherente}.
    \item[\checkmark] Identificar los dos problemas del MPL (predicciones fuera de $[0,1]$ y heterocedasticidad) y por qué Logit/Probit los resuelven.
\end{itemize}

\end{frame}

\section{Motivación}

\begin{frame}{Variables dependientes binarias}

    Muchos problemas de la vida real consisten en explicar una variable dummy.
\begin{itemize}
    \item ¿Qu\'e explica si un alumno va a un colegio p\'ublico o privado?
\item ¿Qui\'enes son las personas que se jubilan anticipadamente?
\item  ¿Qu\'e tipos de personas tienen mas probabilidad de chocar el auto?
\end{itemize}

\pause
\begin{block}{Idea central}
Si $y\in\{0,1\}$, entonces la esperanza condicional \emph{es} una probabilidad:
\[
E[y\mid x] = P(y=1\mid x).
\]
\end{block}

\end{frame}

\section{El Modelo de Probabilidad Lineal}

\begin{frame}{Modelo de probabilidad lineal (MPL)}

    \begin{figure}
        \centering
        \includegraphics[width=0.9\linewidth]{sem11/images/MPL.png}
    \end{figure}
\end{frame}

\begin{frame}{El MPL: definición y coeficientes}

\begin{block}{Definición: Modelo de Probabilidad Lineal}
Si $y\in\{0,1\}$, el MPL modela la probabilidad de éxito de forma lineal:
\[
P(y=1\mid x) = E[y\mid x] = \beta_0 + \beta_1 x_1 + \cdots + \beta_k x_k.
\]
\end{block}

\pause

\begin{itemize}
    \item Como $E[y\mid x]$ es una probabilidad, cada coeficiente mide un \textbf{cambio en la probabilidad}:
    \[
    \frac{\partial P(y=1\mid x)}{\partial x_j} = \beta_j.
    \]
    \item $\beta_j$ es el cambio (en puntos de probabilidad) de $P(y=1)$ ante un aumento de una unidad en $x_j$, \emph{ceteris paribus}.
    \item El efecto marginal es \textbf{constante}, lo cual será una limitación.
\end{itemize}

\end{frame}

\section{Problemas del MPL}

\begin{frame}{El MPL presenta dos problemas}

\begin{enumerate}
    \item Se pueden predecir individuos que tengan probabilidades de \'exito negativas o mayores que uno.
    \begin{itemize}
      \item Si nos interesa el efecto en el promedio, este problema no es muy grave.
      \item Si nos interesa la predicción o efecto para individuos extremos, los modelos no-lineales Logit y Probit son mejores para este tipo de casos (no los vamos a ver en el curso).
    \end{itemize}
    \item No se cumple el supuesto de homocedasticidad.
\end{enumerate}

\end{frame}

\begin{frame}[allowframebreaks]{Problema 2: heterocedasticidad inherente del MPL}

\begin{alertblock}{Proposición: el MPL es heterocedástico por construcción}
En el MPL, la varianza condicional del error \textbf{no es constante}:
\[
\text{Var}(u\mid x) = p(x)\,\bigl(1-p(x)\bigr),\qquad p(x)=E[y\mid x]=x'\beta,
\]
que depende de $x$. Por tanto $\text{Var}(u\mid x)\neq\sigma^2$.
\end{alertblock}

\framebreak

\begin{exampleblock}{Demostración}
Como $y\in\{0,1\}$, condicional a $x$ se distribuye Bernoulli con parámetro $p(x)=P(y=1\mid x)$. El error es
\[
u = y - E[y\mid x] = y - p(x).
\]
\pause
La varianza del error es la varianza de $y$ (restar la constante $p(x)$ no cambia la varianza):
\[
\text{Var}(u\mid x) = \text{Var}(y\mid x).
\]
\pause
Para una Bernoulli$(p)$, $E[y\mid x]=p(x)$ y $E[y^2\mid x]=p(x)$ (pues $y^2=y$ cuando $y\in\{0,1\}$), luego
\[
\text{Var}(y\mid x) = E[y^2\mid x]-\bigl(E[y\mid x]\bigr)^2 = p(x) - p(x)^2 = p(x)\bigl(1-p(x)\bigr).
\]
\pause
Sustituyendo $p(x)=x'\beta$:
\[
\boxed{\;\text{Var}(u\mid x) = x'\beta\,\bigl(1-x'\beta\bigr)\;}
\]
que claramente \textbf{depende de $x$}: hay heterocedasticidad. \hfill$\square$
\end{exampleblock}

\framebreak

\begin{block}{Consecuencias e implicancia práctica}
\begin{itemize}
    \item MCO sigue siendo \textbf{insesgado y consistente}, pero \textbf{no eficiente}.
    \item Los errores estándar clásicos son \textbf{incorrectos} $\Rightarrow$ inferencia inválida.
    \item \textbf{Solución:} usar errores estándar \textbf{robustos} (White), o MCG ponderando por $1/\sqrt{\hat p(1-\hat p)}$.
\end{itemize}
\end{block}

\end{frame}

\section{Ejercicios resueltos}

\begin{frame}{Ejercicio Resuelto 1 — Interpretación del MPL}

\begin{block}{Ejercicio 1}
Considere la regresión, donde la variable dependiente {\color{blue}Accidente} toma valor 1 si la persona tuvo al menos un accidente de tránsito y 0 si no:
\[
\hat{Accidente}_i = 0.30 - 0.05\,educ_i - 0.0001\,ingreso_i
\]
($educ$ en años; $ingreso$ en dólares). Interprete $\hat\beta_0$ y el coeficiente de $educ$; calcule la probabilidad para $ingreso=2000$, $educ=20$.
\end{block}

\end{frame}

\begin{frame}{Ejercicio Resuelto 1 — Solución}
\begin{exampleblock}{Solución}
\textbf{Intercepto:} $\hat\beta_0=0.30$ es la probabilidad de accidente de alguien con $0$ educación y $0$ ingreso: $30\%$.

\textbf{Educación:} cada año adicional de educación \emph{reduce} la probabilidad de accidente en $0.05$ (5 puntos porcentuales), \emph{ceteris paribus}.

\textbf{Predicción} para $educ=20$, $ingreso=2000$:
\[
\hat P = 0.30 - 0.05(20) - 0.0001(2000) = 0.30 - 1.00 - 0.20 = -0.90.
\]
\pause
La predicción es $\mathbf{-0.90}$, \textbf{negativa}: ilustra el \emph{primer problema} del MPL (predicciones fuera de $[0,1]$). En la práctica se trunca a $0$ o se usa Logit/Probit.
\end{exampleblock}
\end{frame}

\begin{frame}{Ejercicio Resuelto 2 — Felicidad (MPL)}
\begin{block}{Ejercicio 2 (fuente: guía de dummies)}
Sea $B_i\in\{0,1\}$ la \textit{felicidad} ($1$ = feliz). Con edad $E_i$, mujer $M_i$, escolaridad $S_i$, ingreso familiar $I_i$ (miles de pesos/año) y salud $H_i$ se plantea:
\[
B_i = \alpha_1 + \alpha_2 E_i + \alpha_3 M_i + \alpha_4 S_i + \alpha_5 I_i + \alpha_6 H_i + e_i.
\]
(1) ¿Se puede estimar por MCO? \; (2) ¿Cuál es $P(B_i=1\mid X_i)$? \; (3) Efecto marginal del ingreso. \; (4) Dos críticas a MCO.
\end{block}
\end{frame}

\begin{frame}{Ejercicio Resuelto 2 — Solución}
\begin{exampleblock}{Solución}
\textbf{(1)} Sí: correr MCO sobre una dependiente binaria es justamente el \textbf{MPL}. Es válido como punto de partida, pero arrastra los problemas que se mencionan en (4).

\textbf{(2)} En el MPL la probabilidad es la esperanza condicional:
\[
P(B_i=1\mid X_i)=E[B_i\mid X_i]=\alpha_1+\alpha_2 E_i+\alpha_3 M_i+\alpha_4 S_i+\alpha_5 I_i+\alpha_6 H_i.
\]
\textbf{(3)} Efecto marginal del ingreso, constante:
$\dfrac{\partial P(B_i=1\mid X_i)}{\partial I_i}=\alpha_5$ (puntos de probabilidad por cada mil pesos).

\textbf{(4)} (i) \textbf{Heterocedasticidad inherente}: $\text{Var}(e_i\mid X_i)=p_i(1-p_i)$ $\Rightarrow$ inferencia inválida sin errores robustos. (ii) \textbf{Predicciones fuera de $[0,1]$} y efecto marginal constante poco realista.
\end{exampleblock}
\end{frame}

\section{Ejemplo aplicado}

\begin{frame}[fragile]{Ejemplo aplicado en R: MPL con errores robustos}
\begin{exampleblock}{Estimación del MPL con corrección de heterocedasticidad}
\begin{verbatim}
library(sandwich); library(lmtest)

# trabaja = 1 si participa en el mercado laboral
mpl <- lm(trabaja ~ educacion + edad + hijos,
          data = casen)

# Errores estandar robustos (White) por la
# heterocedasticidad inherente del MPL
coeftest(mpl, vcov = vcovHC(mpl, type = "HC1"))

# Cuantas predicciones quedan fuera de [0,1]?
mean(fitted(mpl) < 0 | fitted(mpl) > 1)
\end{verbatim}
\end{exampleblock}
\end{frame}

\section{Más allá del MPL}

\begin{frame}{Solución a los problemas: Logit y Probit}
\begin{itemize}
    \item Para evitar predicciones fuera de $[0,1]$ se usa una función $G(\cdot)$ que mapea a $(0,1)$:
    \[
    P(y=1\mid x) = G(\beta_0+\beta_1 x_1+\cdots+\beta_k x_k),\qquad G:\mathbb{R}\to(0,1).
    \]
    \begin{itemize}
        \item \textbf{Logit:} $G$ es la función logística, $G(z)=\dfrac{e^z}{1+e^z}$.
        \item \textbf{Probit:} $G$ es la normal estándar acumulada, $G(z)=\Phi(z)$.
    \end{itemize}
    \pause
    \item Ventajas: las probabilidades quedan en $(0,1)$ y el efecto marginal es no lineal (más realista).
    \item Se estiman por \textbf{máxima verosimilitud}. \textit{No los veremos en detalle en este curso}, pero conviene saber que existen y cuándo usarlos.
    \item A pesar de sus problemas, el MPL es un buen punto de partida y suele dar efectos promedio similares a Logit/Probit.
\end{itemize}
\end{frame}

\hat{aprueba}_i = 0.10 + 0.08\,horas_i.

\begin{itemize}

    \item ¿A partir de cuántas horas la predicción supera $1$? ¿Qué problema ilustra?

\end{frame}

\begin{frame}{Ejercicio Propuesto 2}
\begin{block}{Ejercicio 2 (Propuesto — intermedio)}
Un MPL estima $P(\text{moroso}=1)=0.05+0.0003\,deuda - 0.04\,empleo$, donde \textit{empleo}$=1$ si la persona tiene trabajo formal.
\begin{itemize}
    \item Para un valor ajustado $\hat p=0.2$, calcule la varianza del error $\text{Var}(u\mid x)$.
    \item Repítalo para $\hat p=0.5$. ¿En qué valor de $\hat p$ es máxima la varianza? Demuéstrelo.
\end{itemize}
\end{block}
{\small \textit{Indicación: maximice $p(1-p)$ derivando respecto de $p$.}}
\end{frame}

\begin{frame}{Ejercicio Propuesto 3}
\begin{block}{Ejercicio 3 (Propuesto — desafiante)}
Demuestre que en el MPL los errores estándar clásicos de MCO son inconsistentes y proponga el estimador de varianza robusto (White) apropiado. Luego, explique en qué casos preferiría Logit/Probit por sobre el MPL aun corrigiendo la heterocedasticidad.
\end{block}
{\small \textit{Indicación: parta de $\text{Var}(\hat\beta\mid X)=(X'X)^{-1}X'\,\Omega\,X(X'X)^{-1}$ con $\Omega=\operatorname{diag}(p_i(1-p_i))$.}}
\end{frame}

\section{Resumen}

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 15}
\begin{enumerate}
    \item Cuando $y\in\{0,1\}$, $E[y\mid x]=P(y=1\mid x)$: modelar la media es modelar una probabilidad.
    \item El \textbf{MPL} estima esa probabilidad por MCO; cada $\beta_j$ es un \textbf{cambio en la probabilidad} (efecto marginal constante).
    \item Problema 1: las predicciones pueden caer \textbf{fuera de} $[0,1]$.
    \item Problema 2: \textbf{heterocedasticidad inherente}, $\text{Var}(u\mid x)=p(x)(1-p(x))$ $\Rightarrow$ usar errores robustos.
    \item \textbf{Logit/Probit} resuelven ambos problemas usando $G(\cdot)$ acotada en $(0,1)$; el MPL sigue siendo un buen punto de partida.
\end{enumerate}
\end{block}
