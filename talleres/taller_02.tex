\section{Preguntas}

Use \texttt{wage1} del paquete \texttt{wooldridge} y construya
\texttt{lwage = log(wage)}. Considere como modelo ampliado de referencia:
\[
    \texttt{lwage}_i=\beta_0+\beta_1\texttt{educ}_i
    +\beta_2\texttt{exper}_i+\beta_3\texttt{tenure}_i+u_i.
\]

\subsection*{I. Estimación y bondad de ajuste}

\begin{enumerate}

\item Estime por MCO los siguientes tres modelos:
    \begin{align*}
        \text{Modelo 1: }&\texttt{lwage}\sim\texttt{educ},\\
        \text{Modelo 2: }&\texttt{lwage}\sim\texttt{educ+exper},\\
        \text{Modelo 3: }&\texttt{lwage}\sim\texttt{educ+exper+tenure}.
    \end{align*}
    \begin{enumerate}[label=(\alph*)]
        \item Presente una tabla comparable con los coeficientes, errores
              estándar, observaciones y niveles de significancia.
        \item Interprete el signo, magnitud y significancia económica de los
              coeficientes del Modelo 3. Entregue tanto la interpretación
              aproximada como, cuando corresponda, la transformación exacta
              $100(\exp(\widehat\beta_j)-1)$.
        \item Explique cómo cambia la interpretación de \texttt{educ} al pasar
              del Modelo 1 al Modelo 3.
    \end{enumerate}

\item Compare la bondad de ajuste de los tres modelos.
    \begin{enumerate}[label=(\alph*)]
        \item Reporte $R^2$, $R^2$ ajustado, RMSE y el estadístico $F$ global.
        \item Explique por qué el $R^2$ no disminuye al agregar variables y por
              qué también conviene observar el $R^2$ ajustado.
        \item Construya un gráfico de valores observados versus ajustados para
              el Modelo 3 y comente la calidad del ajuste.
    \end{enumerate}

\end{enumerate}

\subsection*{II. Inferencia individual y restricciones lineales}

\begin{enumerate}[resume]

\item Realice inferencia sobre los parámetros del Modelo 3.
    \begin{enumerate}[label=(\alph*)]
        \item Para cada variable explicativa, contraste
              $H_0:\beta_j=0$ frente a $H_1:\beta_j\neq0$. Reporte el
              estadístico $t$, el valor-$p$ y la decisión.
        \item Construya intervalos de confianza al 95\% para los coeficientes.
              Relacione cada intervalo con el contraste bilateral
              correspondiente.
        \item Repita la inferencia utilizando errores estándar robustos a
              heterocedasticidad. Compare las decisiones con las obtenidas
              usando errores estándar convencionales.
    \end{enumerate}

\item Evalúe una hipótesis sobre la diferencia entre retornos.
    \begin{enumerate}[label=(\alph*)]
        \item Contraste $H_0:\beta_1-\beta_2=0$, es decir, que el retorno de
              un año adicional de educación sea igual al de un año adicional
              de experiencia.
        \item Realice el contraste con una restricción lineal utilizando la
              covarianza entre los estimadores. No compare simplemente los
              dos valores-$p$ individuales.
        \item Interprete el resultado en términos de la pregunta económica del
              proyecto.
    \end{enumerate}

\item Evalúe conjuntamente la relevancia de experiencia y antigüedad.
    \begin{enumerate}[label=(\alph*)]
        \item Plantee $H_0:\beta_2=0$ y $\beta_3=0$ frente a la alternativa de
              que al menos uno de los dos parámetros sea distinto de cero.
        \item Identifique el Modelo 3 como no restringido y estime el modelo
              restringido omitiendo \texttt{exper} y \texttt{tenure}.
        \item Realice la prueba $F$ mediante `linearHypothesis()` y mediante
              la comparación de modelos anidados. Compare ambos resultados.
        \item Explique por qué una prueba conjunta no es equivalente a realizar

    \end{enumerate}

\end{enumerate}

\section{Formulario}

\begin{itemize}
    \item Estimador de MCO:
    \[
        \widehat{\boldsymbol\beta}=(X'X)^{-1}X'y.
    \]

    \item Descomposición de sumas de cuadrados y $R^2$:
    \[
        SST=\sum_{i=1}^{n}(y_i-\bar y)^2,\qquad
        SSE=\sum_{i=1}^{n}(\widehat y_i-\bar y)^2,\qquad
        SSR=\sum_{i=1}^{n}\widehat u_i^2,
    \]
    \[
        SST=SSE+SSR,\qquad
        R^2=\frac{SSE}{SST}=1-\frac{SSR}{SST}.
    \]

    \item Estadístico $t$ para $H_0:\beta_j=a_j$:
    \[
        t=\frac{\widehat\beta_j-a_j}{se(\widehat\beta_j)}.
    \]

    \item Intervalo de confianza de $100(1-\alpha)\%$:
    \[
        \widehat\beta_j\pm t_{\alpha/2,\,n-k-1}
        se(\widehat\beta_j).
    \]

    \item Para una combinación lineal $c'\beta=r$:
    \[
        se(c'\widehat\beta)
        =\sqrt{c'\widehat{\operatorname{Var}}(\widehat\beta)c},
        \qquad
        t=\frac{c'\widehat\beta-r}{se(c'\widehat\beta)}.
    \]

    \item Estadístico $F$ para $j$ restricciones:
    \[
        \widehat F
        =\frac{SSR_R-SSR_{UR}}{SSR_{UR}}\frac{n-k-1}{j}
        =\frac{R^2_{UR}-R^2_R}{1-R^2_{UR}}\frac{n-k-1}{j}.
    \]

    \item Interpretación en un modelo log-nivel:
    \[
        100\left(\exp(\widehat\beta_j)-1\right)\% 
    \]
    es el cambio porcentual exacto aproximado asociado a un aumento de una
    unidad en $x_j$; $100\widehat\beta_j\%$ es la aproximación usual.

    \item Diagnóstico y estimación en R:
    \begin{itemize}[noitemsep]
        \item Estimación y ajuste: \texttt{lm()}, \texttt{summary()}.
        \item Inferencia: \texttt{confint()}, \texttt{coeftest()},
              \texttt{vcovHC()}, \texttt{linearHypothesis()},
              \texttt{anova()}.
        \item Diagnóstico: \texttt{bptest()} y \texttt{vif()}.
    \end{itemize}
\end{itemize}
