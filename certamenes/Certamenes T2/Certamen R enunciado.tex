\begin{enumerate}
    \item (5 puntos) Planteando un modelo donde el salario sólo dependa de la educación universitaria de la forma:
    $$Salario = \beta_0 + \beta_1 \cdot Educ$$
    Encuentre los valores de los estimadores $\hat{\beta_0}$ y $\hat{\beta_1}$ que predicen el valor del salario, e interprete su resultado.
    
    \item (5 puntos) Calcule Bondad de ajuste de este modelo, concluya.
        
    \item (5 puntos) Luego se realiza un modelo incluyendo la variable Experiencia, por lo que el salario estaría explicado por la educación universitaria y la experiencia de la siguiente manera:
    $$Salario = \beta_0 + \beta_1 \cdot Educ + \beta_2 \cdot Exp$$
    Calcule los estimadores $\hat{\beta_0}$, $\hat{\beta_1}$ y $\hat{\beta_2}$, que nos sirven para predecir el salario, sabiendo que:
    \[
(X^tX)^{-1} \;=\;
\begin{pmatrix}
145 & -11.5 & -19 \\
-11.5 & 0.925 & 1.5 \\
-19 & 1.5 & 2.5
\end{pmatrix}
\]

\item (5 puntos) Calcule Bondad de ajuste de este último modelo, concluya.
\end{enumerate}

\item (20 Puntos) Se dispone de una muestra de 250 observaciones con la cual se estimó el siguiente modelo de regresión lineal múltiple:
$$Y_i = \beta_0 + \beta_1\,\textit{L}_i + \beta_2\,\textit{K}_i + u_i$$
Donde . $Y$ es la producción, $L$ es cantidad de dinero puesta en Trabajo, y $K$ la cantidad de dinera puesta en capital.
Se desea contrastar el supuesto de heterocedasticidad y, para ello, se aplicará el método de Breusch-Pagan.
\begin{enumerate}

    \item (10 Puntos) Si el $R^2$ de la regresión auxiliar es $0.05$, contraste, al 5\% de significación, la existencia de heterocedasticidad.
    
    \end{enumerate}

\section*{Formulario}


\begin{itemize}

\item El intervalo de confianza de $100(1-\alpha)\%$ para la media poblacional $\mu$ cuando $\sigma$ es \emph{\textbf{conocido}}:
$\color{red}{\bar{x} \pm z_{\frac{\alpha}{2}}\frac{\sigma}{\sqrt{n}}}$

\item El intervalo de confianza de $100(1-\alpha)\%$ para la diferencia de medias
$\mu_x-\mu_y$ es:
${\color{red}(\bar{X}-\bar{Y}) \pm z_{\frac{\alpha}{2}}\sqrt{\frac{\sigma_x^2}{n_x}+\frac{\sigma_y^2}{n_y}}}$

\item  Estimadores MCO.
{\color{red}
$$\hat{\beta}_0 = \bar{y} - \hat{\beta}_1 \bar{x} \quad \quad   \hat{\beta}_1 = \frac{\sum_{i=1}^{n}(x_i - \bar{x})(y_i-\bar{y})}{\sum_{i=1}^{n}(x_i - \bar{x})^2}$$}
    
\item 
	{\color{red}	\[ SST = \sum_{i=1}^{n} (y_i - \bar{y})^2 \quad \quad  SSE = \sum_{i=1}^{n} (\hat{y}_i - \bar{y})^2 \quad \quad SSR = \sum_{i=1}^{n} \hat{u}_i^2  \]}
		
		
    \item Vector de estimadores:    
{\color{red}    $\hat{\beta}=(X^TX)^{-1}X^Ty$}

\item {\color{red}\[
A = \begin{bmatrix}
a & b \\
c & d
\end{bmatrix}
\qquad
A^{-1} = \frac{1}{det(A)} \begin{bmatrix}
d & -b \\
-c & a
\end{bmatrix}
\]}
\end{itemize}

\begin{itemize}

\item Estadistico t para $\beta_j = a_j$

$$\color{red}{t_{\beta_j}=\frac{\hat{\beta_j}-a_j}{s.e.(\hat{\beta_j})}}$$

\item El intervalo de confianza de $100(1-\alpha)\%$ para el estimador $\beta_j$:

$${\color{red}\hat{\beta_j} \pm t_{\frac{\alpha}{2},n-k-1}s.e.(\hat{\beta_j})}$$


\item para el estadistico F entre dos modelos:

$$\color{red}{F=\frac{SSR_r-SSR_{ur}}{SSR_{ur}}\frac{n-k-1}{j}}$$
$$\color{red}{F=\frac{R^2_{ur}-R^2_{r}}{1-R^2_{ur}}\frac{n-k-1}{j}}$$

\item Test de Chow para Grupos.

    \[\color{red}
    F = \frac{(SSR_r - SSR_{ur}) / ((G - 1)(k + 1))}{SSR_{ur} / (n - G(k + 1))} \sim F((G - 1)(k + 1), n - G(k + 1))
    \]

Con

$$\color{red}SSR_{ur} = \sum_{g=1}^G SSR_g$$

\item Estadistico t para igualdad entre parametros:
 \[\color{red}t_{\text{estadístico}} = \frac{\hat{\beta}_1 - \hat{\beta}_2 - 0}{\sqrt{\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2)}}\]


\item  Varianza de una suma.
{\color{red}
$$\text{Var}(kA + cB) = k^2\text{Var}(A) + c^2\text{Var}(B) + 2kc\text{COV}(A, B)$$}

\item $t_{0.95,n} = 1.64$ cuando $n >50$.
\item $t_{0.975,n} = 1.96$ cuando $n >50$.
\item $t_{0.99,n}=2.33$ cuando $n>50$
