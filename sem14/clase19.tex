\begin{frame}{Objetivos de la Clase}

Al finalizar esta clase, el estudiante será capaz de:

\begin{itemize}
    \item[\checkmark] Distinguir datos de series de tiempo de datos de corte transversal y reconocer por qué el \emph{orden temporal} importa para la causalidad.
    \item[\checkmark] Identificar \textbf{tendencias temporales} (determinísticas y estocásticas) y explicar por qué generan \textbf{correlación espuria}.
    \item[\checkmark] Incluir una tendencia lineal como regresor y reinterpretar los coeficientes ``libres de tendencia''.
    \item[\checkmark] Reconocer la \textbf{persistencia} en un proceso AR(1) y la diferencia entre $|\rho|<1$ (estacionario, $I(0)$) y $\rho=1$ (caminata aleatoria, $I(1)$).
    \item[\checkmark] Demostrar que la caminata aleatoria \emph{no} es estacionaria ($\text{Var}(y_t)=t\sigma^2$) y que su primera diferencia sí lo es.
\end{itemize}

\end{frame}

\begin{frame}[shrink=10]
\frametitle{Tendencias temporales en los datos}

\begin{itemize}
  \item Las series de tiempo tienen una diferencia grande con los datos de corte transversal: los datos están ordenados.
  \item Esto genera varias particularidades que hace que haya que tener \textbf{CUIDADO} a la hora de estimar esos modelos por MCO e interpretarlos como causales.
  \item Uno de ellos es la presencia de tendencias temporales.
  \item Si una serie de tiempo crece o decrece sistemáticamente en el tiempo decimos que tiene tendencia.
  \item ¿Por qué puede ser un problema?
  \begin{itemize}
    \item La altura de un niño aumenta con el tiempo.
    \item La altura de un árbol aumenta con el tiempo.
    \item Si regresamos la altura del niño y del árbol:
    \item ¡vamos a encontrar que están muy correlacionadas pero no hay causalidad!
  \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}{Tipos de tendencia: determinística vs.\ estocástica}

\begin{columns}
  \column{0.5\textwidth}
  \begin{block}{Tendencia determinística}
  La media evoluciona de forma \emph{predecible} con el tiempo:
  \[
  y_t = \beta_0 + \beta_1 t + \varepsilon_t
  \]
  con $\varepsilon_t$ estacionario. La trayectoria esperada es una recta.
  \end{block}

  \column{0.5\textwidth}
  \begin{alertblock}{Tendencia estocástica}
  La serie ``deriva'' por acumulación de choques (caminata aleatoria):
  \[
  y_t = y_{t-1} + \varepsilon_t
  \]
  No hay nivel medio fijo; la dispersión crece con $t$.
  \end{alertblock}
\end{columns}

\vspace{0.3cm}
\begin{itemize}
  \item Ambas producen series que ``suben'' en el tiempo, pero requieren \textbf{tratamientos distintos}: la determinística se quita con un regresor $t$; la estocástica se quita \emph{diferenciando}.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Inclusión de tendencias en los modelos de series temporales}

\begin{itemize}
  \item La "tendencia" de las series es como una variable omitida!
  \item Una manera de solucionar esto es incluir en el modelo una variable que recoja el crecimiento temporal: la tendencia.
  \item Una tendencia lineal \( t \) es una variable que crecerá una unidad cada año, lo más común empezando en 1.
\begin{center}
\begin{tabular}{c c}
\hline
Año & t \\
\hline
1940 & 1 \\
1941 & 2 \\
1942 & 3 \\
\vdots & \vdots \\
1960 & 20 \\
\hline
\end{tabular}
\end{center}
  \item Se pueden ajustar tendencias cuadráticas (\( t^2 \)), o si la variable dependiente está en logs, la tendencia lineal será interpretada como tendencia exponencial.
\end{itemize}

\end{frame}
\begin{frame}
\frametitle{Inclusión de tendencias en los modelos de series temporales}

\begin{itemize}
  \item Cuando incluimos una tendencia determinística, el modelo que estimaremos será:
  \[
  y_t = \alpha_0 + \alpha_1 x_t + \alpha_2 t + \epsilon_t
  \]
  \item \(\alpha_2\) nos dice cuánto varía anualmente \( y_t \):
  \begin{itemize}
    \item \(\alpha_2 > 0\) y significativo: tendencia creciente.
    \item \(\alpha_2 < 0\) y significativo: tendencia decreciente.
  \end{itemize}
  \item Ahora la ecuación la debemos interpretar como “libre de tendencia”.
  \item \(\alpha_1\) nos dirá cuál es la relación entre \( x_t \) y \( y_t \) cuando les “quitamos” la tendencia.
\end{itemize}

\end{frame}

\begin{frame}{¿Por qué incluir $t$ ``limpia'' la correlación espuria?}

\begin{block}{Tendencia $=$ variable omitida (intuición FWL)}
Si tanto $y_t$ como $x_t$ tienen una tendencia común $t$ y \emph{no} la incluimos, $t$ queda en el error: $x_t$ se correlaciona con ese error y $\hat\alpha_1$ recoge la \textbf{co-tendencia}, no la relación causal.
\end{block}

\pause

\begin{exampleblock}{Teorema de Frisch--Waugh--Lovell (idea)}
Estimar $y_t=\alpha_0+\alpha_1 x_t+\alpha_2 t+\epsilon_t$ por MCO es \emph{equivalente} a:
\begin{enumerate}
  \item Regresar $y_t$ sobre $t$ y guardar el residuo $\tilde y_t$ (parte de $y$ libre de tendencia).
  \item Regresar $x_t$ sobre $t$ y guardar el residuo $\tilde x_t$.
  \item Regresar $\tilde y_t$ sobre $\tilde x_t$: el coeficiente es exactamente $\hat\alpha_1$.
\end{enumerate}
Por eso decimos que $\hat\alpha_1$ mide la relación \textbf{una vez removida la tendencia} de ambas variables. \hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}{Estacionalidad}

\begin{itemize}
  \item Muchas series de frecuencia mensual o trimestral exhiben \textbf{estacionalidad}: patrones que se repiten dentro del año (ventas de helado en verano, IMACEC más bajo en febrero, recaudación en diciembre).
  \item Si no la modelamos, queda en el error y puede sesgar la inferencia.
  \item Solución habitual: \textbf{variables dummy estacionales}. Para datos trimestrales (base $=$ Q4):
  \[
  y_t = \beta_0 + \beta_1 x_t + \gamma_1 Q1_t + \gamma_2 Q2_t + \gamma_3 Q3_t + \epsilon_t
  \]
  \item Cada $\gamma_j$ mide el efecto del trimestre $j$ respecto al trimestre base, manteniendo $x_t$ constante.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Persistencia en modelos de series de tiempo}

\begin{itemize}
  \item Muchas veces, las variables del modelo tienen persistencia: $ y_t = \rho y_{t-1} + \epsilon_t $.
  \item Si \(\rho = 1\), se llama una \textbf{caminata aleatoria}:
  \begin{itemize}
    \item Diremos que la serie tiene una raíz unitaria o que es \textbf{integrada de orden 1.}
    \item Tiene tendencia 'estocástica'.
    \item Al igual que la tendencia 'determinística' que vimos antes, cuando tenemos una variable que se comporta de esta manera, vamos a tener que ajustarlo. Pero no va a alcanzar con agregar tendencia en este caso.
  \end{itemize}
  \item Cuando \(|\rho| < 1\), decimos que la serie es un AR(1) \textbf{estacionario} o integrada de orden 0:
  \begin{itemize}
    \item No hay tendencia estocástica.
    \item En este caso, podemos usar MCO sin inconveniente.
  \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}[allowframebreaks]{Demostración: la caminata aleatoria no es estacionaria}

\begin{alertblock}{Proposición}
Sea $y_t = y_{t-1} + \epsilon_t$ con $y_0=0$ y $\epsilon_t$ ruido blanco, $\mathbb{E}[\epsilon_t]=0$, $\text{Var}(\epsilon_t)=\sigma^2$, $\text{Cov}(\epsilon_t,\epsilon_s)=0$ para $t\neq s$. Entonces
\[
\mathbb{E}[y_t]=0, \qquad \text{Var}(y_t)=t\,\sigma^2,
\]
por lo que el proceso \textbf{no es estacionario}: su varianza crece con el tiempo.
\end{alertblock}

\begin{exampleblock}{Demostración}
\textbf{Paso 1: escribir $y_t$ como suma acumulada de choques.}
Iterando la recursión hacia atrás:
\[
y_t = y_{t-1}+\epsilon_t = y_{t-2}+\epsilon_{t-1}+\epsilon_t = \cdots = y_0 + \sum_{j=1}^{t}\epsilon_j = \sum_{j=1}^{t}\epsilon_j.
\]
La caminata aleatoria es literalmente la \emph{acumulación} de todos los choques pasados: ningún choque se ``olvida''.

\framebreak

\textbf{Paso 2: esperanza.}
\[
\mathbb{E}[y_t]=\mathbb{E}\!\left[\sum_{j=1}^{t}\epsilon_j\right]=\sum_{j=1}^{t}\mathbb{E}[\epsilon_j]=0.
\]

\textbf{Paso 3: varianza.} Como los $\epsilon_j$ son no correlacionados,
\[
\text{Var}(y_t)=\text{Var}\!\left(\sum_{j=1}^{t}\epsilon_j\right)=\sum_{j=1}^{t}\text{Var}(\epsilon_j)=\sum_{j=1}^{t}\sigma^2 = t\,\sigma^2.
\]
La varianza \textbf{depende de $t$}: viola la estacionariedad (varianza constante). Además $\text{Cov}(y_t,y_{t-h})=(t-h)\sigma^2$ tampoco depende solo del rezago $h$.

\framebreak

\textbf{Paso 4: la primera diferencia sí es estacionaria.}
Definimos $\Delta y_t = y_t - y_{t-1}$. Sustituyendo:
\[
\Delta y_t = (y_{t-1}+\epsilon_t)-y_{t-1}=\epsilon_t.
\]
Entonces:
\[
\mathbb{E}[\Delta y_t]=0,\quad \text{Var}(\Delta y_t)=\sigma^2 \text{ (constante)},\quad \text{Cov}(\Delta y_t,\Delta y_{t-h})=0.
\]
$\Delta y_t$ es ruido blanco: \textbf{estacionario}. Por eso, cuando una serie es $I(1)$, trabajamos con su \emph{primera diferencia}. \hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}
\frametitle{Series altamente persistentes}

\begin{itemize}
  \item Si \(\rho = 1\), usar estas variables va a introducir correlación espuria entre las series.
  \item Los datos tienen una tendencia en el tiempo, pero no es determinística (crece de forma constante) sino que es estocástica (crece pero de forma variante).
  \item Lo vamos a solucionar tomando la primera diferencia de la serie problemática:
  $
  \Delta y_t = y_t - y_{t-1} = \epsilon_t
  $ y se corre la regresión con la diferencia y no el nivel.
  \item Ejemplo:
$
  salario_t = \beta_0 + \beta_1 productividad_t + \epsilon_t
$
  Supongan que tanto salario como productividad siguen un proceso de raíz unitaria (caminata aleatoria).
  \item Para poder estimar su relación, vamos a transformar el modelo a primeras diferencias.
  Con esto, nos sacamos de encima la persistencia.
\end{itemize}
$
  \Delta salario_t = \beta_0 + \beta_1 \Delta productividad_t + u_t
$
\end{frame}

\section{Ejercicios}

\begin{frame}{Ejercicio Resuelto 1}

\begin{block}{Ejercicio 1 --- Tendencia como variable omitida}
Un analista del Banco Central regresa el \textbf{consumo de electricidad residencial} ($y_t$, en GWh) sobre el \textbf{precio promedio de los smartphones} ($x_t$, en miles de CLP), con datos anuales 1995--2020. Obtiene:
\[
\hat y_t = 120 + 8.4\, x_t, \qquad R^2 = 0.94,\quad t_{\hat\beta_1}=11.2.
\]
¿Es plausible una relación causal? ¿Qué haría usted antes de creerle al modelo?
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Paso 1:} Ambas series \emph{crecen en el tiempo} por razones independientes (más hogares y aparatos $\Rightarrow$ más consumo; mejora tecnológica $\Rightarrow$ precios suben/bajan tendencialmente). El $R^2$ y el $t$ altos reflejan una \textbf{co-tendencia}, no causalidad: \emph{correlación espuria}.

\textbf{Paso 2:} La tendencia $t$ es una variable omitida correlacionada con $x_t$. Incluirla:
\[
y_t = \alpha_0 + \alpha_1 x_t + \alpha_2 t + \epsilon_t.
\]
\textbf{Paso 3:} Se espera que, al ``detrendizar'', $\hat\alpha_1$ caiga drásticamente y deje de ser significativo. La relación original era artefacto de la tendencia común.
\end{exampleblock}

\end{frame}

\begin{frame}{Ejercicio Resuelto 2}

\begin{block}{Ejercicio 2 --- Varianza de una caminata aleatoria}
El tipo de cambio CLP/USD se modela como caminata aleatoria $e_t = e_{t-1}+\epsilon_t$, con $e_0 = 800$, $\epsilon_t$ ruido blanco y $\sigma^2 = 25$ (CLP$^2$). Calcule la desviación estándar de $e_t$ a 1, 4 y 9 períodos. ¿Qué le dice esto sobre pronosticar el dólar a largo plazo?
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Paso 1:} Por la proposición demostrada, $\text{Var}(e_t)=t\sigma^2 = 25\,t$, luego $\text{sd}(e_t)=5\sqrt{t}$.

\textbf{Paso 2:} Evaluamos:
\[
t=1:\ 5\sqrt1 = 5; \quad t=4:\ 5\sqrt4 = 10; \quad t=9:\ 5\sqrt9 = 15.
\]
\textbf{Paso 3:} La incertidumbre crece con $\sqrt{t}$: \emph{nunca} se estabiliza. La mejor predicción puntual es el último valor ($\mathbb{E}[e_{t+h}\mid e_t]=e_t$), pero el intervalo de pronóstico se ensancha sin límite. Por eso los precios de activos en caminata aleatoria son ``impredecibles'' a largo plazo.
\end{exampleblock}

\end{frame}

\begin{frame}[fragile]{Ejemplo Aplicado en R --- Detrendizar el IMACEC}

\begin{exampleblock}{Incluir tendencia lineal en una regresión de series de tiempo}
\begin{verbatim}
# IMACEC mensual vs. ventas del retail (Chile)
datos <- read.csv("imacec_retail.csv")
datos$t <- 1:nrow(datos)   # tendencia lineal 1,2,3,...

# (1) Modelo INGENUO: ignora la tendencia
m0 <- lm(imacec ~ ventas, data = datos)

# (2) Modelo con tendencia determinística
m1 <- lm(imacec ~ ventas + t, data = datos)

summary(m0)   # beta de 'ventas' suele salir grande y signif.
summary(m1)   # al detrendizar, el efecto se reduce mucho
\end{verbatim}
\end{exampleblock}
{\small Compare el coeficiente de \texttt{ventas} entre \texttt{m0} y \texttt{m1}: si cambia mucho, había correlación espuria por tendencia.}

\end{frame}

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 19}
\begin{enumerate}
    \item En series de tiempo los datos están \textbf{ordenados}: el orden temporal abre la puerta a correlaciones espurias y exige cuidado al interpretar MCO como causal.
    \item Una \textbf{tendencia determinística} se elimina incluyendo $t$ como regresor; el coeficiente de interés se reinterpreta ``libre de tendencia'' (FWL).
    \item Una \textbf{tendencia estocástica} (caminata aleatoria, $\rho=1$, $I(1)$) \emph{no} se arregla con $t$: hay que \textbf{diferenciar}.
    \item La caminata aleatoria no es estacionaria: $\text{Var}(y_t)=t\sigma^2$. Su primera diferencia $\Delta y_t=\epsilon_t$ sí lo es.
    \item La estacionalidad se modela con dummies; la persistencia, con AR(1) y diferenciación.
\end{enumerate}
\end{block}
