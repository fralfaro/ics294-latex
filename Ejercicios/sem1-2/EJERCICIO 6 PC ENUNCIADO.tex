\section{Ejercicios}
\begin{enumerate}

\item Para explicar el precio de una vivienda (\emph{precio}) se estima el modelo
        \[
          \text{precio}
          = 30000
          + 900\,\text{msq}
          + 5000\,\text{dorm}
          + 60\,(\text{msq}\times\text{dorm})
          + u ,
        \]
        donde \texttt{msq} denota los metros cuadrados de la vivienda y
        \texttt{dorm} el número de dormitorios.

        \begin{enumerate}[label=\alph*)]
          \item Escriba la expresión del \emph{efecto marginal} sobre el precio de
                \emph{agregar un dormitorio}, manteniendo constante \texttt{msq}.
          \item Explique cómo cambia ese efecto si se evalúa en (i) una vivienda de
                tamaño medio $\overline{\text{msq}}=140m^2$ y (ii) una vivienda situada
                en el percentil 90 de metros cuadrados, $\text{msq}_{90}=220m^2$.
        \end{enumerate}

        


 \item Para una muestra de egresados universitarios se estima la regresión
        \[
          \log(\text{salario}_i)
          = 0.55 \;+\; 0.075\,\text{educ}_i \;+\; u_i ,
        \]
        donde \texttt{educ} representa los años de educación formal.

        \begin{enumerate}[label=\alph*)]
          \item Interprete el coeficiente $\beta_{educ}$.
          \item Mencione una interpretación errónea que se debe evitar al
                comunicar este resultado a un público no técnico.
        \end{enumerate}

        
\item Dada la matriz
        \[
          (X'X)^{-1} =
          \begin{bmatrix}
          0.05 & -0.015 & 0.002 \\
          -0.015 & 0.03 & 0.004 \\
          0.002 & 0.004 & 0.02
          \end{bmatrix},
          \quad \hat{\sigma}^2 = 80,
        \]
        y los coeficientes estimados
        \[
          \hat{\beta} = \begin{bmatrix} 5 \\ 4 \\ 1 \end{bmatrix},
        \]
        testee la hipótesis:
        \[
          H_0: \beta_1 = 2\beta_2
          \quad \text{versus} \quad
          H_1: \beta_1 \neq 2\beta_2
        \]
        al 5\% de significancia.

        

\item Se tiene un modelo del tipo:
$$y = \beta_0 + \beta_1x_1 + \beta_2x_2+\beta_3x_3 + u$$
pero se ha identificado dos grupos, por lo que se plantea un modelo separado para cada grupo:
$$y_g = \beta_{0g} + \beta_{1g}x_1 + \beta_{2g}x_2+\beta_{3g}x_3 + u_g$$
con $g\in {1,2}$, obteniendo los siguentes resultados desde una muestra de 150 datos:
\begin{center}
    \begin{tabular}{|c|c|}
    \hline
        Modelo & SSR \\
        \hline
        Base & 300 \\
        \hline
        1 & 155 \\
        \hline
        2 & 130 \\
        \hline
    \end{tabular}
\end{center}
¿Existen diferencias en y para cada grupo? Use una significancia del 5\%. ($F_{0.95,(4,142)}=2.44$)



\item Considere el modelo de regresión lineal
\[
y_i \;=\; \beta_0 \;+\; \beta_1 x_i \;+\; \beta_2 D_i \;+\; \beta_3 (D_i x_i) \;+\; u_i,
\]
donde
\begin{itemize}
  \item $x_i$ es una variable continua,
  \item $D_i$ es una \emph{dummy} que vale $1$ si la observación pertenece al grupo $A$ y $0$ si pertenece al grupo $B$,
  \item $u_i$ es el término de error con $E[u_i \mid x_i,D_i]=0$.
\end{itemize}

\begin{enumerate}
  \item[(a)] Determine la pendiente $\partial y / \partial x$ para cada grupo ($D=0$ y $D=1$).
  \item[(b)] Interprete el coeficiente $\beta_3$ en términos de la diferencia entre grupos.
  \item[(c)] Suponga que al estimar el modelo se obtiene $\widehat{\beta}_3=-1.2$ y que es estadísticamente significativo al $5\%$.  
        ¿Cómo cambia la relación entre $x$ e $y$ al pasar de $D=0$ a $D=1$?
\end{enumerate}


\end{enumerate}
