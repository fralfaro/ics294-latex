\section{Ejercicios}
\begin{enumerate}
    \item Supongamos un modelo de Regresión Lineal Simple, que describe el salario en función de la educación de la forma:
    $$salary = \beta_0 + \beta_1 educ$$
    Si la solución mediante MCO, los entrega la siguiente forma:
        $$\hat{salary} = 10 + 2\cdot educ + \hat{u}$$
    siendo $\hat{u}$ el error observado.
    \begin{enumerate}
        \item  ¿Como se interpreta el valor de $\hat{\beta_1}$?  \\
        \textbf{el valor nos indica que el salario aumenta en 2 por cada año adicional de educación}

        \item Si dada la gran escala de $salary$, se le aplica $log()$, para tener valores más faciles de trabajar, encontrando la siguente RLS mediante MCO:
        $$\hat{log({salary})} = 2.8 + 0.07\cdot educ + \hat{u}$$
        ¿Como se interpreta el valor de $\hat{\beta_1}$ en este caso?  \\
        \textbf{El valor nos indica que el salario aumenta 7\% por cada 1 año adicional de educación}
        \item Ahora, se le aplica $log()$ también a educ, para tener valores más faciles de trabajar, encontrando la siguente RLS mediante MCO:
        $$\hat{log({salary})} = 1.4 + 0.80\cdot log(educ) + \hat{u}$$
        ¿Como se interpreta el valor de $\hat{\beta_1}$ en este caso? ¿En economia, como se le llama a este valor?\\
        \textbf{$\hat{\beta_1}$ en esta caso nos indica el porcentaje de aumento del salario al aumentar 1\% la educación, es decir, aumenta 0.8\% al aumentar 1\% la educación. Este valor es conocido como la elasticidad.}
    \end{enumerate}


\item Se tiene un modelo que explica el puntaje en la PAES dado el IQ del individuo, de la sigueinte forma:
$$Puntaje = 200 + 10 \cdot IQ -  0.05IQ^2$$
\begin{enumerate}
    \item ¿Cual es el efecto marginal del IQ en el puntaje?
    \textbf{$$\frac{\Delta puntaje}{\Delta IQ} = 10-0.1IQ$$ }
    \item ¿Cual es el valor donde el efecto marginal del IQ es cero?
    \textbf{$10-0.1IQ=0 \implies IQ = 100$}
    \item ¿Que sucede para valores de IQ mayores a 100?
    \textbf{El efecto del IQ en el puntaje empieza a ser negativo}
\end{enumerate}

\item Se tiene la siguiente RLS:
$$y = \beta_0 + \beta_1x$$
mediante MCO podemos encontrar $\hat{\beta_0}$ y $\hat{\beta_1}$ insesgados y eficientes. Si reescalamos $\Tilde{x} = 10x$. 
\begin{enumerate}
    \item Demuestre que $\hat{\beta_1}=10\hat{\Tilde{\beta_1}}$. \\
    \textbf{La RLS podemos definirla como:
    $$\hat{y} = \hat{\beta_0} + \hat{\Tilde{\beta_1}}\Tilde{x}+u = \hat{y} = \hat{\beta_0} + \hat{\Tilde{\beta_1}}(10x)+u$$
    por MCO sabemos que:
    $$\hat{\Tilde{\beta_1}}=\frac{Cov(10x,y)}{Var(10x)}=\frac{10Cov(x,y)}{100Var(x)}=\frac{1}{10}\frac{Cov(x,y)}{Var(x)}$$
    y sabemos que:
    $$\frac{Cov(x,y)}{Var(x)} = \hat{\beta_1}$$
    entonces:
    $$\hat{\Tilde{\beta_1}} = \frac{1}{10}\hat{\beta_1} \implies \hat{\beta_1} = 10\hat{\Tilde{\beta_1}}$$}
\item Demuestra que el valor $t$ del estimador no cambia. \\
\textbf{ Sea el t del estimador:
$$t_{\hat{\beta_1}} = \frac{\hat{\beta_1}}{s.e.(\hat{\beta_1})}$$
y para $\hat{\Tilde{\beta_1}}$ se tiene:
$$t_{\hat{\Tilde{\beta_1}}} = \frac{\hat{\Tilde{\beta_1}}}{s.e.(\hat{\Tilde{\beta_1}})}$$
Notames que 
$$s.e.(\hat{\Tilde{\beta_1}}) = \sqrt{Var(\hat{\Tilde{\beta_1}})}=\sqrt{Var(\frac{1}{10}\hat{{\beta_1}})}=\frac{1}{10}\sqrt{Var(\hat{{\beta_1}})}$$
entonces:
$$t_{\hat{\Tilde{\beta_1}}} = \frac{\frac{1}{10}\hat{{\beta_1}}}{\frac{1}{10}\sqrt{Var(\hat{{\beta_1}})}} = t_{\hat{\beta_1}}$$
}
    
\end{enumerate}

\\
\textbf{Primero se tiene que entrar el valor crítico de t (dos colas) para el nivel de significancia y los grados de libertad. Tenemos: 
$$t_{1-\alpha/2,n-k-1}=t_{0.975,668} = 1.96$$

}

\item Haga el intervalo de confianza para el estimador de la variable $attend$, al 95\% de confianza. Concluya.\\
\textbf{ el IC esta dado por:
$$[\hat{\beta} - t_{1-\alpha/2,n-k-1}s.e(\hat{\beta});\hat{\beta} + t_{1-\alpha/2,n-k-1}s.e(\hat{\beta})]$$
por tanto reemplazando tenemos:
$$[-0.06926 - 1.96\cdot 0.04074;-0.06926 - 1.96\cdot 0.04074]$$
$$[-0.149;0.011]$$
Como ven el intervalo contiene el cero, por lo que con un 95\% el estimador podría ser cero.}

\item Haga el intervalo de confianza para el estimador de la variable $termGPA$, al 95\% de confianza. Concluya.\\
\textbf{ el IC esta dado por:
$$[\hat{\beta} - t_{1-\alpha/2,n-k-1}s.e(\hat{\beta});\hat{\beta} + t_{1-\alpha/2,n-k-1}s.e(\hat{\beta})]$$
por tanto reemplazando tenemos:
$$[3.54031 - 1.96\cdot 0.31395;3.54031 + 1.96\cdot 0.31395]$$
$$[2.925;4.156]$$
Con un 95\% de confianza el estimador de $termGPA$ está entre 2.925 y 4.156.}
\end{enumerate}
