\section{Preguntas}

 
\begin{enumerate}

    \item (50 puntos) En una regresión lineal simple, el coeficiente asociado a la variable
        \(X_1\) fue estimado en:
        \[
          \hat\beta_1 = 3.2,
          \qquad \hat{Var}(\hat\beta_1) = 1.7,
          \qquad n = 300.
        \]
        Considere un nivel de significancia del 5\%.

        \begin{enumerate}[label=\alph*)]
          \item (15 puntos) Plantee las hipótesis nula y alternativa para evaluar si
                \( \beta_1 \) es significativamente distinto de cero.
          \item (20 puntos) Calcule el estadístico \( t \) de prueba, compare su valor
                con el valor crítico correspondiente y determine si se
                rechaza la hipótesis nula al 1\% de significancia.
          \item (15 puntos) Construya un intervalo de confianza al 95\% para
                \( \beta_1 \) e interprete.
        \end{enumerate}

        \textbf{Solución.}
        \begin{enumerate}[label=\alph*)]
          \item Hipótesis a testear:
                \[
                  H_0: \beta_1 = 0
                  \qquad\text{versus}\qquad
                  H_1: \beta_1 \neq 0.
                \]
          \item El estadístico \( t \) es:
                \[
                  t
                  = \frac{\hat\beta_1 - 0}{s.e.(\hat\beta_1)}
                  = \frac{3.2}{\sqrt{1.7}}
                  = 2.45
                \]
                Con 298 grados de libertad (suponiendo pocos regresores)
                y \(n = 300\), podemos aproximar el valor crítico bilateral
                usando la normal estándar:
                \[
                  t_{0.5\%} \approx 2.58.
                \]
                Dado que \(|2.45| < 2.58\), \textbf{ no se rechaza} \( H_0 \).
                Existe evidencia suficiente para afirmar que \( \beta_1 \) no es
                significativamente distinto de cero al 1\% de significancia.

          \item El intervalo de confianza al 95\% es:
                \[
                  \left[
                    \hat\beta_1 - t_{2.5\%} \times s.e.(\hat\beta_1),
                    \hat\beta_1 + t_{2.5\%} \times s.e.(\hat\beta_1)
                  \right]
                  =
                  \left[
                    3.2 - 1.96 \times \sqrt{1.7},
                    3.2 + 1.96 \times \sqrt{1.7}
                  \right]
                  =
                  \left[
                    0.644; 5.756
                  \right].
                \]
                \textbf{Interpretación:} Con 95\% de confianza, se espera que
                el verdadero efecto de \( X_1 \) sobre la variable dependiente
                esté entre 0.644 y 5.756 unidades. Es decir, un aumento en
                \( X_1 \) está asociado, en promedio, a un aumento positivo y
                significativo en la variable dependiente.
        \end{enumerate}


\item (50 puntos) Se tiene un modelo de regresión lineal de la forma:
$$y=\beta_0 + \beta_1 x_1+\beta_2x_2+\beta_3 x3 + u$$
los resultados de la regresión con 250 datos son los siguientes:
        \[
          \hat{\beta} = \begin{bmatrix} 3 \\ 18 \\ 5 \\2 \end{bmatrix},
        \]
y 
        \[
          (X'X)^{-1} =
          \begin{bmatrix}
0.5 & 0  & 0  &  0\\
0 & 0.5  & 0  & -0.5\\
0 & 0  & 1  & -0.5\\
0 & -0.5 & -0.5 &  0.5
\end{bmatrix},
          \quad \hat{\sigma}^2 = 2
        \]
\begin{enumerate}
    \item (15 puntos) Plantee la hipótesis que testea que $\beta_1=4\beta_3$.

    \item (35 puntos) ¿Se puede rechazar la hipótesis nula con la información que disponemos con una significancia del 5\%? 
\end{enumerate}

\textbf{solución}
\begin{enumerate}
    \item $$H_0 : \beta_1-4\beta_3 = 0$$
$$H_1 : \beta_1-4\beta_3 \neq 0$$
\item el estadistico t esta dado por:
$$t = \frac{\beta_1-4\beta_3}{s.e.(\beta_1-4\beta_3)}$$
y $s.e.(\beta_1-4\beta_3)$ esta dado por:
$$s.e.(\beta_1-4\beta_3)=\sqrt{Var(\beta_1-4\beta_3)}=\sqrt{2\times Var(\beta_1)+16\times2\times Var(\beta_3)-8\times 2 \times Cov(\beta_1,\beta_3)}$$
$$\sqrt{Var(\beta_1)+16Var(\beta_3)-8Cov(\beta_1,\beta_3)}=\sqrt{1+16\cdot 1-8 \cdot (-1)}=\sqrt{25}=5$$
por tanto 
$$t = \frac{18-4\cdot2}{5}=2$$
y con el valor $t_{1-\alpha/2,246}=1.96$. Se rechaza $H_0$, y $\beta_1$ es distinto a $4\beta_3$.
\end{enumerate}



\end{enumerate}
