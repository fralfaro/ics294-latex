\begin{frame}[shrink=10]
\frametitle{Incorporación de no-linealidades}
    \begin{enumerate}
        \item {\color{blue}Variable dependiente e independiente en niveles:}
        \begin{itemize}
            \item[] \begin{center}$y_i = \beta_0 + \beta_1 x_{1,i} + \ldots + u_i$\end{center}
            \item → \textbf{Ejemplo:} $salario_i = \beta_0 + \beta_1 educ_i + u_i$
            \item $\beta_1 = \frac{\Delta y}{\Delta x_1}$
            \item Por cada año más de educación se obtiene, en promedio, $\beta_1$ dólares más de salario.
        \end{itemize}
        \item {\color{blue}Variable dependiente en log e independiente en niveles:}
        \begin{itemize}
            \item[] \begin{center} $\log(y_i) = \beta_0 + \beta_1 x_{1,i} + \ldots + u_i$\end{center}
            \item → \textbf{Ejemplo:} $\log(salario)_i = \beta_0 + \beta_1 educ_i + u_i$
            \item $\beta_1 = \frac{\Delta \log y}{\Delta x_1}\approx \frac{\Delta \pc  y}{100\Delta x_1}$
            \item Por cada año más de educación se obtiene, en promedio, un aumento de $100 \cdot \beta_1 \pc$ del salario, manteniendo todas las demás variables constantes.
        \end{itemize}
        \item {\color{blue}Variable dependiente e independiente en logs:}
        \begin{itemize}
            \item[] \begin{center}$\log(y_i) = \beta_0 + \beta_1 \log(x_{1,i}) + \ldots + u_i$\end{center}
            \item → \textbf{Ejemplo:} $\log(salario)_i = \beta_0 + \beta_1 \log(educ)_i + u_i$
            \item $\beta_1 = \frac{\Delta \log y}{\Delta \log x_1}\approx \frac{\Delta\pc  y}{ \Delta\pc  x_1}$
            \item Un aumento de 1\% de los años de educación implica, en promedio, un aumento de $\beta_1 \pc$ del salario manteniendo todas las demás variables constantes. Es decir, representa una {\bf elasticidad}.
        \end{itemize}
    \end{enumerate}


\end{frame}

\section{Variables en logaritmo}

\begin{frame}
\frametitle{Variables en logaritmo}

\begin{itemize}
    \item Diferencias logarítmicas:
    \[
    \Delta \log(X)= \log(X_1)-\log(X_0)= \log\left (\frac{ X_1}{X_0}\right )=\log\left(1+{\Delta X\over X_0} \right)
    \]
    \item Recuerde que en nuestro curso y en el libro $log(X)=ln(X)$.
    \item Sabemos que\footnote{Serie de Taylor centrada en 0, es decir, es una buena aproximación para x cercanos a 0.}:

    \[
    \log(1+x) \approx x
    \]

\end{itemize}



\end{frame}

%========================= DERIVACIÓN APROXIMACIÓN LOG =========================
\begin{frame}[allowframebreaks]{¿Por qué $100\cdot\beta_1$ es el cambio porcentual? (log-lin)}

\begin{alertblock}{Proposición: interpretación porcentual del modelo log-lin}
En el modelo $\log(y) = \beta_0 + \beta_1 x + u$, manteniendo $u$ constante, un aumento de $\Delta x = 1$ produce un cambio porcentual en $y$ de aproximadamente $100\cdot\beta_1\,\pc$, con corrección exacta $100\cdot(e^{\beta_1}-1)\,\pc$.
\end{alertblock}

\begin{exampleblock}{Demostración}
\textbf{Paso 1.} Diferenciando $\log(y) = \beta_0 + \beta_1 x$ ante un cambio $\Delta x$ (con $u$ fijo):
\[
\Delta \log(y) = \beta_1\,\Delta x .
\]

\textbf{Paso 2.} Por definición de diferencia logarítmica, si $y$ pasa de $y_0$ a $y_1$:
\[
\Delta \log(y) = \log(y_1) - \log(y_0) = \log\!\left(\frac{y_1}{y_0}\right) = \log\!\left(1 + \frac{\Delta y}{y_0}\right).
\]

\textbf{Paso 3.} Usando la aproximación de Taylor $\log(1+z)\approx z$ para $z$ pequeño (con $z=\Delta y / y_0$):
\[
\Delta \log(y) \approx \frac{\Delta y}{y_0} = \frac{\Delta\pc  y}{100}.
\]

\textbf{Paso 4.} Igualando los pasos 1 y 3:
\[
\frac{\Delta\pc  y}{100} \approx \beta_1 \,\Delta x \;\;\Longrightarrow\;\; \Delta\pc  y \approx 100\cdot\beta_1\,\Delta x .
\]
Para $\Delta x = 1$ se obtiene $\Delta\pc  y \approx 100\cdot\beta_1$. \hfill $\square$
\end{exampleblock}

\framebreak

\begin{block}{Corrección exacta}
La aproximación $\log(1+z)\approx z$ pierde precisión cuando $\beta_1$ es grande. El cambio porcentual \emph{exacto} ante $\Delta x = 1$ se obtiene sin aproximar:
\[
\Delta \log(y) = \beta_1 \;\Longrightarrow\; \frac{y_1}{y_0} = e^{\beta_1}
\;\Longrightarrow\; \frac{\Delta y}{y_0} = e^{\beta_1} - 1 .
\]
Por lo tanto:
\[
\boxed{\;\Delta\pc  y \text{ (exacto)} = 100\cdot\left(e^{\beta_1}-1\right)\;}
\]
\end{block}

\begin{tcolorbox}[colback=blue!5!white, colframe=blue!75!black, title=Ejemplo numérico]
Si $\beta_1 = 0{,}083$:\quad aproximación $= 8{,}3\pc$; \quad exacto $= 100\cdot(e^{0{,}083}-1) = 8{,}65\pc$.\\
Si $\beta_1 = 0{,}30$:\quad aproximación $= 30\pc$; \quad exacto $= 100\cdot(e^{0{,}30}-1) = 34{,}99\pc$ (la aproximación ya falla).
\end{tcolorbox}

\end{frame}

\begin{frame}
\frametitle{Variables en logaritmo: Caso 2 - Elasticidad}

\begin{itemize}
    \item Cuando $x_j$ e $y$ se encuentran en logaritmo: un cambio de un 1\% en $x_j$ se asocia a un cambio de $\beta_j$\% en $y$.
    \item Por ejemplo, se estima la elasticidad de la demanda de vino en Chile.
    \[
    \log(\text{cantidad vino}) = 6 - 0,2 \log(\text{precio vino})
    \]
    \item Un aumento de un 1\% en el precio del vino corresponde a una caída del 0,2\% en la cantidad consumida de vino.
    \item Se interpreta como una elasticidad. Por ejemplo, la elasticidad de la demanda de vino con respecto al precio es 0,2.
\end{itemize}

\end{frame}

\begin{frame}{Las cuatro formas funcionales: resumen de interpretación}

\begin{table}[H]
    \centering
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{l|c|c|l}
       \textbf{Modelo}  & \textbf{Dep.} & \textbf{Indep.} & \textbf{Interpretación de }$\beta_1$ \\ \hline
       Nivel-nivel  & $y$ & $x$ & $\Delta y = \beta_1\,\Delta x$ (unidades de $y$) \\
       Nivel-log  & $y$ & $\log(x)$ & $\Delta y = \dfrac{\beta_1}{100}\,(\Delta\pc  x)$ \\
       Log-nivel  & $\log(y)$ & $x$ & $\Delta\pc  y = (100\,\beta_1)\,\Delta x$ \;(semielast.) \\
       Log-log  & $\log(y)$ & $\log(x)$ & $\Delta\pc  y = \beta_1\,(\Delta\pc  x)$ \;(elasticidad) \\
    \end{tabular}
    \caption{Interpretación del coeficiente según la forma funcional.}
\end{table}

\begin{block}{Mnemotecnia}
El logaritmo convierte la variable a la que afecta en \textbf{términos porcentuales}: si $y$ está en log $\Rightarrow$ el efecto sobre $y$ es \%; si $x$ está en log $\Rightarrow$ la causa es un cambio \% en $x$.
\end{block}

\end{frame}

\section{Formas cuadráticas}

\begin{frame}[shrink=10]
\frametitle{Formas cuadráticas}

\begin{itemize}
    \item El Modelo de Regresión Lineal se refiere a una relación lineal entre los coeficientes que acompañan a los regresores $X$. Pero las $X$ pueden tomar cualquier forma.
    \item Considere el modelo:
    \[
    y = \beta_0 + \beta_1 x + \beta_2 x^2 + u
    \]
    \item ¿Cuál es la interpretación de $\beta_1$ y $\beta_2$?
    \item \textcolor{blue}{Siempre piensen en términos de derivadas parciales con respecto a la variable explicativa.}
    \item En este caso, la predicción de $y$ es:
    \[
    \hat{y} = \hat{\beta}_0 + \hat{\beta}_1 x + \hat{\beta}_2 x^2
    \]
    Lo que implica que:
    \[
    \Delta \hat{y} \approx (\hat{\beta}_1 + 2 \hat{\beta}_2 x) \Delta x \Rightarrow \frac{\Delta \hat{y}}{\Delta x} \approx \hat{\beta}_1 + 2 \hat{\beta}_2 x
    \]
    \item Noten que ahora un cambio en $x$ genera un cambio esperado en $y$ que depende del nivel de $x$: el efecto parcial de $x$ en $y$ es variable.
    \item $\hat{\beta}_1$ es aproximadamente la pendiente de la relación entre $x$ y $y$ cuando $x = 0$.
\end{itemize}

\end{frame}

%========================= DERIVACIÓN EFECTO MARGINAL Y PUNTO DE QUIEBRE =========================
\begin{frame}{Efecto marginal y punto de quiebre $x^*$}

\begin{alertblock}{Proposición: efecto marginal en el modelo cuadrático}
En $\hat{y} = \hat{\beta}_0 + \hat{\beta}_1 x + \hat{\beta}_2 x^2$, el efecto marginal de $x$ es
\[
\frac{\partial \hat{y}}{\partial x} = \hat{\beta}_1 + 2\hat{\beta}_2 x,
\]
y existe un punto de quiebre $x^*$ donde el efecto cambia de signo, dado por $x^* = -\dfrac{\hat{\beta}_1}{2\hat{\beta}_2}$.
\end{alertblock}

\begin{exampleblock}{Demostración}
\textbf{Paso 1.} Derivamos $\hat{y}$ respecto de $x$ (tratando $x^2$ con la regla de la potencia):
\[
\frac{\partial \hat{y}}{\partial x} = \hat{\beta}_1 + 2\hat{\beta}_2 x .
\]
\textbf{Paso 2.} El efecto marginal cambia de signo donde se anula. Igualando a cero:
\[
\hat{\beta}_1 + 2\hat{\beta}_2 x^* = 0 \;\Longrightarrow\; x^* = -\frac{\hat{\beta}_1}{2\hat{\beta}_2}.
\]
\textbf{Paso 3.} Si $\hat{\beta}_2 < 0$ (y $\hat{\beta}_1>0$), $x^*$ es un \textbf{máximo}: el efecto es positivo para $x<x^*$ y negativo para $x>x^*$. Si $\hat{\beta}_2>0$, $x^*$ es un \textbf{mínimo}. \hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}
\frametitle{Ejemplo: efecto de experiencia en salario}

\begin{itemize}
    \item Considere la siguiente ecuación:
    \[
    \hat{salario} = 3.73 + 0.298\,\text{exper} - 0.0061\,\text{exper}^2
    \]
    \item La experiencia tiene un efecto positivo en el salario pero este efecto va cayendo para niveles altos de experiencia (\(\hat{\beta}_2\) negativo).
    \item ¿Tiene sentido económico esto?
    \item Específicamente, el efecto parcial de la experiencia en el salario es:
\[
\triangle \hat{salario} / \triangle \text{exper} = 0.298 - 2 \cdot 0.0061\, \text{exper}
\]
    \item Cuando \(\beta_1\) y \(\beta_2\) tienen distinto signo, hay un valor de \(x\) a partir del cual el efecto \textit{ceteris paribus} de \(x\) en \(y\) cambia de dirección.
    \item En el ejemplo, este valor es \(x^* = \left|\frac{\beta_1}{2\beta_2}\right| = \frac{0.298}{2\cdot 0.0061} = 24.4\) años.
    \item Otros polinomios de mayor grado también pueden ser incluidos en un MRL.
\end{itemize}

\end{frame}

\]
\textbf{(a)} ¿Cuál es el efecto marginal de un año más de experiencia? \quad \textbf{(b)} ¿En qué nivel de experiencia el salario alcanza su máximo? \quad \textbf{(c)} Interprete el signo de $\hat\beta_2$.
\end{block}

\end{exampleblock}

\section{Términos de interacción}

\begin{frame}[shrink=10, t]
\frametitle{Términos de interacción}

\begin{itemize}
    \item En ciertas ocasiones el efecto parcial de un regresor \(x\) en la variable dependiente \(y\) puede depender del valor de otro regresor \(z\).
    \item En este caso, se incluye en la regresión un término de interacción \(x \times z\).
    \item Ejemplo: suponga el siguiente modelo para el precio de una vivienda
    \[
    \text{precio} = \beta_0 + \beta_1\text{msq} + \beta_2\text{dorm} + \beta_3\,\text{msq} \cdot \text{dorm} + u
    \]
    donde \text{msq} son los metros cuadrados de la vivienda y \text{dorm} el número de dormitorios.
    \item En este caso se cumple que
    \[
    \frac{\triangle \text{precio}}{\triangle \text{dorm}} = \beta_2 + \beta_3\,\text{msq}
    \]
    \item Si \(\beta_2 > 0\) y \(\beta_3 > 0\), la ecuación implica que un dormitorio adicional se relaciona con un incremento en el valor de la vivienda, y estos incrementos son mayores en viviendas de más metros cuadrados \text{msq}.
\end{itemize}

\end{frame}



\begin{frame}
\frametitle{Términos de interacción}

\begin{itemize}
    \item Para resumir el efecto de \text{dorm} en \text{precio} se puede evaluar \(\frac{\triangle \text{precio}}{\triangle \text{dorm}}\) en valores interesantes de \text{msq}, como su media, mediana o algún percentil extremo.
    \item Por ejemplo, sea el tamaño de la casa promedio \(\bar{\text{msq}}\), el efecto de un \text{dorm} más:
    \[
    \frac{\triangle \text{precio}}{\triangle \text{dorm}} = \beta_2 + \beta_3\bar{\text{msq}}. \quad
    \]
    \item \textbf{Cuidado con la interpretación de los efectos principales:} con la interacción, $\beta_2$ ya \emph{no} es ``el efecto de un dormitorio'', sino el efecto de un dormitorio cuando $\text{msq}=0$ (carente de sentido económico). Por eso conviene evaluar en valores relevantes.
\end{itemize}

\end{frame}

\begin{frame}{Ejercicio Resuelto 3: interacción metros$\times$dormitorios}

\begin{block}{Ejercicio 3 (adaptado de Ejercicios/T2/Ejercicios 6)}
Para 500 viviendas se estima (precio en CLP):
\[
\widehat{\text{precio}} = 30{.}000 + 900\,\text{msq} + 5{.}000\,\text{dorm} + 60\,(\text{msq}\cdot\text{dorm})
\]
La mediana es $\widetilde{\text{msq}}=140$ y el percentil 90 es $\text{msq}_{90}=220$. ¿Cuánto sube el precio esperado al agregar un dormitorio en una vivienda mediana y en una grande?
\end{block}

\pause

\begin{exampleblock}{Solución}
El efecto marginal de un dormitorio depende del tamaño:
\[
\frac{\partial \widehat{\text{precio}}}{\partial \text{dorm}} = 5{.}000 + 60\,\text{msq}.
\]
\textbf{Vivienda mediana} ($\text{msq}=140$): $\;5{.}000 + 60(140) = \mathbf{13{.}400}$ CLP.\\
\textbf{Vivienda grande} ($\text{msq}=220$): $\;5{.}000 + 60(220) = \mathbf{18{.}200}$ CLP.\\[4pt]
\textbf{Interpretación:} un dormitorio adicional vale más en casas grandes; el signo positivo de $\hat\beta_3=60$ refleja la \textbf{complementariedad} entre espacio y número de dormitorios.
\end{exampleblock}

\end{frame}

\section{Reescalamiento de variables}

\begin{frame}
\frametitle{Reescalando variables}
¿Qué pasa si re-escalamos \(X\) o \(Y\)? Por ejemplo, para cambiar su unidad de medida?
\begin{itemize}

    \item Los coeficientes estimados, sus varianzas, estadísticos \(t\) y \(F\), cambian de manera tal que se preservan los efectos medidos y los resultados de los tests.
    \item En general, se reescalan variables por motivos estéticos, como reducir el número de ceros en los decimales de una estimación de un coeficiente.
    \item Este procedimiento es usual con variables económicas cuando los montos medidos son grandes o para transformar entre monedas de diferentes países.
\end{itemize}

\end{frame}
\begin{frame}
\frametitle{Ejemplo: reescalando \(Y\)}

\begin{itemize}
    \item Considere el siguiente modelo:
    \[
    {salario_{us\$}} = \beta_0 + \beta_1\text{educacion} + u
    \]
    donde ${salario_{us\$}}$ es medido en dólares.
    \item Quiero estimarlo en pesos a TC=1000 (i.e. \({salario_{chile}} = 1000 \cdot {salario_{us\$}}\)).
    \item Al multiplicar el modelo de los dos lados se obtiene
    \[
    1000 \cdot {salario_{us\$}} = 1000 \beta_0 + 1000 \beta_1 \text{educ} + 1000 \cdot u \quad
    \]
    \[
    {salario_{chile}} = \tilde{\beta}_0 + \tilde{\beta}_1 \text{educ} + \tilde{u} \quad
    \]
    donde \textcolor{red}{\(\tilde{\beta}_j = 1000 \beta_j\).}
    \item Interpretación. Asuma \(\beta_j = 20\)
    \begin{enumerate}
        \item Modelo original: un aumento de un año de educación implica un aumento de 20 dólares en el salario promedio.
        \item Modelo nuevo: un aumento de un año de educación implica un aumento de 20000 pesos en el salario promedio.
    \end{enumerate}
\end{itemize}
\text{}
\end{frame}


\begin{frame}{Reescalando un regresor X }
            \[ y = \beta_0 + \beta_1 x + u\]
Si re-escala la variable $x$ por $\tilde{x} = \alpha x$ se estima:
            \[ y = \beta_0 + \tilde{\beta}_1 \tilde{x} + u\]
        $\Rightarrow$
        \begin{itemize}
            \item $\hat{\tilde{\beta}}_1 = \frac{\hat{\beta}_1}{\alpha}$
            \item $\text{Var}(\hat{\tilde{\beta}}_1) = \dfrac{\text{Var}(\hat{\beta}_1)}{\alpha^2}$
            \item El estadístico $t$ \textbf{no} cambia: $\;t = \dfrac{\hat{\tilde\beta}_1}{\text{ee}(\hat{\tilde\beta}_1)} = \dfrac{\hat\beta_1/\alpha}{\text{ee}(\hat\beta_1)/\alpha}$.
            \item $\hat{\sigma}^2$, SSR, SST, $R^2$ \textbf{no} cambian (no se altera el ajuste).
        \end{itemize}
\end{frame}

%========================= RESUMEN FINAL =========================
\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 9}
\begin{enumerate}
    \item El logaritmo transforma a interpretación \textbf{porcentual}: log en $y\Rightarrow$ efecto en \%; log en $x\Rightarrow$ causa en \% (log-log $=$ elasticidad).
    \item En log-lin, $100\cdot\beta_1$ aproxima el cambio \%; la versión exacta es $100\,(e^{\beta_1}-1)$, relevante cuando $\beta_1$ es grande.
    \item En modelos cuadráticos el efecto marginal es $\beta_1+2\beta_2 x$ y el punto de quiebre es $x^*=-\beta_1/(2\beta_2)$.
    \item Con interacciones, el efecto parcial de una variable depende del nivel de otra: evalúe en media/mediana/percentil.
    \item Reescalar variables cambia magnitudes de coeficientes y varianzas, pero \textbf{no} altera los estadísticos $t$, $F$ ni el $R^2$.
\end{enumerate}
\end{block}
