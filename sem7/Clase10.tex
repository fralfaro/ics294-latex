\section{Distribución de los estimadores}

\begin{frame}
\frametitle{Teorema de Gauss-Markov}
\begin{itemize}
    \item \textbf{Teorema:} Bajo los supuestos 1 a 5, el estimador de Mínimos Cuadrados Ordinarios (MCO) de $\beta$ es de mínima varianza entre los estimadores lineales insesgados de $\beta$.
    \item El estimador MCO es el estimador lineal insesgado de menor varianza (MELI / BLUE).
    \item \textcolor{red}{Pero Gauss-Markov NO nos dice cómo se \emph{distribuye} $\hat{\beta}_j$: solo conocemos su esperanza y su varianza. Para hacer inferencia necesitamos más.}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Supuesto 6: Normalidad de $u$}
\begin{itemize}
    \item Las perturbaciones $u$ se distribuyen como $u \sim N(0, \sigma^2)$.
    \item Este supuesto es importante porque para realizar inferencia acerca de los $\beta$ y la construcción de intervalos de confianza se necesita la distribución de sus estimadores $\hat{\beta}$ (no basta con conocer su esperanza y varianza).
\end{itemize}
\end{frame}

\begin{frame}{Normalidad}
\begin{itemize}
    \item  Como u es la suma de muchos factores distintos no observados que afectan a y, se puede apelar
al teorema del límite central (TLC)  para concluir que u tiene una distribución
aproximadamente normal.

\item Se asume que todos los factores
no observados afectan a y por separado en forma aditiva.  Si u
es una  función no aditiva de los factores no observados, entonces el argumento del TLC no
puede emplearse.
\end{itemize}
\end{frame}
\begin{frame}{Supuestos del modelo lineal clásico (MLC)}

\begin{itemize}
\item Consisten en todos los supuestos de Gauss-Markov más el supuesto  del error
distribuido normalmente.
    \item Si se satisfacen los supuestos del MCL los estimadores de MCO son insesgados de varianza mínima, no es ya necesario restringir la
comparación a estimadores lineales en y.
\end{itemize}

{\color{blue} $$ y|X\sim N(\beta_0+\beta_1x_1+...+\beta_kx_k, \sigma^2)$$}

\end{frame}



\begin{frame}[shrink=10]
\frametitle{Distribución del estimador de MCO}
\begin{itemize}
    \item Teorema: bajo los supuestos 1 a 6 se cumple $\hat{\beta}_j|X \sim N(\beta_j, \text{Var}(\hat{\beta}_j|X))$.
    \item Esto implica que, condicional en las variables regresoras $X$,
    $$\frac{\hat{\beta}_j - \beta_j}{\sqrt{\text{Var}(\hat{\beta}_j|X)}} \sim N(0, 1).$$
    \item Pero, recuerden que no conocemos $\text{Var}(\hat{\beta}_j)$ porque no conocemos $\sigma^2$.
    \item En su lugar, vamos a usar la estimación: $\hat{\text{Var}}(\hat{\beta}_j)$.
    \item Nos queda que:
    $$\color{red}t_{\hat{\beta_j}}=\frac{\hat{\beta}_j - \beta_j}{\text{s.e.}(\hat{\beta}_j)} \sim t_{n-k-1}$$
    donde $\text{s.e.}(\hat{\beta}_j) = \sqrt{\hat{\text{Var}}(\hat{\beta}_j|X)}$.
    \item \textcolor{red}{El término en rojo va a ser clave para hacer inferencia y construir intervalos de confianza.}
\end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Demostración: ¿por qué $\hat{\beta}_j$ es normal?}

\begin{alertblock}{Proposición}
Bajo los supuestos del MLC (1--6),
\[
\hat{\beta}_j \mid X \;\sim\; N\!\big(\beta_j,\ \text{Var}(\hat{\beta}_j\mid X)\big),
\qquad j = 0,1,\ldots,k.
\]
\end{alertblock}

\begin{exampleblock}{Demostración (sketch)}
\textbf{Paso 1: $\hat{\beta}$ es lineal en $u$.} El estimador MCO en forma matricial es
\[
\hat{\beta} = (X^{\top}X)^{-1}X^{\top}\mathbf{y}
           = (X^{\top}X)^{-1}X^{\top}(X\beta + u)
           = \beta + \underbrace{(X^{\top}X)^{-1}X^{\top}}_{\equiv\, A}\, u .
\]
Es decir, condicional en $X$, cada $\hat{\beta}_j$ es una \textbf{combinación lineal} de los errores:
\[
\hat{\beta}_j = \beta_j + \sum_{i=1}^{n} a_{ji}\, u_i .
\]

\textbf{Paso 2: una combinación lineal de normales es normal.} Por el supuesto 6, $u\mid X \sim N(0,\sigma^2 I_n)$. Una transformación lineal $A u$ de un vector normal multivariado es normal multivariada. Por lo tanto $\hat{\beta}\mid X$ es normal.

\framebreak

\textbf{Paso 3: media y varianza.} Tomando esperanza y varianza condicionales en $X$:
\[
\mathbb{E}[\hat{\beta}\mid X] = \beta + A\,\mathbb{E}[u\mid X] = \beta
\quad (\text{insesgadez}),
\]
\[
\text{Var}(\hat{\beta}\mid X) = A\,\text{Var}(u\mid X)\,A^{\top}
= \sigma^2 (X^{\top}X)^{-1}.
\]
Tomando la componente $j$ de la diagonal se obtiene
\[
\hat{\beta}_j\mid X \sim N\!\big(\beta_j,\ \sigma^2 [(X^{\top}X)^{-1}]_{jj}\big).
\hfill \square
\]
\end{exampleblock}

\framebreak

\begin{exampleblock}{De la normal estándar al estadístico $t$}
Estandarizando con la varianza \textbf{verdadera}:
\[
\frac{\hat{\beta}_j - \beta_j}{\sqrt{\sigma^2 [(X^{\top}X)^{-1}]_{jj}}} \sim N(0,1).
\]
Pero $\sigma^2$ es desconocido. Lo reemplazamos por $\hat{\sigma}^2 = \tfrac{\text{SCR}}{n-k-1}$. Dos hechos clave (que se prueban con álgebra de formas cuadráticas):
\begin{enumerate}
    \item $\dfrac{(n-k-1)\hat{\sigma}^2}{\sigma^2} = \dfrac{\text{SCR}}{\sigma^2} \sim \chi^2_{\,n-k-1}$.
    \item $\hat{\beta}_j$ y $\hat{\sigma}^2$ son \textbf{independientes} (residuos $\perp$ valores ajustados bajo normalidad).
\end{enumerate}
Como $t = \dfrac{Z}{\sqrt{\chi^2_{m}/m}}$ con $Z\sim N(0,1)$ independiente del $\chi^2_m$ define una $t_m$:
\[
t_{\hat{\beta}_j} = \frac{(\hat{\beta}_j-\beta_j)/\sqrt{\sigma^2[(X^{\top}X)^{-1}]_{jj}}}{\sqrt{\hat{\sigma}^2/\sigma^2}}
= \frac{\hat{\beta}_j - \beta_j}{\text{s.e.}(\hat{\beta}_j)} \sim t_{\,n-k-1}.
\hfill \square
\]
\end{exampleblock}

\end{frame}

\section{Prueba sobre un parámetro}

\begin{frame}[shrink=10]
\frametitle{Test de hipótesis sobre un parámetro individual}
\begin{itemize}
    \item Tenemos el modelo poblacional
    $$Y_i = \beta_0 + \beta_1X_{i1} + \ldots + \beta_kX_{ik} + u_i$$
    \item Hasta aquí, aprendimos a encontrar el estimador $\hat{\beta}$ de los $\beta$.
    \item Además, aprendimos a estimar su varianza y su función de distribución.
    \item Ahora queremos usar todo esto para testear hipótesis acerca del valor de $\beta$.

\end{itemize}
\end{frame}

\begin{frame}[shrink=10]
\frametitle{Test de hipótesis sobre un parámetro individual}
\begin{itemize}
    \item Suponga que quiere testear si el coeficiente $\beta_j$ que acompaña a $x_j$ es significativamente distinto al valor $a_j$. Donde $a_j$ es una constante (un número). Es decir:
    \begin{align*}
        H_0 &: \beta_j = a_j \\
        H_1 &: \beta_j \neq a_j
    \end{align*}
    \item {\color{red} Para el caso particular de $a_j = 0$, el test busca determinar si la variable $x_j$ es significativa o relevante para el modelo.

     Si se rechaza $H_0$, entonces $x_j$ es estadísticamente significativa.}

    \item Para el test se usa el siguiente estadístico $t$ construido bajo la hipótesis nula:
    $$t_{\hat{\beta}_j} = \frac{\hat{\beta}_j - a_j}{s.e.(\hat{\beta}_j)} \sim t_{n-k-1}$$
    \item $t_{\hat{\beta}_j}$ mide en cuántas desviaciones estándar estimadas de $\hat{\beta}_j$, se aleja $\hat{\beta}_j$ de $a_j$. Valores de $t_{\hat{\beta}_j}$ alejados de $a_j$ son evidencia en contra de la hipótesis nula.
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Test de hipótesis sobre un parámetro individual}
\begin{itemize}
    \item {\color{blue}Supongamos que se fija la probabilidad de cometer error tipo I en $\alpha$. Nivel de significancia
o probabilidad de rechazar $H_0$ cuando en realidad es verdadera.}
    \item {\color{red}Sea $t_{\alpha/2}$ el percentil $(1 - \frac{\alpha}{2})100$ de la distribución t-Student de $n - k - 1$ grados de libertad.}
    \item Si $\left|\frac{\hat{\beta}_j - a_j}{s.e.(\hat{\beta}_j)}\right| \geq t_{\alpha/2}$, se rechaza $H_0$.
\end{itemize}

\begin{frame}
\frametitle{Test de hipótesis sobre un parámetro individual}
\begin{itemize}
    \item El anterior corresponde a un test de dos colas.
    \item Un nivel de significancia del $5\%$ es el valor más común.
    \item Esto significa que de todas las muestras aleatorias que se pueden tomar, en un $5\%$ de ellas se rechaza equivocadamente la hipótesis nula.
    \item Si se rechaza $H_0$ a un nivel de significancia de $\alpha$, entonces también se rechaza para niveles de significancia más altos. Por ejemplo, si una variable es significativa al $1\%$, entonces también lo es al $5\%$, al $10\%$, $20\%$, etc.
    \item \textcolor{blue}{Recuerde que $t_n \rightarrow N(0, 1)$ cuando $n \rightarrow \infty$. }
    %En la práctica, cuando los grados de libertad del estadístico t son más de 120 se suele aproximar la distribución t-Student con la Normal y usar los valores críticos de esta distribución.}
\end{itemize}
\end{frame}

\begin{frame}{Intervalos de Confianza para un Parámetro Individual}
    \begin{itemize}
        \item El intervalo de confianza al $(1-\alpha)100\%$ para $\beta_j$ es:
        \[
        \hat{\beta}_j - t_{\alpha/2} \text{s.e.}(\hat{\beta}_j) \leq \beta_j \leq \hat{\beta}_j + t_{\alpha/2} \text{s.e.}(\hat{\beta}_j)
        \]
        \item Esto nos indica que con un $(1 - \alpha)100\%$ de confianza, $\beta_j$ estará dentro de ese intervalo.
        \item Si $\beta_j$ no está en el intervalo, entonces se rechaza la hipótesis nula al $(\alpha 100\%)$ de significancia.
    \end{itemize}
\end{frame}

\section{Pruebas sobre varios parámetros: la prueba F}

\begin{frame}{¿Por qué necesitamos la prueba $F$?}
\begin{itemize}
    \item La prueba $t$ sirve para \textbf{una} restricción a la vez (un coeficiente).
    \item Muchas hipótesis económicas involucran \textbf{varios} parámetros simultáneamente:
    \begin{itemize}
        \item ¿Son las variables $x_3$, $x_4$ y $x_5$ \emph{conjuntamente} irrelevantes?
        \item ¿Tiene el modelo poder explicativo \emph{global}?
    \end{itemize}
    \pause
    \item \textcolor{red}{No basta con mirar varias pruebas $t$ por separado:} cada una controla el error tipo I individual, pero el error conjunto se acumula (problema de comparaciones múltiples).
    \item Necesitamos un único estadístico que evalúe \textbf{todas las restricciones a la vez}: la prueba $F$.
\end{itemize}
\end{frame}

\begin{frame}{Modelo restringido vs.\ no restringido}
Queremos testear $q$ restricciones lineales sobre los parámetros. Por ejemplo:
\[
H_0:\ \beta_{k-q+1} = \beta_{k-q+2} = \cdots = \beta_{k} = 0,
\qquad
H_1:\ \text{al menos uno}\neq 0 .
\]
\begin{columns}
  \column{0.5\textwidth}
  \begin{block}{Modelo no restringido (nr)}
  Incluye \textbf{todas} las variables. Tiene $k+1$ parámetros.
  \[
  y = \beta_0 + \beta_1 x_1 + \cdots + \beta_k x_k + u
  \]
  Suma de cuadrados residual: $\text{SCR}_{nr}$.
  \end{block}

  \column{0.5\textwidth}
  \begin{alertblock}{Modelo restringido (r)}
  Impone $H_0$ (quita $q$ variables). Tiene $k+1-q$ parámetros.
  \[
  y = \beta_0 + \beta_1 x_1 + \cdots + \beta_{k-q} x_{k-q} + u
  \]
  Suma de cuadrados residual: $\text{SCR}_{r}$.
  \end{alertblock}
\end{columns}
\vspace{2mm}
\textcolor{blue}{Idea clave: al imponer restricciones el ajuste sólo puede empeorar, por lo que $\text{SCR}_r \geq \text{SCR}_{nr}$. La prueba $F$ mide si ese empeoramiento es \emph{grande} relativo al error.}
\end{frame}

\begin{frame}[allowframebreaks]{El estadístico $F$ de restricciones lineales}

\begin{alertblock}{Estadístico $F$ (vía SCR)}
Bajo los supuestos del MLC y bajo $H_0$:
\[
\boxed{\,F = \frac{(\text{SCR}_r - \text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} \;\sim\; F_{\,q,\ n-k-1}\,}
\]
donde $q$ = número de restricciones y $n-k-1$ = grados de libertad del modelo no restringido.
\end{alertblock}

\begin{exampleblock}{Demostración (sketch)}
\textbf{Paso 1.} Bajo normalidad y bajo $H_0$, las dos sumas de cuadrados, escaladas por $\sigma^2$, son chi-cuadrado independientes:
\[
\frac{\text{SCR}_r - \text{SCR}_{nr}}{\sigma^2} \sim \chi^2_{q},
\qquad
\frac{\text{SCR}_{nr}}{\sigma^2} \sim \chi^2_{\,n-k-1},
\]
y ambas son independientes (la primera depende del espacio de columnas extra; la segunda, de los residuos del modelo completo).

\framebreak

\textbf{Paso 2.} Una razón de chi-cuadrado independientes, cada una dividida por sus grados de libertad, define una $F$:
\[
F = \frac{\chi^2_q / q}{\chi^2_{n-k-1}/(n-k-1)}
  = \frac{[(\text{SCR}_r - \text{SCR}_{nr})/\sigma^2]/q}{[\text{SCR}_{nr}/\sigma^2]/(n-k-1)} .
\]
El $\sigma^2$ se cancela en numerador y denominador, dejando el estadístico observable:
\[
F = \frac{(\text{SCR}_r - \text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} \sim F_{q,\,n-k-1}.
\hfill \square
\]
\end{exampleblock}

\framebreak

\begin{block}{Lectura intuitiva}
\begin{itemize}
    \item Numerador: ganancia de ajuste \emph{por restricción} al pasar del modelo restringido al completo.
    \item Denominador: varianza residual del modelo completo ($\hat{\sigma}^2$).
    \item Si las variables omitidas no aportan, $\text{SCR}_r \approx \text{SCR}_{nr}$ y $F$ es pequeño $\Rightarrow$ no se rechaza $H_0$.
    \item Si aportan mucho, $\text{SCR}_r \gg \text{SCR}_{nr}$ y $F$ es grande $\Rightarrow$ se rechaza $H_0$.
    \item \textcolor{red}{La prueba $F$ es siempre de \emph{una cola a la derecha}: sólo valores grandes rechazan.}
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Forma equivalente del $F$ con $R^2$}
\begin{alertblock}{Estadístico $F$ (vía $R^2$)}
Si \textbf{ambos} modelos tienen la \emph{misma} variable dependiente $y$, el $F$ puede escribirse con los $R^2$:
\[
\boxed{\,F = \frac{(R^2_{nr} - R^2_{r})/q}{(1 - R^2_{nr})/(n-k-1)}\,}
\]
\end{alertblock}
\begin{exampleblock}{¿De dónde sale?}
Recordando que $R^2 = 1 - \tfrac{\text{SCR}}{\text{SCT}}$, con \textbf{la misma} SCT en ambos modelos:
\[
\text{SCR}_r = (1-R^2_r)\,\text{SCT}, \qquad \text{SCR}_{nr} = (1-R^2_{nr})\,\text{SCT}.
\]
Sustituyendo en la fórmula vía SCR, la SCT se cancela:
\[
F = \frac{[(1-R^2_r) - (1-R^2_{nr})]\,\text{SCT}/q}{(1-R^2_{nr})\,\text{SCT}/(n-k-1)}
  = \frac{(R^2_{nr}-R^2_r)/q}{(1-R^2_{nr})/(n-k-1)}.
\hfill \square
\]
\end{exampleblock}
\textcolor{red}{Cuidado: la forma con $R^2$ NO es válida si $y$ aparece transformada (p.ej.\ $y$ vs.\ $\log y$): ahí use SCR.}
\end{frame}

\begin{frame}{Caso especial: significancia global del modelo}
\begin{block}{Hipótesis}
\[
H_0:\ \beta_1 = \beta_2 = \cdots = \beta_k = 0
\qquad (\text{aquí } q = k)
\]
\[
H_1:\ \text{al menos un }\beta_j \neq 0 .
\]
\end{block}
El modelo restringido sólo tiene la constante, por lo que $R^2_r = 0$. La fórmula colapsa a:
\[
F = \frac{R^2/k}{(1-R^2)/(n-k-1)} \;\sim\; F_{k,\,n-k-1}.
\]
\begin{itemize}
    \item Es el $F$ que reportan por defecto Stata, R y \texttt{statsmodels}.
    \item Responde: \textcolor{blue}{¿el conjunto de regresores explica algo de $y$, más allá del azar?}
\end{itemize}
\end{frame}

\begin{frame}{Región de rechazo de la prueba $F$}
\begin{figure}
\centering
\begin{tikzpicture}[scale=1.1]
  \draw[->] (0,0) -- (5,0) node[right] {$F$};
  \draw[->] (0,0) -- (0,2.6) node[above] {densidad $F_{q,n-k-1}$};
  % Curva tipo F (asimétrica a la derecha), trazada con coordenadas fijas
  \draw[thick, blue, smooth]
    (0,0) .. controls (0.3,1.9) and (0.7,2.0) .. (1.0,1.7)

          .. controls (3.8,0.13) and (4.4,0.06) .. (4.8,0.04);
  % Zona de rechazo (cola derecha) aproximada con polígono bajo la curva
  \fill[red!30]
    (2.5,0) -- (2.5,0.42) .. controls (3.0,0.30) and (3.8,0.13) .. (4.8,0.04)

  \draw[dashed] (2.5,0) -- (2.5,1.0);
  \node[below] at (2.5,0) {\small $F_{\alpha}$};

  \node[blue] at (1.1,1.3) {\small no rechazo};
\end{tikzpicture}
\end{figure}
\textbf{Regla de decisión:} se rechaza $H_0$ si $F \geq F_{\alpha;\,q,\,n-k-1}$, el percentil $(1-\alpha)$ de la $F$.
\end{frame}

\begin{frame}{Relación entre la prueba $t$ y la prueba $F$}
\begin{alertblock}{Resultado}
Para una \textbf{única} restricción ($q=1$), las pruebas $t$ (de dos colas) y $F$ son \emph{equivalentes}:
\[
\boxed{\,F = t^2\,}
\qquad\text{y}\qquad
F_{\alpha;\,1,\,m} = \left(t_{\alpha/2;\,m}\right)^2 .
\]
\end{alertblock}
\begin{exampleblock}{¿Por qué?}
\begin{itemize}
    \item Por definición, si $T \sim t_m$, entonces $T^2 \sim F_{1,m}$ (una normal al cuadrado dividida por $\chi^2_m/m$).
    \item Numéricamente: para $m$ grande, $t_{0.025} = 1.96$ y $(1.96)^2 = 3.84 = F_{0.05;\,1,\infty}$.
\end{itemize}
\end{exampleblock}
\textcolor{blue}{Moraleja: para testear un solo coeficiente da igual usar $t$ o $F$. La prueba $F$ aporta valor cuando hay $q\geq 2$ restricciones.}
\end{frame}

\section{Ejercicios resueltos}

\begin{frame}{Ejercicio Resuelto 1: significancia conjunta vía $R^2$}

\begin{block}{Ejercicio 1 (Certamen T2, salarios RM)}
Con $n = 206$ trabajadores se estima
\[
\log(\text{salario}) = \beta_0 + \beta_1 educ + \beta_2 exp + \beta_3 tenure + \beta_4 female + \beta_5 union + u,
\]
con $R^2_{nr} = 0.78$. El investigador sospecha que basta con $educ$ y $exp$. Al estimar el modelo restringido (sólo $educ$, $exp$) obtiene $R^2_r = 0.68$. Con $\alpha = 5\%$ y $F_{0.95;\,3,\,200} = 2.65$, ¿son relevantes $tenure$, $female$ y $union$?
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Paso 1 — Hipótesis:} $H_0: \beta_3=\beta_4=\beta_5=0$ vs.\ $H_1:$ al menos uno $\neq 0$. Aquí $q=3$.
\\
\textbf{Paso 2 — Estadístico (misma $y$, uso $R^2$):}
\[
F = \frac{(0.78-0.68)/3}{(1-0.78)/(206-5-1)} = \frac{0.10/3}{0.22/200} = \frac{0.0333}{0.0011} \approx 30.3.
\]
\textbf{Paso 3 — Decisión:} como $F = 30.3 > 2.65 = F_{0.95;3,200}$, \textbf{se rechaza} $H_0$.
\\
\textbf{Paso 4 — Conclusión:} las tres variables son conjuntamente significativas; la propuesta de descartarlas NO se sostiene.
\end{exampleblock}

\end{frame}

\begin{frame}{Ejercicio Resuelto 2: $F$ con dos modelos anidados}

\begin{block}{Ejercicio 2 (Guía Certamen 2)}
Modelo completo con $n=75$ y cuatro regresores ($x_1,\ldots,x_4$): $R^2_{nr}=0.62$. El investigador sospecha que $x_3$ y $x_4$ no aportan, y estima el modelo con sólo $x_1,x_2$: $R^2_r=0.55$. Con $\alpha=5\%$ y $F_{0.95;\,2,\,70}=3.13$, decida.
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Hipótesis:} $H_0:\beta_3=\beta_4=0$ vs.\ $H_1:$ al menos uno $\neq 0$, con $q=2$.
\[
F = \frac{(R^2_{nr}-R^2_r)/q}{(1-R^2_{nr})/(n-k-1)}
  = \frac{(0.62-0.55)/2}{(1-0.62)/(75-4-1)}
  = \frac{0.035}{0.38/70} = \frac{0.035}{0.00543} \approx 6.45.
\]
Como $6.45 > 3.13$, \textbf{se rechaza} $H_0$ al 5\%: $x_3$ y $x_4$ \textbf{sí} son relevantes para el modelo.
\end{exampleblock}

\end{frame}

\begin{frame}[allowframebreaks]{Ejercicio Resuelto 3: restricción lineal vía $t$}

\begin{block}{Ejercicio 3 (Guía Certamen 2, restricción $\beta_1=4\beta_3$)}
Modelo $y = \beta_0 + \beta_1 x_1 + \beta_2 x_2 + \beta_3 x_3 + u$ estimado con $n=250$:
\[
\hat{\beta} = (3,\,9,\,5,\,2)^{\top},\quad
(X'X)^{-1}=\begin{bmatrix} 1&0&0&0\\ 0&1&0&-1\\ 0&0&2&-1\\ 0&-1&-1&1\end{bmatrix},\quad \hat{\sigma}^2 = 2.
\]
Teste $H_0:\beta_1 = 4\beta_3$ al 5\%.
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Paso 1 — Hipótesis (una sola restricción $\Rightarrow$ uso $t$):}
\[
H_0: \beta_1 - 4\beta_3 = 0 \qquad H_1: \beta_1 - 4\beta_3 \neq 0.
\]
\textbf{Paso 2 — Varianza de la combinación.} Con $\hat{V}(\hat{\beta}) = \hat{\sigma}^2 (X'X)^{-1}$, los elementos relevantes son
$\hat{V}(\hat{\beta}_1)=2\cdot 1 = 2$, $\hat{V}(\hat{\beta}_3)=2\cdot 1 = 2$, $\widehat{\text{Cov}}(\hat{\beta}_1,\hat{\beta}_3)=2\cdot(-1)=-2$. Entonces
\[
\hat{V}(\hat{\beta}_1 - 4\hat{\beta}_3) = \hat{V}(\hat{\beta}_1) + 16\,\hat{V}(\hat{\beta}_3) - 8\,\widehat{\text{Cov}}(\hat{\beta}_1,\hat{\beta}_3)
= 2 + 32 - 8(-2) = 50.
\]

\framebreak

\textbf{Paso 3 — Estadístico $t$:}
\[
t = \frac{\hat{\beta}_1 - 4\hat{\beta}_3 - 0}{\sqrt{\hat{V}(\hat{\beta}_1-4\hat{\beta}_3)}}
  = \frac{9 - 4\cdot 2}{\sqrt{50}} = \frac{1}{7.07} \approx 0.14.
\]
\textbf{Paso 4 — Decisión:} con $t_{0.975;\,246}\approx 1.96$ y $|0.14| < 1.96$, \textbf{no se rechaza} $H_0$. Los datos son compatibles con $\beta_1 = 4\beta_3$.

\medskip
\textcolor{blue}{Nota: una restricción lineal cualquiera ($H_0:\,a^{\top}\beta = c$) puede testearse con $t$ usando $\text{Var}(a^{\top}\hat{\beta}) = a^{\top}\hat{V}(\hat{\beta})\,a$, o equivalentemente con un $F_{1,n-k-1}$ (recordando $F=t^2$).}
\end{exampleblock}

\end{frame}

\begin{frame}{Ejercicio Resuelto 4: significancia global y $F=t^2$}

\begin{block}{Ejercicio 4 (gastos de campaña, modelo 5)}
$voteA = \beta_0 + \beta_1 lexpendA + \beta_2 lexpendB + \beta_3 prtystrA + u$, con $n=173$, $R^2 = 0.793$. Realice el test de significancia global.
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Hipótesis:} $H_0: \beta_1=\beta_2=\beta_3=0$ vs.\ $H_1:$ al menos uno $\neq 0$, con $k=q=3$.
\[
F = \frac{R^2/k}{(1-R^2)/(n-k-1)} = \frac{0.793/3}{(1-0.793)/169} = \frac{0.2643}{0.001225} \approx 215.8.
\]
Como $F=215.8$ supera con creces cualquier valor crítico $F_{0.95;3,169}\approx 2.66$, \textbf{se rechaza} $H_0$ a cualquier nivel: el modelo es globalmente significativo.

\medskip
\textbf{Verificación $F=t^2$:} si se testea sólo $\beta_3=0$ con $\hat{\beta}_3 = 0.152$, $s.e.=0.062$, entonces $t = 2.45$ y $t^2 = 6.00 = F_{1,169}$ correspondiente. Coinciden.
\end{exampleblock}

\end{frame}

\begin{frame}[fragile]{Ejemplo aplicado: prueba $F$ en Python}

\begin{exampleblock}{Significancia conjunta con \texttt{statsmodels}}
\begin{verbatim}
import statsmodels.formula.api as smf
import pandas as pd

datos = pd.read_csv("salarios_rm.csv")

# Modelo no restringido (completo)
m_nr = smf.ols("np.log(salario) ~ educ + exp + tenure"
               " + female + union", data=datos).fit()

# F de significancia global (lo da el summary)
print(m_nr.summary())   # F-statistic y Prob (F-statistic)

# F conjunto: H0: tenure = female = union = 0
print(m_nr.f_test("tenure = 0, female = 0, union = 0"))
\end{verbatim}
\end{exampleblock}
\textcolor{blue}{\texttt{f\_test} devuelve el estadístico $F$, su $p$-valor y los grados de libertad $(q,\ n-k-1)$.}

\end{frame}

\section{Cierre}

\begin{frame}
\frametitle{Conclusiones}
\begin{itemize}
    \item Hemos visto cómo se puede hacer tests de hipótesis simples sobre estimadores del MCO.
    \item El test más importante es determinar si el parámetro es significativamente diferente de cero.
    \item La prueba $F$ extiende la inferencia a \emph{varias} restricciones simultáneas.
\end{itemize}
\end{frame}

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 10}
\begin{enumerate}
    \item Bajo los supuestos del MLC (1--6), $\hat{\beta}_j\mid X \sim N(\beta_j, \text{Var}(\hat{\beta}_j\mid X))$; al estimar $\sigma^2$ surge $t_{\hat{\beta}_j} \sim t_{n-k-1}$.
    \item La prueba $t$ y el intervalo de confianza permiten inferir sobre \emph{un} parámetro (o una restricción lineal).
    \item La prueba $F = \dfrac{(\text{SCR}_r-\text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} \sim F_{q,n-k-1}$ evalúa $q$ restricciones a la vez; su forma con $R^2$ requiere la misma $y$.
    \item Significancia global: $H_0:\beta_1=\cdots=\beta_k=0$, con $F = \dfrac{R^2/k}{(1-R^2)/(n-k-1)}$.
    \item Para una sola restricción, $F = t^2$: ambas pruebas coinciden.
\end{enumerate}
\end{block}
