\begin{frame}{Introducción}
    \begin{itemize}
        \item La clase pasada, vimos cómo hacer un test de hipótesis sobre un estimador específico del MCO.
        \item Pero a veces, no es eso lo que buscamos sino respuestas a hipótesis más complejas.
            \begin{itemize}
                \item ¿Es el coeficiente sobre la variable A igual al coeficiente para la variable B?
                \item ¿Puedo rechazar la hipótesis de que tres factores económicos al mismo tiempo no influyen en mi variable de interés?
                \item ¿Puedo excluir la posibilidad de que los sueldos dependen de variables discriminatorias (género, edad, origen)?
            \end{itemize}
      
    \end{itemize}
\end{frame}

\begin{frame}{Hipótesis de igualdad de parámetros}
    \begin{itemize}
        \item Imaginamos que queremos testear: $H_0: \beta_1 = \beta_2$. 
        \item Planteamos el test:
        
        $H_0: \beta_1 - \beta_2 = 0$.

        $H_1: \beta_1 - \beta_2 \neq 0$
        \item Dado que los $\hat{\beta}$ son distribuidos como normales, la diferencia también es una normal.
        \item Se puede, entonces, construir un test $t$ con el $t$-estadístico:
        \[t_{\text{estadístico}} = \frac{\hat{\beta}_1 - \hat{\beta}_2 - 0}{\sqrt{\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2)}}\]
        \item Necesitamos: $\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2)$.
        \item Recuerden propiedades de la varianza. Sea $k,c$ constantes. Sean $A, B$ variables aleatorias:
        \begin{enumerate}
            \item $\text{Var}(kA) = k^2A$
            \item $\text{Var}(kA + cB) = k^2\text{Var}(A) + c^2\text{Var}(B) + 2kc\text{COV}(A, B)$
        \end{enumerate}
    \end{itemize}
\end{frame}

\begin{frame}[shrink=10]{Continuación}
    \begin{itemize}
        \item Entonces, en este caso, usando propiedades de la varianza ($k = 1$, $c = -1$):
        \[\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2) = \textcolor{red}{\hat{Var}(\hat{\beta}_1) + \hat{Var}(\hat{\beta}_2)} - \textcolor{blue}{2\hat{\text{COV}}(\hat{\beta}_1, \hat{\beta}_2)}\]
        \item Recuerden la matriz: $\hat{Var}(\hat{\beta})= (X^{\top}X)^{-1}\hat{\sigma}^2$.
        \begin{enumerate}
            \item \textcolor{red}{$\hat{Var}(\hat{\beta}_j) = \hat{\sigma}^2_{j}$ es el elemento $(j + 1), (j + 1)$ de esa matriz (diagonal principal).}
            \item \textcolor{blue}{$\hat{\text{COV}}(\hat{\beta}_j, \hat{\beta}_l) = \hat{\sigma}_{jl}$ es el elemento $(j + 1), (l + 1)$ de esa matriz.}
        \end{enumerate}
        \item Regla de rechazo: una vez construido el $t$-estadístico, se rechaza la hipótesis nula si $t_{\text{estadístico}} > t_{\alpha/2,n-k-1}$, donde $\alpha$ es el nivel de significatividad requerido.
    \end{itemize}
\end{frame}

\begin{frame}{Ejemplo: Test $\beta_1 = \beta_2$ con $\alpha = 0.05$}

\vspace{0.5cm}



$\mathbf{(X^{\top}X)^{-1}} =
        \begin{bmatrix}
            0.05987 & -0.00177 & -0.00035 \\
            -0.00177 & 0.00030 & -0.00002 \\
            -0.00035 & -0.00002 & 0.00001 \\
        \end{bmatrix}$; $\hat{\sigma}^2 = \frac{SSR}{n-k-1} = 141925$; 

        $\hat{\beta} =
        \begin{bmatrix}
            140 \\
            42 \\
            5 \\
        \end{bmatrix}$; $n=140$

Entonces: $\mathbf{(X^{\top}X)^{-1}\hat{\sigma}^2} =
        \begin{bmatrix}
            8497.04975 & -251.20725 & -49.67375 \\
            -251.20725 & \textcolor{red}{42.5775} & \textcolor{blue}{-2.8385} \\
            -49.67375 & \textcolor{blue}{-2.8385} & \textcolor{red}{1.41925} \\
        \end{bmatrix}$; $\hat{\text{COV}}(\hat{\beta}_1, \hat{\beta}_2) = \textcolor{blue}{-2.8385}$\\
        $\hat{Var}(\hat{\beta}_1)=\textcolor{red}{42.5775} \quad \hat{Var}(\hat{\beta}_2)=\textcolor{red}{1,41925} $
\end{frame}
\begin{frame}{Ejemplo: Test $\beta_1 = \beta_2$ con $\alpha = 0,05$}
Procedimiento:
        \begin{itemize}
            \item Paso 1: Hipótesis: $H_0: \beta_1 - \beta_2 = 0$\\
            $H_1: \beta_1 - \beta_2 \neq 0$.
            \item Paso 2: Construir: $\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2)$:
            \begin{itemize}
          
                \item $\sqrt{\hat{Var}(\hat{\beta}_1 - \hat{\beta}_2)} = \sqrt{42,58 + 1.42 - 2(-2.84)} = 7,05$
            \end{itemize}
            \item Paso 3: $t_e = \frac{42 - 5}{7,05} = 5.3$
            \item Paso 4: Rechazo $H_0$ porque $5.3 > 1.98$  valor crítico de  la t con 137 grados de libertad  con un 5\% de significancia. 
        \end{itemize}
\end{frame}

\begin{frame}{Ejercicio}
Testee la $H_0$ de que el efecto de $x_1$ sobre $y$ es el doble que el efecto de $x_2$ sobre $y:\beta_1=2\beta_2$ frente a la alternativa que es diferente. Suponga $\alpha=5\%$.
    \begin{itemize}
      
          \item $H_0: \beta_1 - 2\beta_2 = 0$
          \item $H_1: \beta_1 - 2\beta_2 \neq 0$.
  \end{itemize}
\end{frame}

\begin{frame}{Ejercicio}
  
   \begin{itemize}
      \item Para construir el t-estadístico empírico: $t_e = \frac{\hat{\beta}_1 - 2\hat{\beta}_2}{\sqrt{\hat{\text{Var}}(\hat{\beta}_1 - 2\hat{\beta}_2)}}$ necesitamos calcular:  
        \begin{itemize}
            \item $\hat{\text{Var}}(\hat{\beta}_1 - 2\hat{\beta}_2) = \hat{\text{Var}}(\hat{\beta}_1) + 4\hat{\text{Var}}(\hat{\beta}_2) - 4\hat{\text{COV}}(\hat{\beta}_1, \hat{\beta}_2)$
            \item Usando la matriz de antes reemplazamos para obtener:
            \begin{align*}
               \hat{ \text{Var}}(\hat{\beta}_1 - 2\hat{\beta}_2) &= 42.5 + 4 \times 1.42 - 4(-2.84) \\
                &= 59.5 \\
                \Rightarrow \text{s.e}(\hat{\beta}_1 - 2\hat{\beta}_2) &= \sqrt{\hat{\text{Var}}(\hat{\beta}_1 - 2\hat{\beta}_2)} = 7.68
            \end{align*}
        \end{itemize}
        \item Construimos el t-empírico: $t_e = \frac{42 - 10}{7.68} = 4.16$
        \item Regla: Rechazo si $|t_e| > 1.98$ valor crítico de  la t con 137 grados de libertad . Entonces, se rechaza $H_0$ con un 5\% de significancia, rechazo la hipótesis de que $\beta_1 = 2\beta_2$.  
   \end{itemize}
\end{frame}
