\begin{frame}{Objetivos de la Clase}

Al finalizar esta clase, el estudiante será capaz de:

\begin{itemize}
    \item[\checkmark] Definir la \textbf{autocorrelación serial} ($\text{Cov}(u_t,u_s)\neq0$) y ubicarla como violación del supuesto de muestreo aleatorio de Gauss--Markov.
    \item[\checkmark] Demostrar las \textbf{consecuencias} para MCO: el estimador sigue \emph{insesgado y consistente}, pero los errores estándar son \emph{inválidos}.
    \item[\checkmark] Aplicar los tests de detección: \textbf{Breusch--Godfrey} (regresión auxiliar de $\hat u_t$ sobre rezagos, $nR^2\sim\chi^2$) y \textbf{Durbin--Watson}.
    \item[\checkmark] Corregir el problema vía \textbf{MCG / Cochrane--Orcutt} o mediante errores estándar \textbf{HAC de Newey--West}.
\end{itemize}

\end{frame}

\begin{itemize}

  \begin{itemize}
    \item \( \triangle y_t = \alpha + (\rho - 1) y_{t-1} + \lambda t + \epsilon_t \)
 \textcolor{red}{   \item \( \triangle y_t = \alpha + \phi y_{t-1} + \lambda t + \epsilon_t \)}

\begin{frame}
\frametitle{El modelo lineal clásico-supuestos: Dependencia de los errores}

\begin{itemize}
  \item El teorema de Gauss-Markov especifica que MCO es MELI cuando:
  \begin{itemize}
    \item Linealidad de los parámetros
    \item Muestreo aleatorio
    \item Colinealidad imperfecta
    \item Media condicional cero
    \item Homoscedasticidad
  \end{itemize}
  \item Pero, ¿qué pasa si se viola el segundo supuesto? ¿Qué pasa si las observaciones no son iid? ¿Si no son independientes? Violación de muestreo aleatorio.
  \item Esta clase vamos a ver qué implicaciones tiene esto en nuestras estimaciones por MCO, cómo se puede detectar y cómo podemos "solucionarlo".
  \item Nos vamos a enfocar en series de tiempo y datos de panel.
\end{itemize}

\end{frame}
\begin{frame}[shrink=10]
\frametitle{Observaciones no independientes - errores correlacionados}

\begin{itemize}
  \item Efecto de no tener observaciones iid: errores correlacionados
  \item Errores correlacionados: el término de error de una observación está correlacionado con el término de error de otra
  \begin{itemize}
    \item \( \text{COV}(u_i, u_j) \neq 0 \quad \forall i \neq j \)
  \end{itemize}
  \item Dependiendo de la estructura de los datos esta dependencia tendrá distinto nombre e interpretación:
  \begin{itemize}
    \item Series de tiempo: hablamos de autocorrelación serial
    \begin{itemize}
      \item El error que cometemos al explicar la relación entre desempleo e inflación en 2017 estará relacionado con el error que cometemos explicando esta relación en 2018. \( \text{Cov}(u_t, u_{t-1}) \neq 0 \)
    \end{itemize}
    \item Sección cruzada: hablamos de correlación espacial
    \begin{itemize}
      \item El error que cometemos explicando el resultado de un niño en una clase está relacionado con el error que cometemos explicando el resultado de otro niño de esa misma clase (efecto del maestro)
    \end{itemize}
    \item Panel: Pueden ocurrir ambas, típicamente pensamos que los errores de la misma observación en el tiempo están correlacionados: \( \text{Cov}(u_{it}, u_{it-1}) \neq 0 \)
    \end{itemize}
  \end{itemize}

\end{frame}
\begin{frame}[shrink=20]
\frametitle{Dependencia de los errores y autocorrelación serial}

\begin{itemize}
  \item Bajo los supuestos clásicos MCO:
  \[
  \Sigma = E(UU') = 
  \begin{pmatrix}
  \sigma^2 & 0 & 0 & \cdots & 0 \\
  0 & \sigma^2 & 0 & \cdots & 0 \\
  0 & 0 & \sigma^2 & \cdots & 0 \\
  \vdots & \vdots & \vdots & \ddots & \vdots \\
  0 & 0 & 0 & \cdots & \sigma^2
  \end{pmatrix}
  \]

  \item Con heteroscedasticidad y sin dependencia:
  \[
  \Sigma = E(UU') = 
  \begin{pmatrix}
  \sigma^2_1 & 0 & 0 & \cdots & 0 \\
  0 & \sigma^2_2 & 0 & \cdots & 0 \\
  0 & 0 & \sigma^2_3 & \cdots & 0 \\
  \vdots & \vdots & \vdots & \ddots & \vdots \\
  0 & 0 & 0 & \cdots & \sigma^2_T
  \end{pmatrix}
  \]

  \item Con homoscedasticidad y dependencia de los errores:
  \[
  \Sigma = E(UU') = 
  \begin{pmatrix}
  \sigma^2 & \sigma^2 \rho_{1,2} & \sigma^2 \rho_{1,3} & \cdots & \sigma^2 \rho_{1,T} \\
  \sigma^2 \rho_{2,1} & \sigma^2 & \sigma^2 \rho_{2,3} & \cdots & \sigma^2 \rho_{2,T} \\
  \sigma^2 \rho_{3,1} & \sigma^2 \rho_{3,2} & \sigma^2 & \cdots & \sigma^2 \rho_{3,T} \\
  \vdots & \vdots & \vdots & \ddots & \vdots \\
  \sigma^2 \rho_{T,1} & \sigma^2 \rho_{T,2} & \sigma^2 \rho_{T,3} & \cdots & \sigma^2
  \end{pmatrix}
  \]
\end{itemize}

\end{frame}
\begin{frame}
\frametitle{¿Qué problema nos trae?}

\begin{itemize}
  \item La correlación de los errores nos trae un problema a la hora de estimar la varianza.
  \item La estimación puntual por MCO NO se ve afectada: el estimador sigue siendo insesgado.
  \item Repasamos: ¿Cuál es el problema de estimar la varianza con sesgo?
  \begin{itemize}
    \item Errores estándar no válidos \(\rightarrow\) Pruebas t no válidas.
    \item ¿Cómo sabemos entonces con certeza si nuestra estimación es estadísticamente cero o no?
  \end{itemize}
  \item Vamos a tener un problema de \textbf{EFICIENCIA} (precisión) pero \textbf{NO DE SESGO }de los estimadores.
\end{itemize}

\end{frame}

\begin{frame}[allowframebreaks]{Demostración: consecuencias de la autocorrelación para MCO}

\begin{alertblock}{Proposición}
En el modelo $y=X\beta+u$ con $\mathbb{E}[u\mid X]=0$ pero $\mathbb{E}[uu'\mid X]=\Sigma\neq\sigma^2 I$ (errores autocorrelacionados):
\begin{enumerate}
  \item $\hat\beta_{MCO}$ sigue siendo \textbf{insesgado} (y consistente).
  \item La fórmula usual $\widehat{\text{Var}}(\hat\beta)=\hat\sigma^2(X'X)^{-1}$ es \textbf{incorrecta}: los errores estándar, los $t$ y los $F$ \textbf{no son válidos}.
\end{enumerate}
\end{alertblock}

\begin{exampleblock}{Demostración}
\textbf{Paso 1 (insesgadez).} El estimador MCO es
\[
\hat\beta = (X'X)^{-1}X'y = (X'X)^{-1}X'(X\beta+u)=\beta+(X'X)^{-1}X'u.
\]
Tomando esperanza condicional, usando $\mathbb{E}[u\mid X]=0$:
\[
\mathbb{E}[\hat\beta\mid X]=\beta+(X'X)^{-1}X'\,\mathbb{E}[u\mid X]=\beta.
\]
La insesgadez \textbf{solo} usa $\mathbb{E}[u\mid X]=0$, \emph{no} la forma de $\Sigma$. Por eso la autocorrelación \textbf{no sesga} $\hat\beta$.

\framebreak

\textbf{Paso 2 (varianza verdadera).} Partiendo de $\hat\beta-\beta=(X'X)^{-1}X'u$:
\[
\text{Var}(\hat\beta\mid X)=(X'X)^{-1}X'\,\mathbb{E}[uu'\mid X]\,X(X'X)^{-1}=(X'X)^{-1}X'\,\Sigma\,X(X'X)^{-1}.
\]
\textbf{Paso 3 (el error de MCO).} La rutina estándar reporta $\hat\sigma^2(X'X)^{-1}$, que es la fórmula anterior \emph{solo si} $\Sigma=\sigma^2 I$. Con autocorrelación $\Sigma$ tiene términos fuera de la diagonal ($\sigma^2\rho_{ij}$), así que
\[
(X'X)^{-1}X'\Sigma X(X'X)^{-1}\ \neq\ \sigma^2(X'X)^{-1}.
\]
Tipicamente, con autocorrelación positiva, el error estándar reportado \textbf{subestima} la verdadera varianza $\Rightarrow$ los $t$ se inflan $\Rightarrow$ rechazamos $H_0$ demasiado seguido (falsos positivos). \hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}

Nuestro modelo lo podemos escribir:

\begin{itemize}
  \item \( y_t = \beta_0 + \beta_1 x_t + \epsilon_t \)
  \item Donde ahora:
  \[
  \epsilon_t = \rho \epsilon_{t-1} + v_t \quad \text{con } v_t \sim \mathcal{F}(0, \sigma^2) \text{ y } |\rho| < 1 \quad
  \]
  \item Entonces \( E(\epsilon_t \epsilon_{t+j}) = \rho \sigma^2 \)
  \item ¿Qué pasa si \(\rho\) es cero?
  \begin{itemize}
    \item \( u_t = \epsilon_t \) \(\rightarrow\) ¡Volvemos a los supuestos de G-M!
  \end{itemize}
  \item El problema de autocorrelación existe cuando \(\rho \neq 0\)!
  \item Cuando el término de error tiene esta estructura decimos que es un AR(1) ya que correlaciona únicamente con 1 período atrás.
\end{itemize}

\end{frame}

\begin{frame}[allowframebreaks]{Estructura de un error AR(1): la función de autocorrelación}

\begin{block}{Resultado}
Si $\epsilon_t=\rho\,\epsilon_{t-1}+v_t$ con $|\rho|<1$ y $v_t$ ruido blanco $(0,\sigma_v^2)$, entonces
\[
\text{Var}(\epsilon_t)=\frac{\sigma_v^2}{1-\rho^2},\qquad
\rho(k)=\text{Corr}(\epsilon_t,\epsilon_{t-k})=\rho^{\,k}.
\]
La autocorrelación \textbf{decae geométricamente} con el rezago $k$.
\end{block}

\pause

\begin{exampleblock}{Demostración}
\textbf{Paso 1 (varianza).} Aplicando varianza a $\epsilon_t=\rho\epsilon_{t-1}+v_t$ y usando que $v_t\perp\epsilon_{t-1}$:
\[
\gamma(0)=\text{Var}(\epsilon_t)=\rho^2\,\gamma(0)+\sigma_v^2 \ \Rightarrow\ \gamma(0)(1-\rho^2)=\sigma_v^2 \ \Rightarrow\ \gamma(0)=\frac{\sigma_v^2}{1-\rho^2}.
\]
\framebreak
\textbf{Paso 2 (autocovarianza de orden 1).}
\[
\gamma(1)=\text{Cov}(\epsilon_t,\epsilon_{t-1})=\text{Cov}(\rho\epsilon_{t-1}+v_t,\ \epsilon_{t-1})=\rho\,\gamma(0).
\]
\textbf{Paso 3 (recursión general).} Para $k\ge1$:
\[
\gamma(k)=\text{Cov}(\rho\epsilon_{t-1}+v_t,\ \epsilon_{t-k})=\rho\,\gamma(k-1)\ \Rightarrow\ \gamma(k)=\rho^{\,k}\gamma(0).
\]
\textbf{Paso 4 (autocorrelación).}
\[
\rho(k)=\frac{\gamma(k)}{\gamma(0)}=\rho^{\,k}.
\]
Por eso un correlograma que decae geométricamente sugiere errores AR(1). \hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}
\frametitle{Test de Breusch-Godfrey para correlación serial de orden (1)}

\begin{itemize}
  \item Para series de tiempo existen varios test.
  \item Con la idea de una autocorrelación de orden 1:
  \[
  y_t = x_t \beta + \epsilon_t
  \]
  \[
  \epsilon_t = \rho_1 \epsilon_{t-1}
  \]
  \item Test LM para testear \( H_0: \rho = 0 \) (si hay un solo rezago, también se puede hacer test t).
  \item Bajo \( H_0 \) cierta (no hay autocorrelación de orden 1) tenemos:
  \begin{itemize}
    \item Si \( \text{LM}_{\text{empírico}} > \chi^2_{\text{tabla}} \) \(\rightarrow\) ¡Rechazo \( H_0 \)!
  \end{itemize}
\end{itemize}

\end{frame}
\begin{frame}
\frametitle{Pasos Breusch-Godfrey}

\begin{itemize}
  \item La prueba de hipótesis la escribimos:
  \[
  H_0 : \rho_1 = 0
  \]
  \[
  H_1 : \rho_1 \neq 0
  \]
  \item Pasos a seguir:
  \begin{enumerate}
    \item Estimar el modelo original por MCO y obtener los residuos \(\hat{\epsilon}\).
    \item Estimar una regresión auxiliar de \(\hat{\epsilon}_t\) sobre el rezago: \(\hat{\epsilon}_t = \rho_0 + \rho_1 \hat{\epsilon}_{t-1}\).
    \item Quedarse con el \(R^2\) y construir el estadístico LM de prueba:
    \[
    \text{LM} = nR^2_\epsilon
    \]
    \item Se compara con una \(\chi^2_p\) donde \(p\) es el número de rezagos (1 en este caso).
    \item Si \(\text{LM} > \chi^2_p\) al \(\alpha \%\) \(\rightarrow\) rechazo la hipótesis nula de ausencia de autocorrelación. ¡Tenemos problema!
  \end{enumerate}
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Breusch--Godfrey de orden $p$ y por qué $nR^2\sim\chi^2_p$}

\begin{itemize}
  \item Para detectar autocorrelación de orden $p$, la regresión auxiliar incluye \emph{también} los regresores originales y $p$ rezagos del residuo:
  \[
  \hat u_t = \gamma_0 + x_t'\delta + \rho_1\hat u_{t-1} + \cdots + \rho_p\hat u_{t-p} + \nu_t.
  \]
  \item $H_0:\ \rho_1=\cdots=\rho_p=0$ (no hay autocorrelación).
  \item \textbf{Por qué la $\chi^2$:} es un test del Multiplicador de Lagrange. Bajo $H_0$, $nR^2$ de la auxiliar converge a una $\chi^2_p$ (un grado de libertad por cada restricción $\rho_j=0$); es el mismo argumento LM que vimos en heterocedasticidad (Breusch--Pagan, White).
  \item Ventaja sobre Durbin--Watson: admite \textbf{varios rezagos} y la presencia de la dependiente rezagada entre los regresores.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Test de Durbin--Watson (autocorrelación de orden 1)}

\begin{block}{Estadístico}
\[
d = \frac{\sum_{t=2}^{T}(\hat u_t - \hat u_{t-1})^2}{\sum_{t=1}^{T}\hat u_t^2}\ \approx\ 2(1-\hat\rho),
\qquad \hat\rho=\frac{\sum \hat u_t\hat u_{t-1}}{\sum \hat u_t^2}.
\]
\end{block}

\begin{itemize}
  \item Rango $d\in[0,4]$. Interpretación rápida:
  \begin{itemize}
    \item $d\approx 2$: \emph{no} hay autocorrelación ($\hat\rho\approx0$).
    \item $d\to 0$: autocorrelación \textbf{positiva} ($\hat\rho\to1$).
    \item $d\to 4$: autocorrelación \textbf{negativa} ($\hat\rho\to-1$).
  \end{itemize}
  \item \textbf{Limitaciones:} solo sirve para AR(1), tiene una ``zona de indecisión'' entre los valores de tabla $d_L$ y $d_U$, y \textbf{no es válido} si el modelo incluye la variable dependiente rezagada. En esos casos, preferir Breusch--Godfrey.
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Soluciones al problema de Autocorrelación de los errores - Similar a heterocedasticidad}

\begin{itemize}
  \item Dos grandes estrategias:
  \begin{enumerate}
    \item \textbf{MCG / Cochrane--Orcutt}: modelar la forma de la autocorrelación y transformar el modelo para recuperar eficiencia (errores ``buenos'').
    \item \textbf{MCO $+$ errores HAC (Newey--West)}: dejar el coeficiente igual y solo \emph{corregir} la varianza estimada.
  \end{enumerate}
  \item Usar MCO y corregir la varianza de los estimadores:
  \begin{itemize}
    \item Para series de tiempo:\textbf{ Errores Estándares Robustos}
 %    \item Para datos de panel: no especificamos la forma de la correlación pero le ponemos un límite, solo puede existir dentro de un “cluster” o grupo
 %    \begin{itemize}
 % \textbf{     \item Clustered Standard errors}
 %    \end{itemize}
  \end{itemize}
\end{itemize}

\end{frame}
\begin{frame}
\frametitle{Estimación por MCO con errores robustos a la autocorrelación}

\begin{itemize}
  \item Al igual que con la heteroscedasticidad, podemos estimar por MCO y obtener errores robustos a la autocorrelación.
  \item \textbf{Newey} y West proponen una forma de estimar los errores robusta a la autocorrelación y la heteroscedasticidad.
  \item Llamamos a estos errores \textbf{HAC} por sus siglas en inglés.
\end{itemize}

\end{frame}
\begin{frame}[shrink=10]
\frametitle{Series de tiempo: Estimador de los errores robustos a la autocorrelación y heteroscedasticidad. Newey y West}

\begin{itemize}
  \item El estimador propuesto para la varianza de los estimadores es el siguiente:
  \[
  s^2_{\hat{\beta}_1} = \left[ \frac{1}{T} \frac{\sigma^2_{\hat{u}}}{\sigma^2_x} \right] \hat{f}_t
  \]
  \item Newey y West proponen estimar \(\hat{f}_t\) con:
  \[
  \hat{f}_t = 1 + 2 \sum_{j=1}^J \left( \frac{J + 1 - j}{J + 1} \right) \hat{\rho}_j
  \]
  \item Donde \(J\) es el número de rezagos que queremos considerar:
  \begin{itemize}
    \item Si creemos que es un AR(1) entonces \(J = 1\)
  \end{itemize}

 % \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}[allowframebreaks]{Solución por MCG: el método de Cochrane--Orcutt}

\begin{block}{Idea: transformar el modelo para ``blanquear'' el error}
Si $\epsilon_t=\rho\epsilon_{t-1}+v_t$ con $v_t$ ruido blanco, una transformación cuasi-diferenciada elimina la autocorrelación y devuelve un modelo MCO válido (MCG factible).
\end{block}

\begin{exampleblock}{Derivación}
\textbf{Paso 1.} Modelo en $t$ y en $t-1$:
\[
y_t=\beta_0+\beta_1 x_t+\epsilon_t,\qquad y_{t-1}=\beta_0+\beta_1 x_{t-1}+\epsilon_{t-1}.
\]
\textbf{Paso 2.} Multiplicamos la segunda por $\rho$ y restamos:
\[
y_t-\rho y_{t-1}=\beta_0(1-\rho)+\beta_1(x_t-\rho x_{t-1})+(\epsilon_t-\rho\epsilon_{t-1}).
\]
\framebreak
\textbf{Paso 3.} Pero $\epsilon_t-\rho\epsilon_{t-1}=v_t$ es ruido blanco. Definiendo variables cuasi-diferenciadas $y_t^*=y_t-\rho y_{t-1}$, $x_t^*=x_t-\rho x_{t-1}$:
\[
y_t^*=\beta_0(1-\rho)+\beta_1 x_t^* + v_t.
\]
Este modelo \textbf{cumple Gauss--Markov} (error ruido blanco): MCO sobre las variables transformadas es eficiente.

\textbf{Paso 4 (factible: $\rho$ desconocido). Algoritmo Cochrane--Orcutt:}
\begin{enumerate}
  \item Estimar el modelo original por MCO; obtener residuos $\hat\epsilon_t$.
  \item Estimar $\hat\rho$ regresando $\hat\epsilon_t$ sobre $\hat\epsilon_{t-1}$.
  \item Cuasi-diferenciar con $\hat\rho$ y reestimar $\beta$.
  \item Iterar 1--3 hasta que $\hat\rho$ converja.
\end{enumerate}
\hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}{¿MCG (Cochrane--Orcutt) o HAC (Newey--West)?}

\begin{columns}
  \column{0.48\textwidth}
  \begin{block}{MCG / Cochrane--Orcutt}
  \begin{itemize}
    \item Corrige el \textbf{coeficiente} y recupera \textbf{eficiencia}.
    \item Requiere \emph{especificar} la forma de la autocorrelación (p.ej.\ AR(1)).
    \item Si la forma está mal, vuelve a fallar.
  \end{itemize}
  \end{block}

  \column{0.48\textwidth}
  \begin{alertblock}{HAC / Newey--West}
  \begin{itemize}
    \item No cambia $\hat\beta$; solo \textbf{corrige los errores estándar}.
    \item \textbf{No} exige conocer la forma de la dependencia.
    \item Robusto también a heterocedasticidad. Es la opción por defecto en la práctica.
  \end{itemize}
  \end{alertblock}
\end{columns}

\end{frame}

\section{Ejercicios}

\begin{frame}{Ejercicio Resuelto 1}

\begin{block}{Ejercicio 1 --- Breusch--Godfrey con datos del IMACEC}
Se estima por MCO un modelo del IMACEC con $T=200$ observaciones. La regresión auxiliar de B--G de \textbf{orden 1} (residuo sobre regresores y $\hat u_{t-1}$) arroja $R^2_{aux}=0.06$. Al 5\%, ¿hay autocorrelación de primer orden? Use $\chi^2_{1,0.95}=3.84$.
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Paso 1 (hipótesis):} $H_0:\rho_1=0$ (sin autocorrelación) vs.\ $H_1:\rho_1\neq0$.

\textbf{Paso 2 (estadístico):} $\text{LM}=nR^2_{aux}=200\times0.06=12.0$.

\textbf{Paso 3 (comparación):} $12.0 > 3.84=\chi^2_{1,0.95}$ $\Rightarrow$ \textbf{se rechaza} $H_0$.

\textbf{Paso 4 (conclusión):} hay evidencia de autocorrelación de primer orden. Los errores estándar de MCO no son válidos; corregir con Newey--West o Cochrane--Orcutt.
\end{exampleblock}

\end{frame}

\begin{frame}{Ejercicio Resuelto 2}

\begin{block}{Ejercicio 2 --- ACF de un error AR(1) (adaptado Certamen 3)}
El error de un modelo sigue $\epsilon_t=0.8\,\epsilon_{t-1}+v_t$, con $v_t$ ruido blanco. Calcule $\rho(1)$, $\rho(2)$ y $\rho(3)$ y describa el correlograma. Compárelo con un MA(1) $X_t=v_t+0.8\,v_{t-1}$, cuyo $\rho(1)=\theta/(1+\theta^2)$.
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Paso 1 (AR(1)):} por el resultado demostrado, $\rho(k)=\rho^k$ con $\rho=0.8$:
\[
\rho(1)=0.8,\quad \rho(2)=0.8^2=0.64,\quad \rho(3)=0.8^3=0.512.
\]
\textbf{Paso 2:} el correlograma del AR(1) \textbf{decae geométricamente} y nunca se corta.

\textbf{Paso 3 (MA(1)):} $\rho(1)=\dfrac{0.8}{1+0.8^2}=\dfrac{0.8}{1.64}\approx0.488$ y $\rho(k)=0$ para $k\ge2$: el correlograma se \textbf{corta} tras el primer rezago.

\textbf{Conclusión:} decaimiento gradual $\Rightarrow$ AR; corte abrupto $\Rightarrow$ MA. \hfill $\square$
\end{exampleblock}

\end{frame}

\begin{frame}[fragile]{Ejemplo Aplicado en R --- Detectar y corregir autocorrelación}

\begin{exampleblock}{Breusch--Godfrey, Durbin--Watson y errores HAC}
\begin{verbatim}
library(lmtest); library(sandwich)

# Modelo del IMACEC sobre el dólar (datos Chile)
m <- lm(imacec ~ dolar, data = datos)

# Deteccion
bgtest(m, order = 1)   # Breusch-Godfrey LM
dwtest(m)              # Durbin-Watson

# Correccion 1: errores HAC de Newey-West
coeftest(m, vcov = NeweyWest(m, lag = 1))

# Correccion 2: Cochrane-Orcutt (MCG factible)
library(orcutt); cochrane.orcutt(m)
\end{verbatim}
\end{exampleblock}

\end{frame}

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 21}
\begin{enumerate}
    \item La \textbf{autocorrelación serial} ($\text{Cov}(u_t,u_s)\neq0$) viola Gauss--Markov: $\Sigma$ deja de ser $\sigma^2 I$.
    \item Consecuencia: $\hat\beta_{MCO}$ sigue \textbf{insesgado y consistente}, pero los errores estándar usuales son \textbf{inválidos} (típicamente subestimados $\Rightarrow$ $t$ inflados).
    \item Detección: \textbf{Breusch--Godfrey} ($nR^2\sim\chi^2_p$, admite varios rezagos y dependiente rezagada) y \textbf{Durbin--Watson} (solo AR(1), $d\approx2(1-\hat\rho)$).
    \item Corrección eficiente: \textbf{MCG / Cochrane--Orcutt} (cuasi-diferenciar para blanquear el error), si conocemos la forma.
    \item Corrección robusta: \textbf{errores HAC de Newey--West}, sin especificar la forma de la dependencia.
\end{enumerate}
\end{block}
