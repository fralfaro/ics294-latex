\section{Ejercicios}
\begin{enumerate}
    \item Considere la siguiente regresión lineal múltiple estimada con \(n = 75\) observaciones:

\[
y \;=\; \beta_{0} + \beta_{1}x_{1} + \beta_{2}x_{2} + \beta_{3}x_{3} + \beta_{4}x_{4} + u.
\]

El modelo arroja un coeficiente de determinación
\[
R^2 = 0.62.
\]

El investigador sospecha que las variables \(x_{3}\) y \(x_{4}\) no aportan capacidad explicativa.  
Para contrastar su hipótesis estima el siguiente modelo:

\[
y \;=\; \gamma_{0} + \gamma_{1}x_{1} + \gamma_{2}x_{2} + e,
\qquad\text{obteniendo}\quad
R^2 = 0.55.
\]
Formule y realice un test de hipótesis que permita decidir, con un 5 \% de significancia, si las variables \(x_{3}\) y \(x_{4}\) son relevantes para el modelo. ($F_{0.95,(2,70)}=3.13$)



\item Considera la base de datos \emph{wagepanel} (hipotética) con $n=50$ trabajadores. Se modela:
\[
y_i=\ln(wage_i).
\]
Se estiman dos modelos por MCO:
\[
\underbrace{UR:\quad y_i=\beta_0+\beta_1\,educ_i+\beta_2\,exper_i+\beta_3\,tenure_i+u_i}_{k_{UR}=4}
\]
\[
\underbrace{R:\quad y_i=\beta_0+\beta_1\,educ_i+v_i}_{k_{R}=2}
\]
Queremos probar $H_0:\beta_2=\beta_3=0$ (es decir, que $exper$ y $tenure$ no aportan). Se dispone de:
\[
SST=200,\qquad R^2_{UR}=0.55,\qquad R^2_{R}=0.38.
\]
\begin{enumerate}
  \item Calcule $SSR_{UR}$ y $SSR_R$.
  \item Pruebe $H_0:\beta_2=\beta_3=0$ al 5\% con un $F$-test. Concluya en términos económicos.
  \item Calcule el $F$ de \emph{significancia global} del modelo $UR$ y concluya.
\end{enumerate}
\textbf{Hint: $F_{0.95,2,46}=3.20$, $F_{0.95,3,46}=2.81$}

\item Se estima por MCO el siguiente modelo con dependiente en logaritmo:
\[
\ln(wage_i)=\beta_0+\beta_1\,female_i+\beta_2\,educ_i+\beta_3\,(female_i\times educ_i)
+\beta_4\,parttime_i+\beta_5\,(female_i\times parttime_i)+u_i,
\]
donde $female=1$ si es mujer, $0$ si es hombre; $parttime=1$ si trabajo a tiempo parcial; $educ$ son años de educación. Suponga los siguientes coeficientes estimados (errores estándar omitidos):
\[
\widehat{\beta}_0=1.80,\quad \widehat{\beta}_1=-0.10,\quad \widehat{\beta}_2=0.070,\quad 
\widehat{\beta}_3=0.010,\quad \widehat{\beta}_4=-0.050,\quad \widehat{\beta}_5=-0.040.
\]
Responda:
\begin{enumerate}
  \item  Interprete $\widehat{\beta}_1$ y $\widehat{\beta}_4$ cuando las interacciones son cero (i.e., $educ$ dado, $parttime$ dado) y aclare el sentido de ``aditivo'' con dependiente en log.
  \item  Interprete $\widehat{\beta}_3$. ¿Cómo cambia el retorno de la educación entre mujeres y hombres?
  \item Calcule el efecto de ser mujer frente a ser hombre para un individuo \emph{full-time} ($parttime=0$) con $educ=12$. Dé la interpretación exacta en \%.
  \item Calcule el efecto de ser \emph{part-time} para hombres y para mujeres (a un $educ$ dado). Dé las interpretaciones exactas en \%.
  \item Escriba las ecuaciones de predicción por subgrupo (hombre/mujer; full-time/part-time) y señale explícitamente el \emph{cambio de pendiente} en educación y el \emph{cambio de nivel} por part-time.
\end{enumerate}


\end{enumerate}
