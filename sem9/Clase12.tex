\section{Introducción}
%==================================================================

\begin{frame}{Objetivos de la Clase}

Al finalizar esta clase, el estudiante será capaz de:

\begin{itemize}
    \item[\checkmark] Distinguir entre el modelo \textbf{restringido} y el \textbf{no restringido}, e identificar el número $q$ de restricciones impuestas por $H_0$.
    \item[\checkmark] Construir el \textbf{estadístico $F$} a partir de las sumas de cuadrados residuales (SCR) y, equivalentemente, a partir de los $R^2$.
    \item[\checkmark] Reconocer su distribución $F_{q,\,n-k-1}$ bajo $H_0$ (sketch de la demostración vía $\chi^2$ independientes).
    \item[\checkmark] Aplicar la prueba $F$ a \textbf{restricciones de exclusión} (significancia conjunta de un grupo de variables) y a la \textbf{significancia global} de la regresión.
    \item[\checkmark] Conectar la prueba $F$ con la prueba $t$ mediante la identidad $F = t^2$ cuando $q=1$.
\end{itemize}

\vspace{0.2cm}
\begin{alertblock}{Contexto}
La clase anterior probamos \textbf{una} restricción lineal (p.ej.\ $\beta_1=\beta_2$) usando un estadístico $t$. Hoy generalizamos a \textbf{varias} restricciones simultáneas: para eso necesitamos la prueba $F$.
\end{alertblock}

\end{frame}

\begin{frame}{¿Por qué no basta con la prueba $t$?}
\begin{itemize}
    \item La prueba $t$ contrasta \textbf{una sola} hipótesis (un coeficiente, o una combinación lineal como $\beta_1=\beta_2$).
    \pause
    \item Pero muchas preguntas económicas involucran \textbf{varios parámetros a la vez}:
    \begin{itemize}
        \item ¿Influyen \emph{conjuntamente} tres factores económicos sobre $y$, o ninguno importa?
        \item ¿Puedo excluir que el salario dependa de variables discriminatorias (género, edad, origen) \emph{en conjunto}?
        \item ¿El modelo \emph{completo} explica algo, o todos los $\beta$ (salvo la constante) son cero?
    \end{itemize}
    \pause
    \item Hacer muchas pruebas $t$ por separado \textbf{infla el error tipo I} y no responde la pregunta conjunta. La herramienta correcta es la prueba $F$.
\end{itemize}
\end{frame}

%==================================================================
\section{Modelo restringido vs no restringido}
%==================================================================

\begin{frame}{Modelo no restringido y modelo restringido}
Partimos del modelo de regresión lineal múltiple (\textbf{no restringido}, $nr$):
\[
    y_i = \beta_0 + \beta_1 x_{1i} + \beta_2 x_{2i} + \dots + \beta_k x_{ki} + u_i .
\]
\begin{itemize}
    \item La hipótesis nula impone $q$ \textbf{restricciones lineales} sobre los parámetros. Ejemplo (restricciones de exclusión):
    \[
        H_0:\ \beta_{k-q+1} = \beta_{k-q+2} = \dots = \beta_k = 0 .
    \]
    \pause
    \item Al imponer $H_0$ obtenemos el modelo \textbf{restringido} ($r$), con menos variables:
    \[
        y_i = \beta_0 + \beta_1 x_{1i} + \dots + \beta_{k-q} x_{(k-q)i} + u_i .
    \]
    \pause
    \item \textbf{Idea central:} si las restricciones son ciertas, quitar esas variables casi no debería empeorar el ajuste. Comparamos cuánto \emph{crece} la suma de cuadrados de los residuos al restringir.
\end{itemize}
\end{frame}

\begin{frame}{Lo que mide la prueba: ¿cuánto se pierde al restringir?}
\begin{itemize}
    \item Sea $\text{SCR}_{nr}$ la suma de cuadrados residuales del modelo no restringido y $\text{SCR}_r$ la del restringido.
    \pause
    \item Como el modelo restringido es un \textbf{caso particular} del no restringido (tiene menos libertad para ajustar), siempre:
    \[
        \text{SCR}_r \;\geq\; \text{SCR}_{nr}, \qquad\text{equivalentemente}\qquad R^2_r \;\leq\; R^2_{nr}.
    \]
    \pause
    \item La pregunta es si el \textbf{aumento} $\text{SCR}_r - \text{SCR}_{nr}$ es \emph{grande} (las variables importan, rechazo $H_0$) o \emph{pequeño} (compatible con $H_0$).
    \pause
    \item El estadístico $F$ \textbf{normaliza} ese aumento por el número de restricciones $q$ y por la varianza estimada del error del modelo no restringido.
\end{itemize}
\end{frame}

%==================================================================
\section{El estadístico F}
%==================================================================

\begin{frame}{El estadístico $F$ de restricciones múltiples}
\begin{tcolorbox}[colback=blue!5!white, colframe=blue!75!black, title=Estadístico $F$ (forma SCR)]
\[
   F = \frac{(\text{SCR}_r - \text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} \;\underset{H_0}{\sim}\; F_{q,\,n-k-1}.
\]
\end{tcolorbox}
\begin{itemize}
    \item $q$ = número de restricciones (grados de libertad del numerador).
    \item $n-k-1$ = grados de libertad del modelo \textbf{no restringido} (denominador).
    \pause
    \item El numerador mide la \emph{pérdida de ajuste por restricción}; el denominador es $\hat{\sigma}^2_{nr}$, la varianza del error sin restringir.
    \pause
    \item $F \geq 0$ siempre. Se rechaza $H_0$ si $F$ es \textbf{grande}: $F > F_{1-\alpha,\,q,\,n-k-1}$ (prueba de una cola, cola derecha).
\end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Demostración (sketch): por qué $F \sim F_{q,\,n-k-1}$}

\begin{alertblock}{Teorema}
Bajo los supuestos clásicos del modelo lineal con errores normales $u\mid \mathbf{X}\sim N(0,\sigma^2 I_n)$ y bajo $H_0$ (las $q$ restricciones son ciertas),
\[
   F = \frac{(\text{SCR}_r - \text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} \sim F_{q,\,n-k-1}.
\]
\end{alertblock}

\begin{exampleblock}{Demostración (sketch)}
\textbf{Paso 1 (definición de la $F$).} Una variable $F$ con $(q, m)$ grados de libertad se construye como el cociente de dos $\chi^2$ \emph{independientes}, cada una dividida por sus grados de libertad:
\[
   F_{q,m} = \frac{\chi^2_q / q}{\chi^2_m / m}.
\]
Debemos identificar dos $\chi^2$ independientes en nuestro estadístico.

\framebreak

\textbf{Paso 2 (la suma residual no restringida).} Bajo normalidad de los errores, un resultado clásico (teorema de Cochran) establece que
\[
   \frac{\text{SCR}_{nr}}{\sigma^2} \sim \chi^2_{\,n-k-1}.
\]
Este es el denominador: $\dfrac{\text{SCR}_{nr}/\sigma^2}{n-k-1}$ es una $\chi^2$ dividida por sus g.l.

\framebreak

\textbf{Paso 3 (el incremento de la suma residual).} El término clave del numerador, $\text{SCR}_r - \text{SCR}_{nr}$, puede escribirse como una forma cuadrática en el vector de error proyectado sobre el subespacio de dimensión $q$ que se ``elimina'' al imponer $H_0$. Bajo $H_0$, esa forma cuadrática satisface
\[
   \frac{\text{SCR}_r - \text{SCR}_{nr}}{\sigma^2} \sim \chi^2_{\,q}.
\]
Intuición: el modelo restringido ``pierde'' $q$ dimensiones de ajuste; bajo $H_0$ esa pérdida es puro ruido y aporta exactamente $q$ grados de libertad.

\framebreak

\textbf{Paso 4 (independencia).} Las dos formas cuadráticas viven en subespacios \emph{ortogonales} (el espacio de proyección y su complemento), y para vectores normales esto implica \textbf{independencia} entre $\text{SCR}_r-\text{SCR}_{nr}$ y $\text{SCR}_{nr}$.

\framebreak

\textbf{Paso 5 (armar el cociente).} Tomamos el cociente de las dos $\chi^2$ independientes, cada una dividida por sus grados de libertad. Nótese que $\sigma^2$ \textbf{se cancela}:
\[
   F = \frac{\big[(\text{SCR}_r - \text{SCR}_{nr})/\sigma^2\big]/q}{\big[\text{SCR}_{nr}/\sigma^2\big]/(n-k-1)}
     = \frac{(\text{SCR}_r - \text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} \sim F_{q,\,n-k-1}.
\]
Esto es justamente el estadístico observable (no depende de $\sigma^2$, que es desconocido). \hfill $\square$
\end{exampleblock}
\end{frame}

\begin{frame}{Forma equivalente con $R^2$}
A partir de $R^2 = 1 - \text{SCR}/\text{SCT}$ se tiene $\text{SCR} = (1-R^2)\,\text{SCT}$. Sustituyendo en el estadístico (la $\text{SCT}$ es la \emph{misma} en ambos modelos y se cancela):

\begin{tcolorbox}[colback=blue!5!white, colframe=blue!75!black, title=Estadístico $F$ (forma $R^2$)]
\[
   F = \frac{(R^2_{nr} - R^2_{r})/q}{(1 - R^2_{nr})/(n-k-1)} \;\underset{H_0}{\sim}\; F_{q,\,n-k-1}.
\]
\end{tcolorbox}

\begin{itemize}
    \item \textbf{Muy útil:} no requiere las SCR, solo los $R^2$ de ambos modelos.
    \pause
    \item \textbf{Cuidado:} la forma con $R^2$ \emph{solo} es válida cuando la variable dependiente es la \textbf{misma} en ambos modelos (igual $\text{SCT}$). No sirve, por ejemplo, si un modelo usa $y$ y el otro $\log(y)$.
    \pause
    \item Si $R^2_{nr} = R^2_{r}$ (restringir no cambia el ajuste), entonces $F=0$: ninguna evidencia contra $H_0$.
\end{itemize}
\end{frame}

\begin{frame}{Región de rechazo de la prueba $F$}
\begin{figure}
\centering
\begin{tikzpicture}[scale=1.15]
  \draw[->] (-0.2,0) -- (4.2,0) node[right] {\scriptsize $F$};
  \draw[->] (0,-0.1) -- (0,2.2) node[above] {\scriptsize $f(F)$};
  % Densidad tipo F (asimétrica a la derecha): aproximación suave
  \draw[domain=0.02:4, smooth, thick, blue, samples=120]
    plot (\x, {3.2*(\x)^(0.5)*exp(-1.1*\x)});
  % Valor crítico
  \draw[dashed] (2.6,0) -- (2.6,0.55);
  \node[below] at (2.6,0) {\scriptsize $F_{1-\alpha,\,q,\,n-k-1}$};
  % Zona de rechazo (cola derecha)
  \fill[red!30, domain=2.6:4, variable=\x]
    (2.6,0) -- plot (\x, {3.2*(\x)^(0.5)*exp(-1.1*\x)}) -- (4,0) -- cycle;
  \node[red] at (3.2,0.55) {\scriptsize Rechazo $H_0$};
  \node[blue] at (1.0,1.3) {\scriptsize No rechazo};
\end{tikzpicture}
\end{figure}
{\small La distribución $F$ vive en $[0,\infty)$ y es asimétrica. La prueba es \textbf{siempre de una cola derecha}: valores grandes de $F$ indican que restringir empeora mucho el ajuste $\Rightarrow$ las restricciones no son creíbles.}
\end{frame}

%==================================================================
\section{Restricciones de exclusión}
%==================================================================

\begin{frame}{Restricciones de exclusión (significancia conjunta)}
El caso más frecuente: probar que un \textbf{grupo} de $q$ variables, \emph{tomadas en conjunto}, no aportan al modelo.
\[
   H_0:\ \beta_{k-q+1} = \dots = \beta_k = 0 \qquad\text{vs}\qquad H_1:\ \text{al menos uno} \neq 0.
\]
\begin{itemize}
    \item \textbf{Modelo no restringido:} incluye todas las $k$ variables $\to$ $R^2_{nr}$.
    \pause
    \item \textbf{Modelo restringido:} elimina las $q$ variables bajo prueba $\to$ $R^2_{r}$.
    \pause
    \item Estadístico: $F = \dfrac{(R^2_{nr}-R^2_r)/q}{(1-R^2_{nr})/(n-k-1)} \sim F_{q,\,n-k-1}$.
    \pause
    \item \textbf{Ojo:} un grupo puede ser \textbf{conjuntamente significativo} aunque \emph{ninguna} variable lo sea individualmente (típico bajo multicolinealidad). Por eso la prueba $F$ no se reemplaza por varias pruebas $t$.
\end{itemize}
\end{frame}

%==================================================================
\section{Significancia global}
%==================================================================

\begin{frame}{Significancia global de la regresión}
Caso especial de exclusión: ¿\emph{algún} regresor explica $y$? Excluimos \textbf{todos} salvo la constante, de modo que $q=k$.
\[
   H_0:\ \beta_1 = \beta_2 = \dots = \beta_k = 0 \qquad\text{vs}\qquad H_1:\ \text{al menos uno}\neq 0.
\]
\begin{itemize}
    \item El modelo restringido es $y_i = \beta_0 + u_i$, cuyo $R^2_r = 0$.
    \pause
    \item Sustituyendo $R^2_r=0$ y $q=k$ en la forma con $R^2$:
\end{itemize}
\begin{tcolorbox}[colback=blue!5!white, colframe=blue!75!black, title=$F$ de significancia global]
\[
   F = \frac{R^2/k}{(1-R^2)/(n-k-1)} \;\underset{H_0}{\sim}\; F_{k,\,n-k-1}.
\]
\end{tcolorbox}
{\small Es el ``$F$ statistic'' que reportan por defecto \texttt{lm} (R) y \texttt{statsmodels} (Python).}
\end{frame}

%==================================================================
\section{Relación con la prueba t}
%==================================================================

\begin{frame}{Puente con la clase anterior: $F = t^2$ cuando $q=1$}
\begin{alertblock}{Proposición}
Si la prueba $F$ contrasta \textbf{una sola} restricción ($q=1$), entonces
\[
   F = t^2,
\]
donde $t$ es el estadístico de la prueba $t$ correspondiente, y $F_{1,\,m} = (t_m)^2$.
\end{alertblock}

\begin{exampleblock}{Justificación}
Una $F_{1,m}$ es, por definición, el cociente $\dfrac{\chi^2_1/1}{\chi^2_m/m}$. Como una variable $t$ con $m$ g.l.\ es $t_m = \dfrac{Z}{\sqrt{\chi^2_m/m}}$ con $Z\sim N(0,1)$, al elevar al cuadrado:
\[
   t_m^2 = \frac{Z^2}{\chi^2_m/m} = \frac{\chi^2_1/1}{\chi^2_m/m} = F_{1,m}.
\]
\hfill $\square$
\end{exampleblock}

\begin{itemize}
    \item \textbf{Consecuencia:} para una hipótesis simple, $t$ y $F$ dan \emph{idéntica} decisión. Con $q\geq 2$ la prueba $t$ ya no aplica y \textbf{solo} la $F$ responde la pregunta conjunta.
\end{itemize}
\end{frame}

%==================================================================
\section{Ejemplo aplicado}
%==================================================================

\begin{frame}[fragile]{Ejemplo aplicado: prueba $F$ en R}
\begin{exampleblock}{Significancia conjunta con datos de salarios (Chile)}
\begin{verbatim}
datos <- read.csv("salarios_chile.csv")

# Modelo no restringido (todas las variables)
nr <- lm(log(salario) ~ educ + exper + tenure
                       + female + sindicato, data = datos)

# Modelo restringido (sin tenure, female, sindicato)
r  <- lm(log(salario) ~ educ + exper, data = datos)

# Prueba F de las 3 restricciones de exclusion
anova(r, nr)               # compara SCR_r vs SCR_nr

# F de significancia global del modelo no restringido
summary(nr)                # ultima linea: F-statistic
\end{verbatim}
\end{exampleblock}
{\small \texttt{anova(r, nr)} entrega el $F$ y su $p$-valor; un $p < \alpha$ rechaza $H_0$: el grupo es conjuntamente significativo.}
\end{frame}

\begin{frame}[fragile]{Ejemplo aplicado: prueba $F$ en Python}
\begin{exampleblock}{statsmodels}
\begin{verbatim}
import statsmodels.formula.api as smf

nr = smf.ols("np.log(salario) ~ educ + exper + tenure"
             " + female + sindicato", data=datos).fit()

# Prueba F de exclusion conjunta (q = 3)
hip = ["tenure = 0", "female = 0", "sindicato = 0"]
print(nr.f_test(hip))      # F, p-valor, gl

# F de significancia global y su p-valor
print(nr.fvalue, nr.f_pvalue)
\end{verbatim}
\end{exampleblock}
{\small \texttt{f\_test} admite cualquier conjunto de restricciones lineales; \texttt{fvalue} es el $F$ global que reporta \texttt{summary()}.}
\end{frame}

\section{Ejercicios resueltos}
%==================================================================

\begin{frame}{Ejercicio Resuelto 1}

\begin{block}{Ejercicio 1 (adaptado de Certamen 2, T2)}
Con $n=206$ trabajadores de la RM se estima
\[ \log(\text{salario}) = \beta_0 + \beta_1\text{educ} + \beta_2\text{exp} + \beta_3\text{tenure} + \beta_4\text{female} + \beta_5\text{sindicato} + u, \]
con $R^2_{nr}=0{,}78$. El investigador sospecha que basta con $\text{educ}$ y $\text{exp}$.
\begin{enumerate}
    \item Plantee $H_0$ y $H_1$ e indique $q$.
    \item Si el modelo solo con $\text{educ}$ y $\text{exp}$ da $R^2_r=0{,}68$, contraste al $5\%$ ($F_{0{,}95;\,3;\,200}=2{,}65$).
\end{enumerate}
\end{block}

\pause

\begin{exampleblock}{Solución (1)}
Restricciones de exclusión sobre las tres últimas variables:
\[ H_0:\ \beta_3=\beta_4=\beta_5=0 \quad\text{vs}\quad H_1:\ \text{al menos uno}\neq 0,\qquad q=3. \]
$k=5$, $n=206 \Rightarrow n-k-1 = 200$.
\end{exampleblock}

\end{frame}

\begin{frame}{Ejercicio Resuelto 1 (continuación)}
\begin{exampleblock}{Solución (2): contraste al $5\%$}
Usamos la forma con $R^2$:
\[
   F = \frac{(R^2_{nr}-R^2_r)/q}{(1-R^2_{nr})/(n-k-1)}
     = \frac{(0{,}78-0{,}68)/3}{(1-0{,}78)/200}
     = \frac{0{,}10}{0{,}22}\cdot\frac{200}{3} \approx 30{,}30.
\]
\textbf{Distribución bajo $H_0$:} $F \sim F_{3,\,200}$. \textbf{Valor crítico:} $F_{0{,}95;\,3;\,200}=2{,}65$.

\medskip
Como $F = 30{,}30 > 2{,}65$, \textbf{se rechaza} $H_0$.

\medskip
\textbf{Conclusión:} las variables $\text{tenure}$, $\text{female}$ y $\text{sindicato}$ son \emph{conjuntamente} relevantes; la propuesta de quedarse solo con $\text{educ}$ y $\text{exp}$ no es estadísticamente adecuada.
\end{exampleblock}
\end{frame}

\begin{frame}{Ejercicio Resuelto 2}
\begin{block}{Ejercicio 2 (significancia global — datos voteA, Wooldridge)}
Con $n=173$ se estima la cuota de votación $\text{voteA}$ sobre $\text{lexpendA}$, $\text{lexpendB}$ y $\text{prtystrA}$ ($k=3$), obteniendo $R^2=0{,}793$. Lleve a cabo la prueba de \textbf{significancia global} al $5\%$.
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{Hipótesis:} $H_0:\beta_1=\beta_2=\beta_3=0$ vs $H_1:$ al menos uno $\neq 0$; aquí $q=k=3$, $n-k-1=169$.
\[
   F = \frac{R^2/k}{(1-R^2)/(n-k-1)} = \frac{0{,}793/3}{(1-0{,}793)/169} = \frac{0{,}2643}{0{,}001225} \approx 215{,}8.
\]
\textbf{Distribución:} $F\sim F_{3,\,169}$; valor crítico $F_{0{,}95;\,3;\,169}\approx 2{,}66$.

\medskip
Como $F=215{,}8 \gg 2{,}66$, \textbf{se rechaza} $H_0$ a cualquier nivel usual: el modelo es globalmente significativo.
\end{exampleblock}
\end{frame}

\begin{frame}{Ejercicio Resuelto 3}
\begin{block}{Ejercicio 3 (forma SCR y la identidad $F=t^2$)}
Un modelo con $k=4$ regresores y $n=80$ tiene $\text{SCR}_{nr}=150$. Al imponer una \textbf{única} restricción ($\beta_4=0$) el modelo restringido da $\text{SCR}_r=160$.
\begin{enumerate}
    \item Calcule el estadístico $F$.
    \item Verifique que coincide con $t^2$, donde $t$ es el estadístico de la prueba $t$ de $\beta_4=0$.
\end{enumerate}
\end{block}

\pause

\begin{exampleblock}{Solución}
\textbf{(1)} Aquí $q=1$ y $n-k-1 = 80-4-1 = 75$:
\[
   F = \frac{(\text{SCR}_r-\text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} = \frac{(160-150)/1}{150/75} = \frac{10}{2} = 5.
\]
\textbf{(2)} Por la proposición $F=t^2$ con $q=1$, el estadístico $t$ de $\beta_4=0$ cumple $t=\pm\sqrt{5}=\pm 2{,}236$. La prueba $F$ (cola derecha) y la $t$ (dos colas) dan exactamente la \emph{misma} decisión: a $5\%$, $F_{0{,}95;\,1;\,75}\approx 3{,}97 = (1{,}992)^2 = t^2_{0{,}975;\,75}$. Como $5>3{,}97$, se rechaza $H_0$.
\end{exampleblock}
\end{frame}

\section{Ejercicios propuestos}

\begin{frame}{Ejercicios Propuestos}

\begin{block}{Ejercicio 4 (Propuesto — rutinario)}
Un modelo con $k=5$ regresores y $n=100$ tiene $R^2_{nr}=0{,}62$. Al excluir $q=2$ variables, el modelo restringido da $R^2_r=0{,}55$. Contraste al $5\%$ la significancia conjunta de esas dos variables.
\end{block}
{\small \textit{Indicación: $F=\dfrac{(R^2_{nr}-R^2_r)/q}{(1-R^2_{nr})/(n-k-1)}\sim F_{2,\,94}$; valor crítico $F_{0{,}95;\,2;\,94}\approx 3{,}09$.}}

\begin{block}{Ejercicio 5 (Propuesto — intermedio)}

\section{Resumen}
%==================================================================

\begin{frame}{Resumen}

\begin{block}{Puntos Clave de la Clase 12}
\begin{enumerate}
    \item Para probar \textbf{varias} restricciones a la vez ($q\geq 2$) usamos la prueba $F$, no la $t$.
    \item Comparamos el modelo \textbf{restringido} ($\text{SCR}_r$, $R^2_r$) con el \textbf{no restringido} ($\text{SCR}_{nr}$, $R^2_{nr}$): siempre $\text{SCR}_r\geq\text{SCR}_{nr}$.
    \item $F=\dfrac{(\text{SCR}_r-\text{SCR}_{nr})/q}{\text{SCR}_{nr}/(n-k-1)} = \dfrac{(R^2_{nr}-R^2_r)/q}{(1-R^2_{nr})/(n-k-1)} \sim F_{q,\,n-k-1}$ bajo $H_0$ (cola derecha).
    \item \textbf{Significancia global}: $H_0:\beta_1=\dots=\beta_k=0 \Rightarrow F=\dfrac{R^2/k}{(1-R^2)/(n-k-1)}$.
    \item Con $q=1$ se recupera la prueba $t$ vía $F=t^2$ (puente con la clase anterior).
\end{enumerate}
\end{block}

\begin{alertblock}{Para la próxima clase}
Variables artificiales (\emph{dummies}) e interacciones: cómo modelar diferencias de grupo y cómo la prueba $F$ contrasta su significancia conjunta (test de Chow).
\end{alertblock}

\end{frame}
